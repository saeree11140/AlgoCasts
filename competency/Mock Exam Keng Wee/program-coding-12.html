ให้ statuses เป็นผล request ตามลำดับเวลา 1 หมายถึงสำเร็จ 0 หมายถึงล้มเหลว

วิธีคิดทีละขั้น

  1. ตั้ง count, run และ best เป็น 0 เพราะยังไม่อ่าน request
  2. อ่านทีละค่า: ถ้าเป็น 1 เพิ่ม count และ run; ถ้าเป็น 0 เริ่ม run ใหม่ที่ 0

  4. คืน count และ best หลังอ่านครบ

ไล่ตัวอย่าง

  1. [1,1] ทำให้ count=2, run=2, best=2
  2. เจอ 0: run=0 แต่ count และ best คงเดิม
  3. ช่วง [1,1,1] ต่อมาได้ count=5 และ best=3

เหตุผลที่วิธีนี้ถูก

หลังอ่านแต่ละตำแหน่ง run คือจำนวน 1 ติดต่อกันที่ลงท้ายตรงนี้ ส่วน best คือค่าสูงสุดในทุกตำแหน่งที่อ่านแล้ว จึงครอบคลุมทุกช่วงสำเร็จ

เทียบวิธีตรงไปตรงมา

ไล่ทุกช่วงแล้วเช็กว่าเป็น 1 ทั้งหมดทำงานซ้ำมาก ใช้ตัวนับช่วงเดียวพอ

Time: O(n)
Extra space: O(1)

Reference TypeScript

export function summarizeRequests(statuses: number[]): { successCount: number; longestRun: number } {
  let successCount = 0;
  let longestRun = 0;
  let run = 0;
  for (const status of statuses) {
    if (status === 1) {
      successCount++;
      run++;
      longestRun = Math.max(longestRun, run);
    } else {
      run = 0;
    }
  }
  return { successCount, longestRun };
}

export function firstUniqueCode(codes: string): string | null {
  const counts = new Array<number>(26).fill(0);
  for (const c of codes) counts[c.charCodeAt(0) - 97]++;
  for (const c of codes) {
    if (counts[c.charCodeAt(0) - 97] === 1) return c;
  }
  return null;
}

เหตุผลที่วิธีนี้ถูก

เทียบวิธีตรงไปตรงมา

nested loops ตรวจทุกคู่เป็น O(n²) จำนวน 200000 รายการมีเกือบ 2 หมื่นล้านคู่

Time: expected O(n)
Extra space: O(u) โดย u คือจำนวนค่าต่างกัน

Reference TypeScript

export function countTargetPairs(amounts: number[], target: number): number {
  const seen = new Map<number, number>();
  let answer = 0;
  for (const value of amounts) {
    // Only earlier indices are eligible; insert after counting.
    answer += seen.get(target - value) ?? 0;
    seen.set(value, (seen.get(value) ?? 0) + 1);
  }
  return answer;
}

  2. เก็บ x ก่อน lookup แล้วจับตัวเอง

sort อักษรแต่ละคำเพื่อเป็น key ก็ถูก แต่ frequency key มีต้นทุน O(length+26) และไม่ต้องเปลี่ยนลำดับคำ

export function groupCodes(codes: string[]): string[][] {
  const groups: string[][] = [];
  const indexByKey = new Map<string, number>();
  for (const code of codes) {
    const counts = new Array<number>(26).fill(0);
    for (const c of code) counts[c.charCodeAt(0) - 97]++;
    const key = counts.join(",");
    let index = indexByKey.get(key);
    if (index === undefined) {
      index = groups.length;
      indexByKey.set(key, index);
      groups.push([]);
    }
    groups[index].push(code);
  }
  return groups;
}

export function countAffordablePairs(prices: number[], limit: number): number {
  const sorted = [...prices].sort((a, b) => a - b);
  let left = 0;
  let right = sorted.length - 1;
  let answer = 0;
  while (left < right) {
    if (sorted[left] + sorted[right] <= limit) {
      answer += right - left;
      left++;
    } else {
      right--;
    }
  }
  return answer;
}

export function isValidConfiguration(text: string): boolean {
  const stack: string[] = [];
  const opener: Record<string, string> = { ")": "(", "]": "[", "}": "{" };
  for (const c of text) {
    if (c === "(" || c === "[" || c === "{") stack.push(c);
    else if (stack.pop() !== opener[c]) return false;
  }
  return stack.length === 0;
}

export function peakRequests(timestamps: number[], duration: number): number {
  let left = 0;
  let answer = 0;
  for (let right = 0; right < timestamps.length; right++) {
    while (timestamps[right] - timestamps[left] >= duration) left++;
    answer = Math.max(answer, right - left + 1);
  }
  return answer;
}

export function minimumCapacity(weights: number[], days: number): number {
  let low = 0;
  let high = 0;
  for (const weight of weights) {
    low = Math.max(low, weight);
    high += weight;
  }
  const feasible = (capacity: number): boolean => {
    let usedDays = 1;
    let load = 0;
    for (const weight of weights) {
      if (load + weight > capacity) {
        usedDays++;
        load = 0;
      }
      load += weight;
      if (usedDays > days) return false;
    }
    return true;
  };
  while (low < high) {
    const mid = low + Math.floor((high - low) / 2);
    if (feasible(mid)) high = mid;
    else low = mid + 1;
  }
  return low;
}

  3. query [2,1] ใช้ (7-1)+1=7; [3,3] ใช้ 7-6=1

เหตุผลที่วิธีนี้ถูก

prefix[b]-prefix[a] คือผลรวม index a..b-1 ช่วงวนแยกเป็นสองส่วนที่ไม่ทับกัน จึงรวมครบโดยไม่มีวันซ้ำ

เทียบวิธีตรงไปตรงมา

วนรวมแต่ละ query เป็น O(nq); prefix ใช้ O(n) เตรียมแล้ว O(1) ต่อ query

Time: O(n+q)
Extra space: O(n) นอก output ของ q queries

Reference TypeScript

export function circularTotals(values: number[], queries: [number, number][]): number[] {
  const prefix = new Array<number>(values.length + 1).fill(0);
  for (let i = 0; i < values.length; i++) prefix[i + 1] = prefix[i] + values[i];
  return queries.map(([left, right]) => {
    if (left <= right) return prefix[right + 1] - prefix[left];
    return prefix[values.length] - prefix[left] + prefix[right + 1];
  });
}

จุดพลาด

  1. รวมต่อ query O(n), ลืม +1, l=r ถูกตีเป็นเต็มรอบ

export function minimumHops(n: number, edges: [number, number][], source: number, target: number): number {
  if (source === target) return 0;
  const graph: number[][] = Array.from({ length: n }, () => []);
  for (const [u, v] of edges) {
    graph[u].push(v);
    graph[v].push(u);
  }
  const distance = new Array<number>(n).fill(-1);
  distance[source] = 0;
  const queue = [source];
  for (let head = 0; head < queue.length; head++) {
    const u = queue[head];
    for (const v of graph[u]) {
      if (distance[v] !== -1) continue;
      distance[v] = distance[u] + 1;
      if (v === target) return distance[v];
      queue.push(v);
    }
  }
  return -1;
}

export function maximumReward(rewards: number[]): number {
  let twoDaysBack = 0;
  let yesterday = 0;
  for (const reward of rewards) {
    const today = Math.max(yesterday, twoDaysBack + reward);
    twoDaysBack = yesterday;
    yesterday = today;
  }
  return yesterday;
}

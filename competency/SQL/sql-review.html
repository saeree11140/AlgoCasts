  customer_id integer PRIMARY KEY,
  name        text NOT NULL,
  city        text,
  email       text UNIQUE
);
CREATE TABLE products (
  product_id   integer PRIMARY KEY,
  product_name text NOT NULL,
  category     text NOT NULL,
  price        numeric(10,2) NOT NULL CHECK (price >= 0)
);
CREATE TABLE orders (
  order_id    integer PRIMARY KEY,
  customer_id integer NOT NULL REFERENCES customers(customer_id),
  order_date  date NOT NULL,
  status      text NOT NULL CHECK (status IN ('pending','paid','shipped','cancelled'))
);
CREATE TABLE order_items (
  order_id   integer REFERENCES orders(order_id),
  product_id integer REFERENCES products(product_id),
  qty        integer NOT NULL CHECK (qty > 0),
  unit_price numeric(10,2) NOT NULL,
  PRIMARY KEY (order_id, product_id)
);

Table, Row, Column และ Key

ฐานข้อมูลเชิงสัมพันธ์ (relational database) เก็บข้อมูลเป็นตาราง แต่ละตารางพูดถึงเรื่องเดียว เช่น customers พูดถึงลูกค้า orders พูดถึงการสั่งซื้อ แต่ละแถว (row) คือข้อมูลหนึ่งรายการ แต่ละคอลัมน์ (column) คือคุณสมบัติหนึ่งอย่าง และมีชนิดข้อมูลกำกับเสมอ ส่วนตารางต่าง ๆ เชื่อมกันด้วยค่าที่ตรงกัน ไม่ได้ฝังข้อมูลซ้อนกันเหมือนเอกสาร JSON

ตัวที่ทำให้ตารางเป็น relational จริง ๆ คือ key ซึ่งมีหลายชนิดที่ข้อสอบชอบถามแยกกัน

คำความหมายตัวอย่างในฐานข้อมูลนี้
Primary key (PK)ค่าที่ระบุแถวได้แถวเดียว ห้ามซ้ำ ห้าม NULL มีได้ตารางละหนึ่งชุดcustomers.customer_id

เรื่อง surrogate กับ natural key มีข้อถกเถียงอยู่ แต่ในงานจริงคนส่วนใหญ่เลือก surrogate เป็น PK แล้วใส่ UNIQUE ให้ natural key อีกชั้น เหตุผลคืออีเมลเปลี่ยนได้ และถ้า PK เปลี่ยน ทุกตารางที่อ้างมาถึงก็ต้องตามแก้ ส่วนเลขบัตรประชาชนเป็นข้อมูลส่วนบุคคลที่ไม่ควรไปปรากฏเป็น FK อยู่ทั่วทุกตาราง

ความสัมพันธ์ระหว่างตารางมีสามแบบ แบบ one-to-many ทำด้วย FK ฝั่ง "many" เช่น ลูกค้าหนึ่งคนมีหลายออเดอร์ แบบ many-to-many ต้องมีตารางกลาง (junction table) เพราะออเดอร์หนึ่งมีหลายสินค้า และสินค้าหนึ่งอยู่ในหลายออเดอร์ ตาราง order_items ทำหน้าที่นี้ และแบบ one-to-one ใช้น้อย มักเกิดตอนแยกข้อมูลที่ไม่ค่อยใช้ออกจากตารางหลัก

ตารางนี้ใช้งานได้ แต่เกิดปัญหาสามอย่างที่ตำราเรียกว่า anomaly ถ้า Arisa ย้ายเมือง ต้องแก้ทุกแถวของเธอ (update anomaly) ถ้าอยากเพิ่มสินค้าใหม่ที่ยังไม่มีใครสั่ง ทำไม่ได้เพราะ order_id เป็นส่วนของ PK จะว่างไม่ได้ (insert anomaly) และถ้าลบออเดอร์สุดท้ายที่มี Laptop ข้อมูลของ Laptop หายไปด้วย (delete anomaly) Normalization คือขั้นตอนแยกตารางเพื่อกำจัด anomaly เหล่านี้ โดยไล่ไปตามระดับที่เรียกว่า normal form

1NF ทุกช่องต้องเก็บค่าเดียว และห้ามมีกลุ่มคอลัมน์ซ้ำ ถ้ามีคอลัมน์ products ที่เก็บ 'Keyboard, Mouse' ไว้ในช่องเดียว ก็ผิด 1NF เพราะค้นหา นับ หรือ JOIN ไม่ได้โดยไม่ต้อง parse string การแก้คือหนึ่งสินค้าหนึ่งแถว เหมือนตารางด้านบน

2NF ต้องผ่าน 1NF และคอลัมน์ที่ไม่ใช่ key ทุกตัวต้องขึ้นกับ key ทั้งก้อน ไม่ใช่แค่บางส่วน ข้อนี้มีความหมายเมื่อ PK เป็น composite ในตารางข้างบน customer_name ขึ้นกับ order_id อย่างเดียว และ product_name ขึ้นกับ product_id อย่างเดียว ซึ่งเรียกว่า partial dependency ต้องแยกออกไปเป็นตาราง orders กับ products

3NF ต้องผ่าน 2NF และคอลัมน์ที่ไม่ใช่ key ต้องไม่ขึ้นกับคอลัมน์อื่นที่ไม่ใช่ key (transitive dependency) ถ้าตาราง employees เก็บ dept_id และ dept_name ไว้คู่กัน dept_name ขึ้นกับ dept_id ไม่ได้ขึ้นกับ emp_id โดยตรง ต้องแยก departments ออกมา ประโยคที่คนใช้จำกันคือทุกคอลัมน์ต้องขึ้นกับ "key, ทั้ง key, และไม่มีอะไรนอกจาก key"

BCNF เป็นรุ่นที่เข้มกว่า 3NF นิดหน่อย กฎคือทุกคอลัมน์ที่เป็นตัวกำหนดค่าของคอลัมน์อื่น (determinant) ต้องเป็น candidate key ในงานปกติตาราง 3NF ส่วนใหญ่ผ่าน BCNF อยู่แล้ว ข้อสอบระดับนี้มักถามแค่ 1NF ถึง 3NF

พอแยกตามนี้ครบ ก็จะได้ schema ชุดเดียวกับที่ใช้ในหน้านี้ ลูกค้าอยู่ที่เดียวใน customers สินค้าอยู่ที่เดียวใน products และ order_items เก็บเฉพาะว่าออเดอร์ไหนซื้ออะไร กี่ชิ้น ย้ายเมืองของ Arisa ก็แก้แถวเดียว

จำให้ติดมือNormalization ลดข้อมูลซ้ำและป้องกัน anomaly แลกกับการต้อง JOIN มากขึ้น ถ้าข้อสอบถามว่า "คอลัมน์นี้ผิด normal form ข้อไหน" ให้ถามตัวเองว่าคอลัมน์นั้นขึ้นกับอะไร ถ้าขึ้นกับ key บางส่วนคือ 2NF ถ้าขึ้นกับคอลัมน์ธรรมดาอีกตัวคือ 3NF ถ้าช่องเดียวมีหลายค่าคือ 1NF

Denormalization ทำเมื่อไร

บางครั้งเราตั้งใจเก็บข้อมูลซ้ำ เช่น ตารางสรุปยอดขายรายวันสำหรับ dashboard ที่ต้องอ่านเร็ว หรือคอลัมน์ order_count ในตารางลูกค้าเพื่อไม่ต้องนับทุกครั้ง เหตุผลคือ query ที่ JOIN หลายตารางกับข้อมูลก้อนใหญ่ช้าเกินไป แต่ต้นทุนคือต้องมีกลไกรักษาให้ข้อมูลตรงกันเอง เช่น trigger, job ที่คำนวณใหม่, หรือ materialized view ตัวเลือกนี้ควรมาหลังจากวัดแล้วว่า schema แบบ normalize ช้าจริง ไม่ใช่ทำไว้ก่อนเพราะกลัวว่าจะช้า

มีกรณีที่ดูเหมือนข้อมูลซ้ำแต่ไม่ใช่ order_items.unit_price เก็บราคาตอนที่ลูกค้าซื้อ ส่วน products.price คือราคาปัจจุบัน สองค่านี้มีความหมายต่างกัน ถ้าตัดคอลัมน์ unit_price ออกเพราะคิดว่าซ้ำ พอสินค้าขึ้นราคา ยอดของออเดอร์เก่าจะเปลี่ยนตามไปด้วย

ข้อผิดพลาดที่เจอบ่อย

  • เก็บหลายค่าในช่องเดียวด้วย comma เช่น tags = 'sql,index,join' แล้วค้นด้วย LIKE '%join%' ซึ่งช้า และจับผิดตัวได้ง่าย (เช่น 'disjoint') ควรทำตาราง post_tags
  • ใช้ชื่อหรืออีเมลเป็น FK แทน id พอแก้ชื่อทีเดียวต้องไล่แก้ทุกตาราง
  • ไม่ประกาศ FK เพราะ "ให้ application ตรวจเอง" ซึ่งสุดท้ายมักมีข้อมูลกำพร้า (orphan row) หลงอยู่ ดูต่อในหัวข้อ Data Integrity
  • แยกตารางละเอียดเกินจนทุก query ต้อง JOIN สิบตาราง Normalization เป็นเครื่องมือ ไม่ได้มีเป้าหมายให้ตารางเยอะที่สุด

สังเกตว่า Anan กับ Malee เงินเดือนเท่ากัน ถ้าเรียงแค่ salary DESC ลำดับของสองคนนี้ไม่ได้ถูกรับประกัน ฐานข้อมูลจะคืนแบบไหนก็ได้ และอาจเปลี่ยนไปเมื่อข้อมูลหรือ plan เปลี่ยน ดังนั้นเวลาต้องการผลที่แน่นอน (โดยเฉพาะตอนแบ่งหน้า) ต้องใส่คอลัมน์ที่ทำให้ลำดับไม่กำกวมต่อท้ายเสมอ เช่น emp_id ซึ่งเรียกว่า tie-breaker

ตัวกรองที่ใช้บ่อยมีดังนี้ =, <> (หรือ !=), <, >, BETWEEN a AND b (รวมทั้งสองปลาย), IN (...), LIKE กับ wildcard สองตัวคือ % แทนกี่ตัวอักษรก็ได้ และ _ แทนหนึ่งตัวอักษร ตัวอย่างเช่น name LIKE '_a%' หมายถึงชื่อที่ตัวอักษรที่สองเป็น a ใน PostgreSQL ถ้าต้องการไม่สนตัวพิมพ์ใหญ่เล็กใช้ ILIKE ส่วน MySQL ที่ใช้ collation ปกติ LIKE ไม่สนตัวพิมพ์ใหญ่เล็กอยู่แล้ว

### AND กับ OR: AND มาก่อน

-- ตั้งใจ: แผนก 2 หรือ 3 ที่เงินเดือนเกิน 50000
SELECT emp_name, dept_id, salary FROM employees
WHERE dept_id = 2 OR dept_id = 3 AND salary > 50000;
 emp_name | dept_id |  salary
----------+---------+----------
 Nida     |       2 | 65000.00

AND ถูกคำนวณก่อน OR เหมือนคูณก่อนบวก query ข้างบนจึงอ่านว่า dept_id = 2 หรือ (dept_id = 3 AND salary > 50000) ทำให้ Preecha ติดมาทั้งที่เงินเดือน 50000 พอดี วิธีแก้คือใส่วงเล็บให้ตรงกับที่ตั้งใจ WHERE (dept_id = 2 OR dept_id = 3) AND salary > 50000 จะได้ Nida กับ Kanya สองคน หรือเขียนเป็น dept_id IN (2, 3) ก็ได้ผลเดียวกันและอ่านง่ายกว่า

NULL: ค่าที่ไม่ใช่ค่า

NULL หมายถึง "ไม่มีข้อมูล" หรือ "ไม่รู้ค่า" ไม่ใช่ศูนย์และไม่ใช่สตริงว่าง ผลที่ตามมาคือการเปรียบเทียบอะไรกับ NULL ก็ตาม ผลลัพธ์ไม่ใช่ true หรือ false แต่เป็น UNKNOWN และ WHERE เก็บเฉพาะแถวที่เงื่อนไขเป็น true เท่านั้น แถวที่เป็น UNKNOWN จึงหายไปเงียบ ๆ

SELECT emp_name FROM employees WHERE dept_id = NULL;     -- 0 แถว (ไม่ error แต่ไม่เคยเป็นจริง)
SELECT emp_name FROM employees WHERE dept_id IS NULL;    -- Suda

SELECT count(*) FROM employees WHERE dept_id <> 1;                       -- 5   (Suda ไม่ถูกนับ)
SELECT count(*) FROM employees WHERE dept_id <> 1 OR dept_id IS NULL;    -- 6

ตารางมีพนักงาน 10 คน แผนก 1 มี 4 คน เราจึงคิดว่า dept_id <> 1 ควรได้ 6 แต่ได้แค่ 5 เพราะ Suda มี dept_id เป็น NULL และ NULL <> 1 เป็น UNKNOWN เรื่องนี้เป็นต้นเหตุของบั๊กที่เจอบ่อยที่สุดในงานจริงเรื่องหนึ่ง

### NOT IN กับ NULL

-- แผนกไหนไม่มีพนักงานเลย
SELECT dept_name FROM departments
WHERE dept_id NOT IN (SELECT dept_id FROM employees);
-- ได้ 0 แถว ทั้งที่ Legal ไม่มีพนักงาน

subquery คืนค่าเป็น 1,1,1,1,2,2,2,3,4,NULL และ x NOT IN (...) แปลว่า x <> 1 AND x <> 2 AND ... AND x <> NULL เทอมสุดท้ายเป็น UNKNOWN ทำให้ทั้งเงื่อนไขไม่มีวันเป็น true ทางแก้ที่ปลอดภัยคือ NOT EXISTS หรือกรอง NULL ออกจาก subquery

SELECT dept_name FROM departments d
WHERE NOT EXISTS (SELECT 1 FROM employees e WHERE e.dept_id = d.dept_id);   -- Legal

### ฟังก์ชันรับมือ NULL

COALESCE(a, b, c) คืนค่าแรกที่ไม่ใช่ NULL เป็นมาตรฐาน SQL ใช้ได้ทุก DBMS ส่วน NULLIF(a, b) คืน NULL เมื่อ a เท่ากับ b มักใช้กันหารด้วยศูนย์ เช่น total / NULLIF(count, 0) และ ISNULL (SQL Server) หรือ IFNULL (MySQL) เป็นรุ่นสองอาร์กิวเมนต์ที่แต่ละค่ายทำเอง

SELECT name, COALESCE(city, 'ไม่ระบุ') AS city FROM customers;   -- Ek จะได้ 'ไม่ระบุ'

CASE, DISTINCT และการแบ่งหน้า

CASE คือ if-else ของ SQL ใช้ได้ทั้งใน SELECT, WHERE, ORDER BY และข้างใน aggregate เงื่อนไขแรกที่เป็นจริงชนะ ถ้าไม่มีข้อไหนตรงและไม่ใส่ ELSE ผลจะเป็น NULL

SELECT emp_name, salary,
       CASE WHEN salary >= 80000 THEN 'high'
            WHEN salary >= 55000 THEN 'mid'
            ELSE 'low' END AS band
FROM employees;

Aggregate, GROUP BY, HAVING และลำดับการประมวลผลของ Query

Beginner

Apply basic SQL queries with aggregate functions, grouping and filtering to summarize data correctly.

โจทย์ SQL ระดับ Basic ของ HackerRank ที่บอกไว้ว่าวัด "simple queries, relationships, and aggregators" ส่วนที่เป็น aggregator คือหัวข้อนี้ ข้อที่คนพลาดบ่อยมีสองเรื่อง คือไม่รู้ว่า COUNT(*) กับ COUNT(column) ต่างกัน และใช้ WHERE กับ HAVING สลับกัน

Aggregate function ทำอะไรกับ NULL

COUNT, SUM, AVG, MIN, MAX ยุบหลายแถวให้เหลือค่าเดียว ข้อควรจำคือทุกตัวยกเว้น COUNT(*) จะ ข้ามค่า NULL

SELECT count(*)              AS c_star,      -- 10  นับทุกแถว
       count(dept_id)        AS c_dept,      -- 9   นับเฉพาะที่ไม่ใช่ NULL
       count(DISTINCT dept_id) AS c_distinct -- 4   นับค่าที่ไม่ซ้ำ (ไม่รวม NULL)
FROM employees;

กับ AVG ก็เป็นแบบเดียวกัน ค่าเฉลี่ยของ 10, 20, NULL คือ 15 ไม่ใช่ 10 เพราะแถวที่เป็น NULL ไม่ถูกนำมาหาร ถ้าเจตนาให้ NULL นับเป็นศูนย์ต้องเขียน AVG(COALESCE(x, 0)) เอง อีกจุดที่ต้องระวังคือ SUM ของกลุ่มที่ทุกค่าเป็น NULL ได้ NULL ไม่ใช่ 0 ในขณะที่ COUNT ได้ 0

GROUP BY

GROUP BY แบ่งแถวเป็นกลุ่มตามค่าของคอลัมน์ที่ระบุ แล้วคำนวณ aggregate ต่อกลุ่ม

SELECT dept_id, count(*) AS headcount, round(avg(salary), 2) AS avg_salary
FROM employees
GROUP BY dept_id
ORDER BY dept_id;
 dept_id | headcount | avg_salary
---------+-----------+------------
       1 |         4 |   72500.00
       2 |         3 |   56666.67
       3 |         1 |   48000.00
       4 |         1 |   80000.00
         |         1 |   40000.00      <- Suda (dept_id เป็น NULL) ถูกรวมเป็นกลุ่มของตัวเอง

กฎของ GROUP BY คือทุกคอลัมน์ใน SELECT ต้องเป็นคอลัมน์ที่อยู่ใน GROUP BY หรือเป็นผลของ aggregate ไม่เช่นนั้นฐานข้อมูลไม่รู้ว่าจะเลือกชื่อพนักงานคนไหนมาแสดงในกลุ่มที่มี 4 คน PostgreSQL ตอบด้วย error column "employees.emp_name" must appear in the GROUP BY clause or be used in an aggregate function ส่วน MySQL ที่เปิดโหมด ONLY_FULL_GROUP_BY (เป็นค่าเริ่มต้นตั้งแต่ 5.7) ก็ error เช่นกัน ถ้าโหมดนี้ถูกปิด MySQL จะเลือกค่าจากแถวใดแถวหนึ่งให้เงียบ ๆ ซึ่งเป็นบั๊กที่ตรวจจับยาก

WHERE กับ HAVING

สองตัวนี้กรองเหมือนกัน แต่กรองคนละเวลา WHERE กรอง แถว ก่อนจัดกลุ่ม ส่วน HAVING กรอง กลุ่ม หลังจัดกลุ่มและคำนวณ aggregate แล้ว เงื่อนไขที่อ้าง aggregate เช่น count(*) >= 2 จึงใส่ใน WHERE ไม่ได้

-- แผนกที่มีพนักงานตั้งแต่ 2 คนขึ้นไป
SELECT dept_id, count(*) AS headcount
FROM employees
GROUP BY dept_id
HAVING count(*) >= 2;        -- แผนก 1 (4 คน) และแผนก 2 (3 คน)

-- จำนวนออเดอร์ที่จ่ายแล้วของลูกค้าแต่ละคน: กรองสถานะก่อนนับ ใช้ WHERE
SELECT customer_id, count(*) AS n
FROM orders
WHERE status = 'paid'
GROUP BY customer_id;

หลักคิดง่าย ๆ คือถ้าเงื่อนไขพูดถึงค่าในแถวเดียว ให้ใส่ WHERE เพราะกรองก่อนทำให้ต้องคำนวณน้อยลง ถ้าพูดถึงผลรวมหรือจำนวนของกลุ่ม ใส่ HAVING

ลำดับการประมวลผลจริงของ Query

SQL เขียนขึ้นต้นด้วย SELECT แต่ฐานข้อมูลคิดตามลำดับตรรกะอีกแบบ ถ้าจำลำดับนี้ได้ จะอธิบายได้ว่าทำไมบางอย่างเขียนไม่ได้

ลำดับขั้นผลที่ตามมา
1FROM / JOINเริ่มจากรวมตารางก่อน
2WHEREกรองแถว ยังไม่มี alias ของ SELECT และยังไม่มี aggregate
3GROUP BYจัดกลุ่ม
4HAVINGกรองกลุ่ม ใช้ aggregate ได้
5SELECTคำนวณนิพจน์และตั้ง alias, window function ทำงานตรงนี้
6DISTINCTตัดแถวซ้ำ
7ORDER BYเรียง ใช้ alias ของ SELECT ได้
8LIMIT / OFFSET / TOPตัดจำนวนแถว

ตัวอย่างผลของลำดับนี้ SELECT salary * 12 AS yearly FROM employees WHERE yearly > 600000 จะ error เพราะ WHERE ทำงานก่อน SELECT ยังไม่รู้จัก yearly ต้องเขียนนิพจน์ซ้ำใน WHERE หรือห่อด้วย subquery (ผ่อนปรนอยู่ในบางค่าย เช่น MySQL ยอมให้ใช้ alias ใน HAVING แต่ไม่ควรพึ่ง) ในทางกลับกัน ORDER BY yearly ทำได้เพราะ ORDER BY อยู่หลัง SELECT

Conditional aggregation

เทคนิคที่ใช้บ่อยมากในงานรายงานและโจทย์ pivot คือเอา CASE ไปไว้ข้างใน aggregate เพื่อนับหรือรวมเฉพาะบางเงื่อนไขในผลลัพธ์แถวเดียว

SELECT dept_id,
       SUM(CASE WHEN salary >= 60000 THEN 1 ELSE 0 END) AS n_high,
       COUNT(*) FILTER (WHERE salary < 60000)           AS n_low      -- PostgreSQL / SQLite
FROM employees
GROUP BY dept_id
ORDER BY dept_id;
 dept_id | n_high | n_low
---------+--------+-------
       1 |      4 |     0
       2 |      1 |     2
       3 |      0 |     1
       4 |      1 |     0
         |      0 |     1

FILTER (WHERE ...) เป็นมาตรฐาน SQL ที่ PostgreSQL กับ SQLite รองรับ แต่ MySQL และ SQL Server ไม่มี ให้ใช้ SUM(CASE ...) ซึ่งใช้ได้ทุกค่าย จึงเป็นรูปแบบที่ปลอดภัยที่สุดตอนไม่รู้ว่าข้อสอบรันบน DBMS ไหน

อีกเรื่องที่ควรฝึกคือนับยอดขายจากหลายตาราง ยอดรวมของทุกออเดอร์ในฐานข้อมูลนี้คำนวณจาก order_items เพราะเราไม่เก็บ total ไว้ในตาราง orders

SELECT o.status, SUM(oi.qty * oi.unit_price) AS revenue
FROM orders o
JOIN order_items oi USING (order_id)
GROUP BY o.status
ORDER BY o.status;

-----------+----------
 cancelled |  7500.00
 paid      | 89800.00
 pending   |  2400.00
 shipped   | 15000.00

ถ้าโจทย์ถาม "ยอดขาย" ให้ดูว่าควรนับสถานะ cancelled หรือไม่ ข้อสอบจริงมักซ่อนเงื่อนไขแบบนี้ไว้ในคำอธิบาย ไม่ได้บอกตรง ๆ ในชื่อ query

FULL JOINเก็บทั้งสองฝั่งเห็นทั้ง Suda และ Legal ในผลเดียว (MySQL ไม่มี FULL JOIN ต้องใช้ LEFT JOIN UNION RIGHT JOIN)
CROSS JOINไม่มีเงื่อนไข จับคู่ทุกแถวกับทุกแถวcustomers 6 คน × products 6 ชิ้น = 36 แถว
-- ทุกแผนกพร้อมจำนวนพนักงาน รวมแผนกที่ไม่มีคน
SELECT d.dept_name, COUNT(e.emp_id) AS n
FROM departments d
LEFT JOIN employees e ON e.dept_id = d.dept_id
GROUP BY d.dept_name
ORDER BY d.dept_name;
  dept_name  | n
-------------+---
 Engineering | 4
 Finance     | 1
 HR          | 1
 Legal       | 0
 Sales       | 3

ตรง COUNT(e.emp_id) ตั้งใจใช้คอลัมน์ของตารางฝั่งขวา ถ้าเขียน COUNT(*) Legal จะได้ 1 เพราะแถวที่ฝั่งขวาเป็น NULL ก็ยังนับเป็นหนึ่งแถว

### ON กับ WHERE ใน LEFT JOIN ไม่เหมือนกัน

นี่คือจุดที่โจทย์ชอบวางกับดัก เงื่อนไขที่ใส่ใน ON ใช้ตัดสินว่า "จับคู่ได้ไหม" ส่วนเงื่อนไขใน WHERE ใช้กรองผลลัพธ์หลัง JOIN เสร็จแล้ว ถ้าใส่เงื่อนไขของตารางฝั่งขวาไว้ใน WHERE แถวที่ฝั่งขวาเป็น NULL จะถูกกรองทิ้ง LEFT JOIN จึงกลายเป็น INNER JOIN โดยไม่ตั้งใจ

-- A: ลูกค้าทุกคน พร้อมจำนวนออเดอร์ที่จ่ายแล้ว (Ek และ Fah ได้ 0)
SELECT c.name, COUNT(o.order_id)
FROM customers c
LEFT JOIN orders o ON o.customer_id = c.customer_id AND o.status = 'paid'
GROUP BY c.name;

-- B: เงื่อนไขเดียวกันย้ายไป WHERE (เหลือเฉพาะลูกค้าที่มีออเดอร์ paid: 4 คน)
SELECT c.name, COUNT(o.order_id)
FROM customers c
LEFT JOIN orders o ON o.customer_id = c.customer_id
WHERE o.status = 'paid'
GROUP BY c.name;

A ได้ 6 แถว B ได้ 4 แถว เพราะ Ek กับ Fah ไม่มีออเดอร์ o.status จึงเป็น NULL แล้ว NULL = 'paid' ไม่เป็นจริง

### หาแถวที่ "ไม่มี" ในอีกตาราง (anti-join)

ลูกค้าที่ไม่เคยสั่งของ เขียนได้สองแบบที่ให้ผลเหมือนกัน

-- แบบ LEFT JOIN แล้วกรองแถวที่หาคู่ไม่เจอ
SELECT c.name
FROM customers c
LEFT JOIN orders o ON o.customer_id = c.customer_id
WHERE o.order_id IS NULL;

-- แบบ NOT EXISTS อ่านตรงกับความหมายมากกว่า
SELECT c.name
FROM customers c
WHERE NOT EXISTS (SELECT 1 FROM orders o WHERE o.customer_id = c.customer_id);

ทั้งคู่ได้ Ek กับ Fah ส่วน NOT IN (SELECT customer_id FROM orders) ก็ใช้ได้เมื่อ orders.customer_id ไม่มีทาง NULL (ซึ่งตารางนี้ตั้ง NOT NULL ไว้) แต่ถ้าคอลัมน์ฝั่ง subquery เป็น NULL ได้ จะเจอปัญหาที่เล่าไว้ในหัวข้อ 2 ซึ่งเป็นเหตุผลที่คนในทีมส่วนใหญ่ตั้งกติกาให้ใช้ NOT EXISTS

Self join และ JOIN หลายตาราง

Self join คือการ JOIN ตารางกับตัวเอง ต้องตั้ง alias ให้สองฝั่งต่างกัน ตัวอย่างคลาสสิกคือหาชื่อหัวหน้าของพนักงาน

SELECT e.emp_name AS employee, m.emp_name AS manager
FROM employees e
LEFT JOIN employees m ON m.emp_id = e.manager_id
ORDER BY e.emp_id;

ใช้ LEFT JOIN เพื่อให้พนักงานที่ไม่มีหัวหน้า (Somchai, Nida, Wirat, Tanawat) ยังอยู่ในผล โดยช่อง manager เป็น NULL

JOIN สามตารางขึ้นไปก็คือ JOIN ต่อกันเป็นทอด ๆ คิดทีละคู่ตามลำดับ ตัวอย่างหายอดซื้อของลูกค้าแต่ละคนแยกตามหมวดสินค้า

SELECT c.name, p.category, SUM(oi.qty * oi.unit_price) AS amount
FROM customers c
JOIN orders o       ON o.customer_id = c.customer_id
JOIN order_items oi ON oi.order_id   = o.order_id
JOIN products p     ON p.product_id  = oi.product_id
WHERE o.status IN ('paid', 'shipped')
GROUP BY c.name, p.category
ORDER BY c.name, p.category;

คำสั่ง USING (col) กับ NATURAL JOIN เป็นทางลัดเมื่อชื่อคอลัมน์ตรงกัน USING ใช้ได้ดีและอ่านง่าย แต่ NATURAL JOIN จับคู่ทุกคอลัมน์ที่ชื่อซ้ำกันโดยอัตโนมัติ ถ้าวันหนึ่งมีคนเพิ่มคอลัมน์ชื่อซ้ำ (เช่น created_at) เข้าไปทั้งสองตาราง ผลจะเปลี่ยนเงียบ ๆ จึงไม่แนะนำในโค้ดจริง

Fan-out: ผลรวมที่ใหญ่เกินจริง

JOIN แบบ one-to-many ทำให้แถวฝั่ง "one" ซ้ำตามจำนวนแถวฝั่ง "many" ถ้านับหรือรวมค่าของฝั่ง one หลัง JOIN ผลจะเพี้ยน

SELECT c.name,
       COUNT(*)                 AS n_rows,
       COUNT(DISTINCT o.order_id) AS n_orders
FROM customers c
JOIN orders o       ON o.customer_id = c.customer_id
JOIN order_items oi ON oi.order_id   = o.order_id
GROUP BY c.name;
  name   | n_rows | n_orders
---------+--------+----------
 Arisa   |      5 |        3
 Boonmee |      2 |        2
 Chai    |      6 |        3
 Dao     |      3 |        2

Arisa มี 3 ออเดอร์ แต่หลัง JOIN กับ order_items ได้ 5 แถว ถ้าโจทย์ถาม "ลูกค้าแต่ละคนมีกี่ออเดอร์" แล้วเขียน COUNT(*) จะได้ 5 ซึ่งผิด ทางแก้มีสองแบบ ใช้ COUNT(DISTINCT o.order_id) หรือรวมฝั่ง many ให้เหลือหนึ่งแถวต่อออเดอร์ก่อนแล้วค่อย JOIN (ใช้ CTE ได้ ดูหัวข้อถัดไป) แบบหลังปลอดภัยกว่าเมื่อต้อง SUM ค่าของฝั่ง one เพราะ DISTINCT กับ SUM ไม่ช่วยถ้าสองออเดอร์มียอดเท่ากัน

UNION, INTERSECT, EXCEPT

Set operator ต่อผลลัพธ์ของสอง query ในแนวตั้ง ทั้งสอง query ต้องมีจำนวนคอลัมน์เท่ากัน และชนิดข้อมูลของแต่ละตำแหน่งต้องเข้ากันได้ ชื่อคอลัมน์ของผลลัพธ์ใช้ตาม query แรก

ตัวดำเนินการผลลัพธ์ตัดแถวซ้ำ
UNIONแถวที่อยู่ในฝั่งใดฝั่งหนึ่งตัด
UNION ALLทุกแถวของทั้งสองฝั่งไม่ตัด
INTERSECTแถวที่อยู่ทั้งสองฝั่งตัด
EXCEPT (Oracle เรียก MINUS)แถวฝั่งซ้ายที่ไม่อยู่ฝั่งขวาตัด
SELECT count(*) FROM (SELECT city FROM customers UNION ALL SELECT city FROM customers) t;   -- 12
SELECT count(*) FROM (SELECT city FROM customers UNION     SELECT city FROM customers) t;   -- 4  (Bangkok, Chiang Mai, Phuket, NULL)

UNION ต้องเทียบและตัดแถวซ้ำ ซึ่งเป็นงานเรียงหรือ hash เพิ่ม ถ้าเรารู้อยู่แล้วว่าสองฝั่งไม่มีแถวซ้ำกัน หรือซ้ำก็ไม่เป็นไร ให้ใช้ UNION ALL ซึ่งเร็วกว่าและให้ผลที่คาดเดาได้ ส่วน INTERSECT และ EXCEPT เหมือนกับ UNION ตรงที่มองว่า NULL สองตัวเท่ากัน ต่างจากเครื่องหมาย = ที่ NULL ไม่เท่ากับอะไรเลย (MySQL เพิ่ม INTERSECT และ EXCEPT ใน 8.0.31 เวอร์ชันก่อนหน้าไม่มี)

วิธีเช็กผล JOIN ก่อนส่งงานก่อนใส่ aggregate ให้รัน JOIN เปล่า ๆ แล้วนับแถว เทียบกับที่คาดจากข้อมูล ถ้าจำนวนแถวเกินจำนวนแถวของตารางที่เป็นฐาน แปลว่ามี fan-out ถ้าแถวของฝั่งซ้ายหายไป ให้ตรวจว่า INNER JOIN หรือเงื่อนไขใน WHERE ตัดมันทิ้งหรือเปล่า

Subquery, CTE, Recursive CTE และ Window Function

Intermediate

Develop optimized complex SQL queries using subqueries and CTEs, including window functions for ranking and running calculations.

พอ query เริ่มซับซ้อน ปัญหาแรกที่เจอคือเขียนในก้อนเดียวไม่ไหว ต้องใช้ผลของ query หนึ่งไปเป็นข้อมูลให้อีก query หนึ่ง SQL มีเครื่องมือสามตระกูลสำหรับเรื่องนี้ คือ subquery, CTE และ window function แต่ละตัวเหมาะกับงานต่างกัน

Subquery สี่แบบ

Subquery คือ query ที่ซ้อนอยู่ในอีก query ใช้ได้ในหลายตำแหน่ง ที่เจอบ่อยมีสี่แบบ

-- 1) Scalar subquery: คืนค่าเดียว ใช้แทนค่าคงที่ได้
SELECT emp_name, salary
FROM employees
WHERE salary > (SELECT AVG(salary) FROM employees);        -- Somchai, Tanawat, Malee, Anan, Nida

-- 2) IN / EXISTS: ตรวจว่ามีแถวที่ตรงเงื่อนไขหรือไม่
SELECT name FROM customers c
WHERE EXISTS (SELECT 1 FROM orders o
              WHERE o.customer_id = c.customer_id AND o.status = 'cancelled');   -- Boonmee

-- 3) Correlated subquery: อ้างคอลัมน์ของ query ข้างนอก จึงคิดใหม่ทุกแถว (ในทางตรรกะ)
SELECT emp_name, salary, dept_id
FROM employees e
WHERE salary > (SELECT AVG(salary) FROM employees x WHERE x.dept_id = e.dept_id);   -- Somchai, Nida

-- 4) Derived table: subquery ใน FROM ต้องตั้ง alias
SELECT t.dept_id, t.avg_salary
FROM (SELECT dept_id, AVG(salary) AS avg_salary FROM employees GROUP BY dept_id) t
WHERE t.avg_salary > 60000;

คำว่า "คิดใหม่ทุกแถว" ของ correlated subquery เป็นเรื่องความหมาย ตัว optimizer มักแปลงเป็น JOIN หรือ semi-join ให้เอง จึงไม่ได้ช้าเสมอไป แต่ถ้าเจอกรณีที่ช้า การเขียนใหม่เป็น JOIN กับ subquery ที่ GROUP BY ไว้ก่อนมักช่วยได้ ใช้ EXPLAIN ยืนยันเสมอ ไม่ควรเดา

IN กับ EXISTS ให้ผลเหมือนกันเมื่อไม่มี NULL มาเกี่ยว ต่างกันที่ EXISTS หยุดค้นทันทีที่เจอแถวแรก และไม่สนว่า subquery เลือกคอลัมน์อะไรออกมา (เขียน SELECT 1 เป็นธรรมเนียม) ส่วนเรื่อง NOT IN กับ NULL ที่อันตรายนั้นบอกไว้แล้วในหัวข้อ 2

CTE: ตั้งชื่อให้ subquery

CTE (Common Table Expression) ขึ้นต้นด้วย WITH ตั้งชื่อให้ผลลัพธ์ชั่วคราว แล้วเอาไปใช้ใน query หลักเหมือนตาราง ข้อดีหลักคืออ่านง่ายขึ้น เพราะแตก query ยาวเป็นขั้น ๆ อ่านจากบนลงล่าง และใช้ชื่อเดิมซ้ำได้หลายที่

WITH order_totals AS (
  SELECT o.order_id, o.customer_id, o.status,
         SUM(oi.qty * oi.unit_price) AS total
  FROM orders o
  JOIN order_items oi ON oi.order_id = o.order_id
  GROUP BY o.order_id, o.customer_id, o.status
)
SELECT c.name, COUNT(*) AS n_orders, SUM(t.total) AS spent
FROM order_totals t
JOIN customers c ON c.customer_id = t.customer_id
WHERE t.status IN ('paid', 'shipped')
GROUP BY c.name
ORDER BY spent DESC;
  name   | n_orders |  spent
---------+----------+----------
 Arisa   |        3 | 43100.00
 Dao     |        1 | 33500.00
 Chai    |        3 | 26400.00
 Boonmee |        1 |  1800.00

ขั้นแรกรวมยอดต่อออเดอร์ให้เหลือหนึ่งแถวต่อหนึ่งออเดอร์ ซึ่งแก้ปัญหา fan-out ที่เล่าในหัวข้อ JOIN ไปในตัว แล้วขั้นที่สองค่อยนับและรวมต่อลูกค้า เรื่องที่ต้องรู้เกี่ยวกับ CTE มีสองข้อ

  • ประสิทธิภาพขึ้นกับเวอร์ชัน ใน PostgreSQL ก่อน 12 CTE ถูกคำนวณแยกเป็นก้อนเสมอ (optimization fence) ตั้งแต่ 12 เป็นต้นไป ถ้า CTE ไม่มี side effect และถูกอ้างถึงครั้งเดียว optimizer จะ inline เข้า query หลัก และสั่งบังคับได้ด้วย AS MATERIALIZED หรือ AS NOT MATERIALIZED ส่วน MySQL 8.0 ขึ้นไปและ SQL Server ก็ถือว่า CTE เป็นแค่ตัวช่วยเขียน ไม่ได้รับประกันว่าจะคำนวณครั้งเดียว
  • MySQL เริ่มรองรับ CTE ใน 8.0 ถ้าข้อสอบรันบน MySQL 5.7 ต้องใช้ derived table แทน

### Recursive CTE

ใช้กับข้อมูลที่เป็นต้นไม้หรือกราฟ เช่น ผังหัวหน้า-ลูกน้อง ประกอบด้วยสองส่วนที่ต่อกันด้วย UNION ALL ส่วนแรกคือจุดเริ่ม (anchor) ส่วนที่สองคือส่วน recursive ที่ JOIN กลับมาหา CTE ตัวเอง ทำซ้ำจนไม่มีแถวใหม่

WITH RECURSIVE chain AS (
  SELECT emp_id, emp_name, 1 AS lvl, emp_name::text AS path
  FROM employees
  WHERE manager_id IS NULL
  UNION ALL
  SELECT e.emp_id, e.emp_name, c.lvl + 1, c.path || ' > ' || e.emp_name
  FROM employees e
  JOIN chain c ON e.manager_id = c.emp_id
  WHERE c.lvl < 10                              -- กันวนไม่รู้จบถ้าข้อมูลมี cycle
)
SELECT lvl, path FROM chain ORDER BY path;
 lvl |         path
-----+-----------------------
   1 | Nida
   2 | Nida > Kanya
   2 | Nida > Preecha
   1 | Somchai
   2 | Somchai > Anan
   2 | Somchai > Malee
   3 | Somchai > Malee > Pim
   2 | Somchai > Suda
   1 | Tanawat
   1 | Wirat

ถ้าข้อมูลมีวงวน เช่น A เป็นหัวหน้า B และ B เป็นหัวหน้า A query จะไม่จบเอง จึงต้องมีตัวจำกัดระดับ (lvl < 10) หรือเก็บ path แล้วเช็กว่าไม่เคยเจอ id นี้ซ้ำ ส่วนไวยากรณ์ต่างกันเล็กน้อย PostgreSQL และ MySQL ต้องเขียน WITH RECURSIVE ส่วน SQL Server เขียนแค่ WITH แม้จะ recursive

Window function

Aggregate ธรรมดาที่มี GROUP BY ยุบหลายแถวเหลือแถวเดียว ส่วน window function คำนวณจากกลุ่มของแถวแต่ ไม่ยุบแถว ทุกแถวยังอยู่ แล้วเพิ่มคอลัมน์ผลคำนวณเข้าไป เขียนด้วย ฟังก์ชัน() OVER (PARTITION BY ... ORDER BY ...) โดย PARTITION BY คือการแบ่งกลุ่ม (ไม่ใส่แปลว่ากลุ่มเดียวทั้งตาราง) และ ORDER BY คือลำดับภายในกลุ่ม

### ROW_NUMBER, RANK, DENSE_RANK

สามตัวนี้ให้ลำดับ ต่างกันตรงวิธีจัดการกับค่าที่เท่ากัน ลองดูแผนก Engineering ที่ Malee กับ Anan เงินเดือนเท่ากันที่ 70000

SELECT emp_name, salary,
       ROW_NUMBER() OVER (ORDER BY salary DESC, emp_id) AS rn,
       RANK()       OVER (ORDER BY salary DESC)         AS rk,
       DENSE_RANK() OVER (ORDER BY salary DESC)         AS drk
FROM employees
WHERE dept_id = 1;
 emp_name |  salary  | rn | rk | drk
----------+----------+----+----+-----
 Somchai  | 90000.00 |  1 |  1 |   1
 Malee    | 70000.00 |  2 |  2 |   2
 Anan     | 70000.00 |  3 |  2 |   2
 Pim      | 60000.00 |  4 |  4 |   3

ROW_NUMBER ให้เลขเรียงไม่ซ้ำ แม้ค่าเท่ากันก็ต้องมีใครสักคนได้เลขก่อน (ถ้าไม่ใส่ tie-breaker ลำดับของสองคนนี้ไม่แน่นอน) RANK ให้ค่าเท่ากันได้เลขเดียวกัน แล้วข้ามเลขถัดไป Pim จึงเป็นอันดับ 4 และ DENSE_RANK ให้ค่าเท่ากันได้เลขเดียวกันเหมือนกัน แต่ไม่ข้ามเลข Pim เป็นอันดับ 3

### Top-N ต่อกลุ่ม

โจทย์ที่ออกบ่อยที่สุดของ window function คือ "เอา N อันดับแรกในแต่ละกลุ่ม" เช่น พนักงานเงินเดือนสูงสุดสองอันดับในแต่ละแผนก window function ใช้ใน WHERE ตรง ๆ ไม่ได้ เพราะมันทำงานตอน SELECT ซึ่งอยู่หลัง WHERE PostgreSQL จะตอบว่า window functions are not allowed in WHERE ต้องห่อไว้ใน CTE หรือ subquery ก่อน

WITH ranked AS (
  SELECT emp_name, dept_id, salary,
         DENSE_RANK() OVER (PARTITION BY dept_id ORDER BY salary DESC) AS r
  FROM employees
  WHERE dept_id IS NOT NULL
)
SELECT dept_id, emp_name, salary
FROM ranked
WHERE r <= 2
ORDER BY dept_id, salary DESC, emp_name;

### LAG, LEAD และ running total

LAG(col) ดึงค่าจากแถวก่อนหน้า LEAD(col) ดึงจากแถวถัดไป ส่วน aggregate ที่ใส่ OVER (ORDER BY ...) จะคำนวณสะสมจากแถวแรกจนถึงแถวปัจจุบัน

WITH daily AS (
  SELECT o.order_date, SUM(oi.qty * oi.unit_price) AS amt
  FROM orders o JOIN order_items oi ON oi.order_id = o.order_id
  WHERE o.status IN ('paid', 'shipped')
  GROUP BY o.order_date
)
SELECT order_date, amt,
       SUM(amt) OVER (ORDER BY order_date)      AS running,
       amt - LAG(amt) OVER (ORDER BY order_date) AS diff_from_prev
FROM daily
ORDER BY order_date;

Index ทำงานอย่างไร

Index ของ DBMS ส่วนใหญ่เป็น B-tree คือโครงสร้างต้นไม้ที่เก็บค่าของคอลัมน์แบบเรียงลำดับ พร้อมตัวชี้ไปยังแถวจริง เหมือนสารบัญท้ายเล่ม เราไม่ต้องไล่อ่านทุกหน้าเพื่อหาคำว่า "index" แต่เปิดสารบัญแล้วกระโดดไปหน้าที่ถูกต้องได้เลย จำนวนขั้นที่ต้องเดินโตช้ามากเมื่อข้อมูลเพิ่ม ข้อมูลหลักล้านแถวยังใช้เพียงไม่กี่ระดับ

Index มีต้นทุนด้วย ทุกครั้งที่ INSERT, UPDATE คอลัมน์ที่ถูก index, หรือ DELETE ฐานข้อมูลต้องปรับ index ทุกตัวของตารางนั้น และ index กินพื้นที่ดิสก์และหน่วยความจำ การใส่ index ให้ทุกคอลัมน์จึงไม่ใช่คำตอบ ควรสร้างตามรูปแบบ query ที่ใช้จริง และตรวจเป็นระยะว่ามี index ที่ไม่เคยถูกใช้ (PostgreSQL ดูได้จาก pg_stat_user_indexes)

คอลัมน์ที่มีค่าแตกต่างกันน้อยมาก เช่น status ที่มีสี่ค่า ถ้าสร้าง index เดี่ยว ๆ มักไม่ช่วย เพราะแต่ละค่าตรงกับแถวเป็นสิบเปอร์เซ็นต์ของตาราง อ่านผ่าน index แล้วกระโดดไปอ่านตารางสลับไปมาอาจช้ากว่าอ่านตารางตรง ๆ ทั้งก้อน optimizer จึงเลือก Seq Scan เอง ซึ่งเป็นเรื่องปกติ ไม่ใช่ว่า index เสีย

Composite index และลำดับของคอลัมน์

Index หลายคอลัมน์เรียงค่าตามคอลัมน์แรกก่อน แล้วค่าที่เท่ากันเรียงตามคอลัมน์ที่สอง และต่อไปเรื่อย ๆ เหมือนสมุดโทรศัพท์ที่เรียงตามนามสกุลแล้วตามชื่อ ผลที่ตามมาคือ index (customer_id, status, order_date) ช่วยได้เมื่อเงื่อนไขเริ่มจากคอลัมน์ซ้ายสุดต่อเนื่องกันไป (กฎ leftmost prefix) แต่ถ้า query มีแค่ status กับ order_date ก็ใช้ไม่ได้ เหมือนหาชื่อ "สมชาย" ในสมุดโทรศัพท์โดยไม่รู้นามสกุล

หลักการเลือกลำดับคอลัมน์ที่ใช้กันทั่วไปคือ เอาคอลัมน์ที่ใช้เทียบเท่ากับ (=) ไว้ก่อน แล้วตามด้วยคอลัมน์ที่ใช้เป็นช่วง (>, <, BETWEEN) หรือใช้เรียง ลองดู query นี้ ซึ่งหาห้าออเดอร์ล่าสุดที่จ่ายแล้วของลูกค้าคนหนึ่ง

Intermediate

Implement data integrity constraints so the database itself rejects invalid data.

Application ตรวจข้อมูลก่อนบันทึกอยู่แล้ว ทำไมยังต้องมี constraint ในฐานข้อมูลอีก คำตอบคือฐานข้อมูลมักมีมากกว่าหนึ่งทางเข้า มีทั้งแอปหลายตัว สคริปต์ import ข้อมูล คนที่รัน SQL เองตอนแก้ปัญหา และบั๊กที่ผ่านการตรวจของแอปไปได้ Constraint คือด่านสุดท้ายที่รับประกันว่าข้อมูลที่เข้าตารางแล้วต้องเป็นไปตามกติกา ไม่ว่าจะเขียนมาจากทางไหน

UNIQUE กับ NULL

UNIQUE ปฏิบัติกับ NULL ไม่เหมือนกันในแต่ละ DBMS และเป็นข้อที่ชอบถาม

DBMSอนุญาตให้ NULL กี่ตัวใน UNIQUE column
PostgreSQLหลายตัว เพราะ NULL ไม่เท่ากับ NULL (PostgreSQL 15 เพิ่มตัวเลือก UNIQUE NULLS NOT DISTINCT ถ้าต้องการให้นับเป็นค่าซ้ำ)
MySQL (InnoDB)หลายตัว
Oracleหลายตัวสำหรับ unique คอลัมน์เดียว (คีย์หลายคอลัมน์ที่เป็น NULL บางส่วนมีกติกาเฉพาะของ Oracle ให้เช็กเอกสาร)
SQL Serverได้ตัวเดียว เพราะมองว่า NULL สองตัวซ้ำกัน ถ้าต้องการหลายตัวให้ใช้ filtered unique index WHERE col IS NOT NULL

Foreign key และการจัดการเมื่อลบหรือแก้พ่อแม่

เมื่อประกาศ foreign key เรากำหนดได้ว่าถ้าแถวพ่อแม่ถูกลบหรือถูกแก้ key จะให้เกิดอะไรกับแถวลูก

การกระทำผลต่อแถวลูกใช้เมื่อ
NO ACTION / RESTRICT (ค่าเริ่มต้น)ปฏิเสธการลบพ่อแม่ที่ยังมีลูก (NO ACTION ตรวจตอนจบ statement, RESTRICT ตรวจทันที)ข้อมูลทางธุรกิจที่ห้ามหายเงียบ ๆ เช่นลูกค้ากับออเดอร์
CASCADEลบลูกตามไปด้วยลูกไม่มีความหมายเมื่อไม่มีพ่อแม่ เช่นรายการสินค้าในออเดอร์ที่ถูกลบ
SET NULLตั้ง FK ของลูกเป็น NULL (คอลัมน์ต้องยอมให้เป็น NULL)ความสัมพันธ์เสริม เช่นพนักงานที่แผนกถูกยุบ
SET DEFAULTตั้งเป็นค่า defaultใช้น้อย
CREATE TABLE child (
  id  int PRIMARY KEY,
  pid int REFERENCES parent (id) ON DELETE CASCADE
);
-- ลบ parent id=1 → แถว child ที่ pid=1 ถูกลบตาม

CASCADE สะดวกแต่เป็นดาบสองคม การลบแถวเดียวอาจลบข้อมูลต่อเนื่องเป็นพันแถวในหลายตารางโดยไม่รู้ตัว ใช้กับข้อมูลที่ลูกเป็นส่วนหนึ่งของพ่อแม่จริง ๆ และอย่าใช้กับข้อมูลที่ต้องเก็บไว้เป็นหลักฐาน เช่นออเดอร์กับบัญชีผู้ใช้ ซึ่งมักใช้ soft delete (คอลัมน์ deleted_at หรือ is_active) แทน

foreign key ที่เป็น NULL ไม่ถูกตรวจ เพราะ NULL หมายถึงไม่มีความสัมพันธ์ ดังนั้นถ้าต้องการให้ลูกต้องมีพ่อแม่เสมอ ต้องใส่ NOT NULL กำกับด้วย ตาราง orders.customer_id ของเราทำแบบนี้ ส่วน employees.dept_id ปล่อยให้เป็น NULL ได้ จึงมี Suda ที่ไม่มีแผนก

เรื่องของ index ที่มักลืมPostgreSQL ไม่สร้าง index ให้คอลัมน์ foreign key ฝั่งลูกอัตโนมัติ (ตัว primary key ฝั่งพ่อแม่มี index อยู่แล้ว) พอลบแถวพ่อแม่หรือ JOIN จากพ่อแม่ไปลูก ฐานข้อมูลต้องหาแถวลูก ซึ่งถ้าไม่มี index จะ scan ทั้งตารางลูก ส่วน MySQL InnoDB สร้าง index ให้คอลัมน์ FK อัตโนมัติถ้ายังไม่มี

CHECK, DEFAULT และข้อควรระวังเรื่องเวอร์ชัน

CHECK ตรวจนิพจน์ต่อแถว ใช้ได้กับกติกาที่เป็นอิสระต่อแถวอื่น เช่น price >= 0, end_date >= start_date, status IN (...) แต่ตรวจข้ามแถวไม่ได้ เช่น "ห้ามช่วงเวลาจองซ้อนกัน" หรือ "ผลรวมของรายการต้องเท่ากับยอดรวม" กรณีแบบนั้นต้องใช้ trigger, unique index เฉพาะทาง หรือ exclusion constraint ของ PostgreSQL

CREATE EXTENSION btree_gist;
CREATE TABLE booking (
  room   int,
  during daterange,
  EXCLUDE USING gist (room WITH =, during WITH &&)      -- ห้องเดียวกัน ช่วงเวลาห้ามทับกัน
);
INSERT INTO booking VALUES (1, '[2026-01-01,2026-01-05)');
INSERT INTO booking VALUES (1, '[2026-01-04,2026-01-08)');
-- ERROR: conflicting key value violates exclusion constraint "booking_room_during_excl"

เรื่องเวอร์ชันที่ควรรู้คือ MySQL อ่านไวยากรณ์ CHECK ได้มานานแต่ ไม่บังคับใช้ จนกระทั่งเวอร์ชัน 8.0.16 ถ้าข้อสอบหรือระบบเก่าอยู่บน MySQL 5.7 การใส่ CHECK ไม่ได้ป้องกันอะไรเลย ต้องตรวจในแอปหรือ trigger แทน

### เพิ่ม constraint ให้ตารางที่มีข้อมูลอยู่แล้ว

การ ALTER TABLE ... ADD CONSTRAINT บนตารางใหญ่ต้องตรวจทุกแถว และอาจล็อกตารางไว้นาน PostgreSQL มีทางลดผลกระทบ คือเพิ่มแบบ NOT VALID ก่อน ซึ่งบังคับใช้กับแถวใหม่ทันทีแต่ยังไม่ตรวจของเก่า แล้วค่อยรัน VALIDATE CONSTRAINT ภายหลัง ซึ่งล็อกเบากว่า

ALTER TABLE t ADD CONSTRAINT amt_pos CHECK (amount > 0) NOT VALID;
INSERT INTO t VALUES (3, -1);                     -- ถูกปฏิเสธแล้ว
ALTER TABLE t VALIDATE CONSTRAINT amt_pos;        -- ถ้ามีแถวเก่าที่ผิด จะ error จนกว่าจะแก้ข้อมูล

ถ้ามีข้อมูลเก่าที่ผิดกติกาอยู่ ต้องหาก่อนด้วย SELECT ... WHERE NOT (amount > 0) แล้วตัดสินใจร่วมกับเจ้าของข้อมูลว่าจะแก้หรือลบ ไม่ใช่ปิด constraint ทิ้ง

Deferrable constraint

ปกติฐานข้อมูลตรวจ constraint ทันทีที่จบแต่ละ statement ในบางกรณีเราต้องแทรกข้อมูลสองตารางที่อ้างกันเอง (a ชี้ b และ b ชี้ a) ซึ่งไม่มีลำดับไหนที่ผ่านการตรวจทันที PostgreSQL และ Oracle มี DEFERRABLE INITIALLY DEFERRED ให้เลื่อนการตรวจ foreign key ไปตอน COMMIT ภายใน transaction เดียวกันเราใส่ข้อมูลตามลำดับใดก็ได้ ขอแค่ตอนจบทุกอย่างสอดคล้องกัน SQL Server และ MySQL ไม่มีความสามารถนี้ ต้องแก้ด้วยการออกแบบ เช่น ตั้ง FK ให้เป็น NULL ได้ แล้วค่อย update ทีหลัง

ใช้ constraint คู่กับการตรวจในแอปตรวจในแอปเพื่อให้ผู้ใช้ได้ข้อความ error ที่เข้าใจง่ายตั้งแต่ต้น ตรวจในฐานข้อมูลเพื่อรับประกันความถูกต้อง สองอย่างไม่ได้ซ้ำซ้อนกัน อย่างแรกคือ UX อย่างหลังคือความปลอดภัยของข้อมูล

ระดับ Advance

ระดับนี้ต้องปรับแต่ง query กับข้อมูลปริมาณมากได้ อ่าน execution plan และ profile หาจุดช้า เขียน stored procedure, trigger และ function ได้อย่างมีเหตุผล และเข้าใจ transaction, concurrency และ locking พอจะอธิบายว่าทำไมผลลัพธ์ถึงถูกหรือผิดเมื่อมีคนใช้พร้อมกัน

Query Optimization สำหรับข้อมูลขนาดใหญ่

Advance

Design and implement advanced query optimization strategies for large and complex datasets.

ระดับ Advanced ไม่ได้ถามว่า "รู้จัก index ไหม" แต่ถามว่า query ที่ช้าบนข้อมูลหลายสิบล้านแถวควรไล่แก้อย่างไร และแต่ละทางเลือกแลกกับอะไร ข้อมูลโตขึ้นสิบเท่า เวลาที่ใช้ไม่ได้โตตามสัดส่วนเสมอไป บาง query ที่เร็วตอนข้อมูลน้อยจะช้าลงแบบก้าวกระโดดเมื่อถึงจุดที่ข้อมูลไม่พอดีหน่วยความจำ ตัวอย่างในหัวข้อนี้ใช้ตาราง orders 200,000 แถวชุดเดียวกับหัวข้อ 6 และ plan ที่แสดงมาจากการรันจริงบน PostgreSQL 16

วิธีคิด: ถามก่อนว่าช้าที่ไหน

ความช้าของ query ส่วนใหญ่มาจากสี่สาเหตุ และแต่ละสาเหตุแก้คนละวิธี

  1. อ่านข้อมูลเกินจำเป็น เช่น scan ทั้งตารางเพื่อหาไม่กี่แถว หรือดึงทุกคอลัมน์ทั้งที่ใช้สองตัว
  2. optimizer เลือก plan ผิดเพราะประมาณจำนวนแถวผิด (หัวข้อ 9 จะลงลึก)
  3. ยิง query มากเกินไป เช่น N+1 หรือวน loop ทำทีละแถว
  4. รอ lock จาก transaction อื่น (หัวข้อ 11)

ก่อนแก้อะไรให้วัดก่อนเสมอ ใช้ EXPLAIN ANALYZE กับข้อมูลที่มีขนาดและการกระจายใกล้ของจริง การทดสอบบนตาราง 100 แถวบอกอะไรไม่ได้ เพราะ optimizer จะเลือก seq scan ทุกกรณี และอย่าตัดสินด้วยเวลาที่เห็นครั้งเดียว ครั้งแรกอาจช้าเพราะ cache ยังเย็น

Pagination: OFFSET ลึกแล้วช้า

LIMIT 20 OFFSET 100000 ไม่ได้กระโดดไปแถวที่ 100,001 ได้เลย ฐานข้อมูลต้องอ่านและทิ้งแถวก่อนหน้า 100,000 แถวก่อน ยิ่งหน้าลึก งานยิ่งมาก

SELECT order_id, order_date FROM orders
ORDER BY order_date DESC, order_id DESC
LIMIT 20 OFFSET 100000;
 Limit (actual rows=20 loops=1)
   ->  Index Only Scan using idx_orders_date_id on orders (actual rows=100020 loops=1)
         Heap Fetches: 0

บรรทัดที่สองบอกทุกอย่าง ต้องอ่าน 100,020 แถวเพื่อส่งคืน 20 แถว ทางเลือกคือ keyset pagination (บางที่เรียก seek method) เก็บค่าของแถวสุดท้ายที่ผู้ใช้เห็นไว้ แล้วขอ "แถวที่อยู่หลังจากนี้"

-- index: (order_date DESC, order_id DESC)   หน้าถัดไปต่อจาก (2025-03-01, 123456)
SELECT order_id, order_date FROM orders
WHERE (order_date, order_id) < ('2025-03-01', 123456)
ORDER BY order_date DESC, order_id DESC
LIMIT 20;
 Limit (actual rows=20 loops=1)
   ->  Index Only Scan using idx_orders_date_id on orders (actual rows=20 loops=1)
         Index Cond: (ROW(order_date, order_id) < ROW('2025-03-01'::date, 123456))

คราวนี้อ่านแค่ 20 แถว ไม่ว่าจะอยู่หน้าไหน นอกจากเร็วคงที่แล้ว ยังไม่เจอปัญหาข้อมูลเลื่อนเมื่อมีแถวใหม่เข้ามาระหว่างที่ผู้ใช้เปลี่ยนหน้า (OFFSET จะทำให้เห็นแถวซ้ำหรือตกหล่น) ข้อแลกเปลี่ยนคือกระโดดไปหน้าที่ N ตรง ๆ ไม่ได้ ใช้ได้ดีกับ infinite scroll และ API ที่ส่ง cursor ส่วนหน้าที่ต้องมีเลขหน้าให้กดอาจต้องยอมใช้ OFFSET แต่จำกัดความลึก เงื่อนไขสำคัญของ keyset คือคอลัมน์ที่ใช้เรียงต้องรวมกันแล้ว ไม่ซ้ำ (ตัวอย่างนี้เติม order_id เป็น tie-breaker) และ row-value comparison (a, b) < (x, y) เป็นไวยากรณ์ของ PostgreSQL, MySQL และ SQLite ถ้า DBMS ไหนไม่รองรับ ต้องเขียนกระจายเป็น a < x OR (a = x AND b < y) ซึ่งบาง optimizer ใช้ index ได้ไม่ดีเท่า

Index ที่ออกแบบเฉพาะงาน

Partial index สร้างเฉพาะแถวที่ตรงเงื่อนไข เหมาะกับข้อมูลที่ query สนใจเพียงส่วนน้อย เช่น ออเดอร์ที่ยังค้างอยู่ ตัวอย่างนี้ index เล็กกว่า index เต็มของคอลัมน์เดียวกันมาก (ในข้อมูลทดสอบ 248 kB เทียบกับ 1,448 kB) แถมการ INSERT ที่ไม่ใช่ pending ไม่ต้องแตะ index นี้เลย ใช้ได้ใน PostgreSQL และ SQL Server (เรียกว่า filtered index) แต่ MySQL ไม่มี

CREATE INDEX idx_orders_pending ON orders (order_date) WHERE status = 'pending';

SELECT order_id FROM orders
WHERE status = 'pending' AND order_date >= '2026-06-01';      -- ต้องมี status = 'pending' ใน query ด้วย planner ถึงจะใช้ได้

Covering index ที่เห็นในหัวข้อ 6 ลดการกลับไปอ่านตาราง BRIN index (PostgreSQL) เหมาะกับตารางใหญ่ที่เพิ่มข้อมูลต่อท้ายตามเวลา เช่น log หรือ event เพราะเก็บแค่ค่าต่ำสุดและสูงสุดของแต่ละช่วงหน้า ขนาดเล็กมากเมื่อเทียบกับ B-tree แต่ได้ผลเฉพาะเมื่อค่าในคอลัมน์สัมพันธ์กับลำดับการเก็บจริงบนดิสก์ Expression index (index บนนิพจน์) ช่วยเมื่อต้องกรองด้วยค่าที่คำนวณ เช่น LOWER(email)

ทุก index แลกกับต้นทุนการเขียน ตารางที่รับ insert หลายพันครั้งต่อวินาทีและมี index สิบตัว จะเจอว่าการเขียนช้าลงอย่างเห็นได้ชัด ให้ตรวจ index ที่ไม่ถูกใช้และลบทิ้งเป็นระยะ

Partitioning

Partitioning แบ่งตารางใหญ่เป็นชิ้นย่อยตามช่วงค่า (เช่นรายเดือน) หรือ hash หรือ list ฐานข้อมูลมองเป็นตารางเดียว แต่เก็บแยกกัน ประโยชน์หลักสามข้อคือ partition pruning ที่ query ซึ่งกรองด้วยคีย์ที่แบ่งจะอ่านแค่ชิ้นที่เกี่ยวข้อง การลบข้อมูลเก่าทำได้ด้วยการ drop หรือ detach ชิ้นนั้นทั้งก้อนแทนที่จะ DELETE ทีละล้านแถว และ index แต่ละชิ้นเล็กลง

งานเขียนข้อมูลจำนวนมาก

คิดเป็นเซต ไม่ใช่ทีละแถว loop ที่ UPDATE ทีละแถวจากแอป หรือ cursor ใน stored procedure ช้ากว่าคำสั่งเดียวที่ทำทั้งเซตหลายเท่า เพราะทุกแถวต้องเสียค่า round trip และค่า parse/plan ซ้ำ ถ้าจำเป็นต้องวน ให้วนเป็นก้อน (batch) ไม่ใช่ทีละแถว

แต่ถ้าก้อนใหญ่เกินก็มีปัญหา UPDATE หรือ DELETE ที่แตะหลายสิบล้านแถวใน transaction เดียวจะถือ lock นาน สร้าง transaction log ปริมาณมาก ทำให้ replica ตามไม่ทัน และใน PostgreSQL ทิ้งแถวตายไว้ให้ vacuum เก็บกวาด วิธีที่ใช้กันคือแบ่งเป็นรอบ รอบละหนึ่งหมื่นถึงหนึ่งแสนแถว (ปรับตามเครื่อง) และ commit ทุกรอบ ตัวอย่าง procedure ที่ย้ายออเดอร์เก่าเข้าตาราง archive เป็นรอบ ๆ อยู่ในหัวข้อ 10

โหลดข้อมูลเข้าด้วยเครื่องมือเฉพาะ COPY ของ PostgreSQL, LOAD DATA ของ MySQL, BULK INSERT ของ SQL Server เร็วกว่า INSERT ทีละแถวมาก ถ้ายังต้องใช้ INSERT ให้รวมหลายแถวในคำสั่งเดียว (VALUES (...), (...), ...) และห่อด้วย transaction เดียว

N+1 query คืออาการที่แอปดึงรายการ 50 ออเดอร์ด้วย query หนึ่งครั้ง แล้ววนยิง query อีก 50 ครั้งเพื่อดึงลูกค้าของแต่ละออเดอร์ รวมเป็น 51 query มักเกิดจาก ORM แบบ lazy loading แก้ด้วย JOIN หรือดึงรวมด้วย WHERE customer_id IN (...) ครั้งเดียว ฝั่ง GraphQL แก้ด้วย DataLoader

ทักษะที่ต่างระหว่างคนที่ "เขียน SQL เป็น" กับคนที่ "จูน SQL เป็น" คือการอ่าน execution plan ได้อย่างมีหลักการ ไม่ใช่ไล่ดูว่ามี Seq Scan หรือเปล่าแล้วแปะ index เพิ่ม หัวข้อนี้เน้นวิธีหาว่า query ไหนควรแก้ก่อน อ่านตัวเลขใน plan ให้เป็น และเครื่องมือของแต่ละ DBMS ตัวอย่างหลักรันจริงบน PostgreSQL 16 ส่วนของ MySQL และ SQL Server เขียนตามเอกสารทางการ ไม่ได้รันในเครื่องที่ใช้เตรียมเนื้อหานี้

-- PostgreSQL 13 ขึ้นไป (เวอร์ชันเก่ากว่าใช้ชื่อคอลัมน์ total_time / mean_time)

       round(total_exec_time::numeric, 1) AS total_ms,
       round(mean_exec_time::numeric, 2)  AS mean_ms,
       rows,
       left(query, 80)                    AS query
FROM pg_stat_statements
ORDER BY total_exec_time DESC
LIMIT 10;

### 1. ประมาณผิด: ตัวเลขประมาณต่างจากของจริงมาก

นี่คือต้นเหตุของ plan แย่ ๆ จำนวนมาก planner เลือก Nested Loop เพราะคิดว่าเหลือ 10 แถว แต่ความจริงมี 1,000 แถว ทุกอย่างต่อจากนั้นผิดตามกันหมด ตัวอย่างที่สร้างได้ง่ายคือสองคอลัมน์ที่ขึ้นต่อกัน

-- ตาราง corr 100,000 แถว คอลัมน์ a และ b ได้ค่าเท่ากันเสมอ (g % 100)
EXPLAIN ANALYZE SELECT * FROM corr WHERE a = 5 AND b = 5;
 Seq Scan on corr  (cost=0.00..2076.00 rows=11 width=12) (actual rows=1000 loops=1)
   Filter: ((a = 5) AND (b = 5))
   Rows Removed by Filter: 99000

คาด 11 แถว ได้จริง 1,000 แถว (ต่างกันราว 90 เท่า) เพราะ planner สมมติว่า a = 5 และ b = 5 เป็นอิสระต่อกัน จึงเอาความน่าจะเป็นมาคูณกัน (1% × 1% ของ 100,000 ≈ 10) ในความจริงทั้งสองคอลัมน์ขึ้นต่อกันโดยสมบูรณ์ ทางแก้ใน PostgreSQL 10 ขึ้นไปคือบอกให้เก็บสถิติข้ามคอลัมน์

ANALYZE corr;

ตอนนี้ประมาณ 1,020 ใกล้ความจริงมาก สาเหตุอื่นที่ทำให้ประมาณผิดมีอีกหลายอย่าง เช่น statistics เก่า (ตารางเพิ่งโหลดข้อมูลใหม่แล้วยังไม่ได้ ANALYZE, autovacuum ยังตามไม่ทัน), ข้อมูลกระจายไม่สม่ำเสมอ (ค่าบางค่าพบบ่อยมาก), การกรองด้วยนิพจน์ที่ planner ไม่มีสถิติ และ parameter ที่ planner เดาค่าไม่ได้

### 2. อ่านเกินจำเป็น

Rows Removed by Filter: 199992 ในหัวข้อ 6 คือตัวอย่าง ให้ดูสัดส่วนระหว่างแถวที่อ่านกับแถวที่ได้ ถ้าอ่านมาหมื่นแต่ได้หนึ่ง index ที่เหมาะสมน่าจะช่วย ถ้าอ่านมาหมื่นแล้วได้เก้าพัน แปลว่า query ต้องการข้อมูลเยอะอยู่แล้ว index ไม่ช่วยอะไร

### 4. Plan ที่เปลี่ยนตามค่า parameter

query เดียวกัน ค่าต่างกัน plan ต่างกันได้ ลูกค้ารายใหญ่ที่มี 500,000 ออเดอร์ควรใช้ plan คนละแบบกับลูกค้ารายเล็กที่มี 3 ออเดอร์ ปัญหาเกิดเมื่อ plan ถูกเก็บไว้ใช้ซ้ำ (cache) จาก parameter ครั้งแรกซึ่งอาจเป็นรายเล็ก SQL Server เรียกอาการนี้ว่า parameter sniffing และแก้ได้หลายทาง เช่น OPTION (RECOMPILE), OPTIMIZE FOR หรือดู plan ที่ต่างกันใน Query Store ส่วน PostgreSQL ที่ใช้ prepared statement จะลอง custom plan ห้าครั้งแรก แล้วตัดสินใจเลือกระหว่าง custom กับ generic plan เอง และปรับพฤติกรรมได้ด้วย plan_cache_mode (PostgreSQL 12 ขึ้นไป)

อ่าน plan ใน MySQL และ SQL Server

MySQL คำสั่ง EXPLAIN แสดงตารางที่คอลัมน์สำคัญคือ type (วิธีเข้าถึงตาราง เรียงจากดีไปแย่ ได้แก่ const, eq_ref, ref, range, index, ALL โดย ALL คือ scan ทั้งตาราง), key (index ที่ใช้จริง), rows (จำนวนแถวที่ประมาณ) และ Extra ที่ควรสังเกต Using filesort (เรียงเองโดยไม่ได้อาศัย index), Using temporary (สร้างตารางชั่วคราว), Using index (ตอบจาก index อย่างเดียว) ตั้งแต่ MySQL 8.0.18 มี EXPLAIN ANALYZE ที่รันจริงและรายงานเวลาและจำนวนแถวจริงเป็นรูปแบบ tree ส่วน EXPLAIN FORMAT=TREE ให้ภาพรวมแบบ tree โดยไม่รัน

SQL Server ใช้ Actual Execution Plan ใน SSMS (Ctrl+M แล้วรัน query) ซึ่งแสดงเป็นแผนภาพ ให้ดูลูกศรที่หนาที่สุด (แถวมาก), ตัวเลข estimated เทียบ actual rows, Index Seek (เดินตาม index ตรง ๆ ดี) กับ Index Scan (อ่านทั้ง index), และ Key Lookup ที่เกิดจาก index ไม่ครอบคลุมคอลัมน์ที่ต้องการ ต้องกลับไปอ่านตารางทีละแถว ซึ่งถ้าเกิดซ้ำหลายหมื่นครั้งจะช้ามาก แก้ได้ด้วย INCLUDE ใน index นอกจากนี้ SET STATISTICS IO, TIME ON แสดงจำนวน logical reads และเวลา ส่วนคำแนะนำ "missing index" ที่ plan แสดงให้ ควรอ่านเป็นข้อมูลประกอบ ไม่ใช่คำสั่ง เพราะมันไม่ได้คิดเรื่องต้นทุนการเขียนหรือ index ที่มีอยู่แล้วซ้ำซ้อน

ขั้นตอนที่ใช้ได้จริง

  1. หา query ที่คุ้มค่าที่สุดที่จะแก้ จากเวลารวม ไม่ใช่ความรู้สึก
  2. ทำซ้ำปัญหาด้วยข้อมูลขนาดใกล้ของจริง รัน EXPLAIN (ANALYZE, BUFFERS) ดูหลายรอบ (ครั้งแรกอาจเป็น cache เย็น)
  3. หาโหนดที่ตัวเลขใหญ่ผิดปกติ ให้ดูจากล่างขึ้นบน โหนดที่ใช้เวลาจริงเยอะ โหนดที่ actual เทียบ estimate ต่างกันมาก โหนดที่ Rows Removed สูง
  4. ตั้งสมมติฐานทีละข้อ (statistics เก่า, ขาด index, เงื่อนไขไม่ sargable, join order, หน่วยความจำ) และเปลี่ยนทีละอย่าง
  5. วัดซ้ำและเทียบกับผลเดิม แล้วจดผลไว้ ถ้าแก้แล้วไม่ดีขึ้นก็ย้อนกลับ
  6. ป้องกันไม่ให้ถอยหลัง เช่นเพิ่ม query ที่สำคัญเข้าชุดทดสอบ performance หรือตั้ง alert จาก statistics ที่เก็บไว้
ถ้าต้องตอบสั้น ๆ ในข้อสอบเปรียบเทียบ estimated rows กับ actual rows ก่อน ถ้าต่างกันมากให้แก้ statistics ถ้าตรงกันแต่ช้า ให้ดูว่าอ่านแถวมากเกินจำเป็นหรือเปล่า (Rows Removed, index ไม่ครอบคลุม, เงื่อนไขไม่ sargable) และถ้าเวลาหายไปกับ Sort หรือ Hash ที่ล้นลงดิสก์ ให้ดูเรื่องหน่วยความจำและ index ที่เรียงมาให้แล้ว

Advance

Write complex stored procedures, triggers, and user-defined functions for automation and business logic.

Function ที่ตรวจเงื่อนไขและรายงานผล

ตัวอย่างนี้จองสต็อกสินค้า จุดสำคัญคือไม่ได้อ่านจำนวนคงเหลือมาตรวจก่อนแล้วค่อยลด (เสี่ยง race condition ตามที่อธิบายในหัวข้อ 11) แต่ใส่เงื่อนไข qty_on_hand >= p_qty ไว้ใน UPDATE เองให้เป็นคำสั่งอะตอมมิกเดียว แล้วใช้ FOUND ตรวจว่ามีแถวถูกแก้หรือไม่

CREATE TABLE stock (
  product_id  int PRIMARY KEY,
  qty_on_hand int NOT NULL CHECK (qty_on_hand >= 0)
);
INSERT INTO stock VALUES (1, 5), (2, 0);

CREATE FUNCTION reserve_stock(p_product int, p_qty int) RETURNS boolean
LANGUAGE plpgsql AS $$
BEGIN
  IF p_qty <= 0 THEN
    RAISE EXCEPTION 'qty must be positive, got %', p_qty USING ERRCODE = '22023';
  END IF;
  UPDATE stock
     SET qty_on_hand = qty_on_hand - p_qty
   WHERE product_id = p_product AND qty_on_hand >= p_qty;
  RETURN FOUND;
END $$;

Function ที่คืนเป็นตารางเขียนได้ทั้งแบบ SQL ธรรมดาและ PL/pgSQL ตัวอย่างนี้ใช้แบบ SQL เพราะง่ายและ optimizer มีโอกาส inline เข้า query หลักได้

CREATE FUNCTION top_customers(p_limit int DEFAULT 3)
RETURNS TABLE (customer_id int, spent numeric)
LANGUAGE sql STABLE AS $$
  SELECT o.customer_id, SUM(oi.qty * oi.unit_price)
  FROM orders o JOIN order_items oi ON oi.order_id = o.order_id
  WHERE o.status IN ('paid', 'shipped')
  GROUP BY o.customer_id
  ORDER BY 2 DESC, 1
  LIMIT p_limit
$$;

SELECT * FROM top_customers(2);      -- customer 1 (43100.00) และ 4 (33500.00)

คำว่า STABLE คือ volatility ของ function PostgreSQL แบ่งเป็นสามระดับ IMMUTABLE (ผลขึ้นกับพารามิเตอร์อย่างเดียวเสมอ เช่น ฟังก์ชันคำนวณทางคณิตศาสตร์), STABLE (ผลคงที่ภายใน statement เดียว เช่น อ่านตาราง) และ VOLATILE (อาจเปลี่ยนทุกครั้งที่เรียก เช่น random(), หรือมี side effect ค่าเริ่มต้น) ยิ่งบอกระดับที่เข้มถูกต้อง optimizer ยิ่งทำงานให้ได้มาก เช่น function ที่จะเอาไปสร้าง expression index ต้องเป็น IMMUTABLE เท่านั้น แต่ถ้าประกาศผิด (บอกว่า immutable ทั้งที่ไม่ใช่) ผลลัพธ์จะผิดแบบไม่มี error

Procedure ที่ทำงานเป็นรอบและ commit ระหว่างทาง

งานที่ต้องย้ายหรือลบข้อมูลเยอะ ๆ ไม่ควรทำใน transaction เดียว (เหตุผลดูหัวข้อ 8) ใน PostgreSQL 11 ขึ้นไป procedure ใช้ COMMIT ภายในได้ ต่างจาก function

CREATE PROCEDURE archive_old_orders(cutoff date, batch_size int DEFAULT 20000)
LANGUAGE plpgsql AS $$
DECLARE moved int;
BEGIN
  LOOP
    WITH moved_rows AS (
      DELETE FROM orders
      WHERE order_id IN (SELECT order_id FROM orders
                         WHERE order_date < cutoff
                         ORDER BY order_id LIMIT batch_size)
      RETURNING *
    )
    INSERT INTO orders_archive SELECT * FROM moved_rows;

    GET DIAGNOSTICS moved = ROW_COUNT;
    EXIT WHEN moved = 0;
    COMMIT;                                  -- ปล่อย lock และ log ของรอบนี้ ก่อนเริ่มรอบถัดไป
  END LOOP;
END $$;

CALL archive_old_orders('2025-07-01');

ในการทดสอบกับตารางสุ่ม 200,000 แถวชุดเดียวกับหัวข้อ 6 ผลคือย้ายไป 49,479 แถว (ทุกแถวที่ order_date ก่อน 2025-07-01) เหลืออยู่ 150,521 แถว รวมกันได้ 200,000 พอดี (ตัวเลขนี้มาจากข้อมูลสุ่มชุดนั้น สร้างใหม่แล้วจะต่างไปเล็กน้อย แต่ผลรวมต้องเท่าเดิมเสมอ) DELETE ... RETURNING ส่งแถวที่ลบไปให้ INSERT ในคำสั่งเดียว จึงไม่มีช่วงที่แถวหายไปจากทั้งสองตาราง ข้อควรจำคือถ้า procedure ถูกเรียกจากใน transaction block ที่เปิดไว้แล้ว (BEGIN; CALL ...;) จะสั่ง COMMIT ภายในไม่ได้ และถ้า procedure ล้มกลางทาง รอบที่ commit ไปแล้วจะไม่ย้อนกลับ ต้องออกแบบให้รันซ้ำได้อย่างปลอดภัย (ตัวอย่างนี้รันซ้ำได้ เพราะรอบที่ย้ายไปแล้วจะไม่เจอแถวนั้นอีก)

Trigger

Trigger คือฟังก์ชันที่ฐานข้อมูลเรียกให้เองเมื่อเกิดเหตุการณ์กับตาราง (INSERT, UPDATE, DELETE) แบ่งตามจังหวะเป็น BEFORE ที่ทำก่อนแถวถูกเขียนและแก้ค่าแถวใหม่ได้ หรือสั่งยกเลิกได้ กับ AFTER ที่ทำหลังเขียนแล้ว เหมาะกับการบันทึกประวัติหรือแจ้งเหตุการณ์ และแบ่งตามขอบเขตเป็น row-level (รันทีละแถว) กับ statement-level (รันครั้งเดียวต่อคำสั่ง)

ตัวอย่างต่อไปใช้ trigger สองตัวกับตารางพนักงาน ตัวแรกแบบ BEFORE ห้ามลดเงินเดือนเกิน 10% และอัปเดตเวลาแก้ไขให้อัตโนมัติ ตัวที่สองแบบ AFTER เขียนประวัติลงตาราง audit

CREATE TABLE emp (
  emp_id int PRIMARY KEY, salary numeric(10,2) NOT NULL,
  updated_at timestamptz NOT NULL DEFAULT now()
);
CREATE TABLE salary_audit (
  audit_id bigserial PRIMARY KEY, emp_id int, old_salary numeric, new_salary numeric,
  changed_by text DEFAULT current_user, changed_at timestamptz DEFAULT now()
);

CREATE FUNCTION trg_emp_before() RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
  IF NEW.salary < OLD.salary * 0.9 THEN
    RAISE EXCEPTION 'salary cut of more than 10%% is not allowed (emp %)', OLD.emp_id;
  END IF;
  NEW.updated_at := now();
  RETURN NEW;                                    -- BEFORE: ต้อง return แถวที่จะถูกเขียน
END $$;
CREATE TRIGGER emp_before_update BEFORE UPDATE ON emp
  FOR EACH ROW EXECUTE FUNCTION trg_emp_before();

CREATE FUNCTION trg_emp_audit() RETURNS trigger LANGUAGE plpgsql AS $$
BEGIN
  IF NEW.salary IS DISTINCT FROM OLD.salary THEN
    INSERT INTO salary_audit (emp_id, old_salary, new_salary)
    VALUES (OLD.emp_id, OLD.salary, NEW.salary);
  END IF;
  RETURN NULL;                                   -- AFTER row trigger: ค่าที่ return ถูกมองข้าม
END $$;
CREATE TRIGGER emp_after_update AFTER UPDATE ON emp
  FOR EACH ROW EXECUTE FUNCTION trg_emp_audit();

UPDATE emp SET salary = salary * 1.1 WHERE emp_id = 1;     -- 50000 → 55000 ผ่าน และมีแถวใน salary_audit
UPDATE emp SET salary = 30000 WHERE emp_id = 2;            -- ERROR: salary cut of more than 10% is not allowed (emp 2)

ผลที่รันได้คือการขึ้นเงินเดือน emp 1 ผ่านและมีแถวใน salary_audit เป็น (1, 50000.00, 55000.00) ส่วนการลดของ emp 2 ถูกปฏิเสธ และเงินเดือนยังเป็น 60000 เหมือนเดิม สังเกต IS DISTINCT FROM ที่ใช้เทียบเพราะมันจัดการ NULL ได้ถูกต้อง ถ้าใช้ <> และเงินเดือนเดิมเป็น NULL ผลจะเป็น UNKNOWN และ trigger จะไม่บันทึก

### ใช้ trigger เมื่อไร และอย่าใช้เมื่อไร

เหมาะกับงานที่ต้องเกิดแน่นอนไม่ว่าใครแก้ข้อมูลจากทางไหน ได้แก่ audit trail, คอลัมน์ที่คำนวณอัตโนมัติ เช่น updated_at, และกติกาข้ามตารางที่ CHECK ทำไม่ได้ ส่วนที่ต้องระวัง

  • ตรรกะซ่อนอยู่ นักพัฒนาที่ UPDATE ตารางไม่รู้ว่ามี trigger อีกสามตัวทำงานตาม ถ้า trigger ไปแก้ตารางอื่นที่มี trigger ต่อ อาจเป็นสายโซ่ที่ debug ยาก หรือวนซ้ำไม่จบ
  • ต้นทุนต่อแถว row-level trigger ทำงานทุกแถว UPDATE หนึ่งล้านแถวจึงรัน function หนึ่งล้านครั้ง ถ้าตรรกะทำได้ทีเดียวทั้งเซต ให้ใช้ statement-level trigger (PostgreSQL มี transition table ให้อ่านแถวที่เปลี่ยนทั้งก้อนด้วย REFERENCING NEW TABLE AS ...)
  • งานโหลดข้อมูลปริมาณมาก ต้องรู้ว่าจะปิด trigger ชั่วคราวได้หรือไม่ และมีผลต่อ audit อย่างไร
  • ลำดับการทำงาน ถ้าตารางมี trigger หลายตัวในเหตุการณ์เดียวกัน PostgreSQL เรียงตามชื่อตัวอักษร ส่วน MySQL เรียงตามลำดับที่สร้าง และกำหนดด้วย FOLLOWS / PRECEDES ได้

11

Transaction, Concurrency และ Locking

Advance

Manage transactions, concurrency, and locking for reliable data consistency and integrity.

ถ้ามีผู้ใช้คนเดียว SQL ที่เขียนถูกก็ให้ผลถูกเสมอ แต่ระบบจริงมีคนใช้พร้อมกัน และคำสั่งสองชุดที่ถูกต้องเมื่อรันทีละชุดอาจให้ผลผิดเมื่อรันพร้อมกัน หัวข้อนี้ว่าด้วยการรักษาความถูกต้องของข้อมูลในสถานการณ์นั้น ตัวอย่างที่เกี่ยวกับพฤติกรรมพร้อมกันรันจริงด้วย psql สองหน้าต่างบน PostgreSQL 16 แต่พฤติกรรมเริ่มต้นของ isolation level ต่างกันมากระหว่าง DBMS จึงมีตารางเทียบไว้ให้

Transaction และ ACID

Transaction คือชุดคำสั่งที่ฐานข้อมูลรับประกันว่าจะสำเร็จทั้งหมดหรือไม่เกิดเลย ตัวอย่างโอนเงินที่ต้องหักบัญชีหนึ่งและเพิ่มอีกบัญชีหนึ่งต้องไปพร้อมกัน

BEGIN;
UPDATE accounts SET balance = balance - 500 WHERE id = 1;
UPDATE accounts SET balance = balance + 500 WHERE id = 2;
COMMIT;            -- ถ้ามีอะไรผิดก่อนถึงตรงนี้ ใช้ ROLLBACK ย้อนทั้งหมด

ACID คือคุณสมบัติสี่ข้อที่ transaction รับประกัน

ตัวอักษรความหมายตัวอย่างจากการโอนเงิน
Atomicityทำทั้งหมดหรือไม่ทำเลยหักเงินแล้วระบบล่ม เงินต้องไม่หายครึ่งทาง
Consistencyหลังจบ transaction ข้อมูลต้องผ่าน constraint ทั้งหมดยอดเงินติดลบไม่ได้ถ้ามี CHECK

Durabilityเมื่อ commit แล้วข้อมูลต้องคงอยู่แม้ไฟดับฐานข้อมูลเขียน transaction log ลงดิสก์ก่อนตอบว่า commit สำเร็จ

คำถามที่ถามบ่อยคือ "autocommit คืออะไร" ปกติ client ส่วนใหญ่รันแต่ละคำสั่งเป็น transaction ของตัวเอง ซึ่ง commit ให้ทันทีหลังจบคำสั่ง ถ้าอยากรวมหลายคำสั่งเป็นก้อนเดียว ต้อง BEGIN (หรือ START TRANSACTION) เอง

ปัญหาเมื่อรันพร้อมกัน (anomaly)

มาตรฐาน SQL นิยามปัญหาสามอย่างที่ isolation level ใช้อธิบาย บวกอีกสองอย่างที่พบในงานจริง

Non-repeatable readอ่านแถวเดิมสองครั้งใน transaction เดียว ได้ค่าไม่เท่ากัน เพราะมีคนอื่น commit การแก้ไขคั่นกลาง

Lost updateสอง transaction อ่านค่าเดียวกัน คำนวณแล้วเขียนทับกัน การเปลี่ยนแปลงของคนแรกหาย
Write skewสอง transaction อ่านข้อมูลชุดเดียวกัน แล้วต่างคนต่างแก้แถวคนละแถว ผลรวมแล้วละเมิดกติกาที่แต่ละคนตรวจไว้

### ดู Non-repeatable read ด้วยตา

เปิดสอง session session A อ่านยอดสองครั้งใน transaction เดียว ระหว่างนั้น session B แก้ค่าและ commit

-- Session A (Read Committed)                      -- Session B
BEGIN;
SELECT bal FROM acct WHERE id = 1;     -- 100
                                                   UPDATE acct SET bal = 150 WHERE id = 1;  -- autocommit
SELECT bal FROM acct WHERE id = 1;     -- 150   ← ต่างจากครั้งแรก
COMMIT;

ถ้าเปลี่ยน session A เป็น BEGIN ISOLATION LEVEL REPEATABLE READ; อ่านครั้งที่สองจะได้ 100 เหมือนเดิม เพราะใช้ snapshot ตั้งแต่ตอนเริ่ม แต่มีราคาที่ต้องจ่าย ถ้า A พยายามแก้แถวเดียวกันที่ B เพิ่งแก้ไปหลัง snapshot ของ A จะได้ error

ERROR:  could not serialize access due to concurrent update

และแอปต้องเตรียมรับ error ชนิดนี้ แล้วลองทำ transaction ใหม่ทั้งก้อน

Lost update และวิธีแก้

รูปแบบที่ผิดคืออ่านค่ามาที่แอป คำนวณ แล้วเขียนกลับ

-- แอป: SELECT qty FROM stock WHERE product_id = 1;  → 5
--      (คำนวณ 5 - 1 = 4 ในโค้ด)
-- แอป: UPDATE stock SET qty = 4 WHERE product_id = 1;

ถ้าสองคำขอทำพร้อมกัน ทั้งคู่อ่านได้ 5 ทั้งคู่เขียน 4 สต็อกถูกหักแค่ครั้งเดียวทั้งที่ขายสองชิ้น มีสามวิธีแก้ เลือกตามสถานการณ์

1. ให้ฐานข้อมูลคำนวณเอง ด้วย UPDATE เดียว เป็นวิธีที่ง่ายและเร็วที่สุด ถ้างานนั้นทำได้ในคำสั่งเดียว

UPDATE stock
   SET qty = qty - 1
 WHERE product_id = 1 AND qty >= 1;        -- ตรวจจำนวนแถวที่ได้: 0 แปลว่าของหมด

การอ่านค่าปัจจุบันและเขียนค่าใหม่เกิดเป็นขั้นตอนเดียวในแถวที่ถูกล็อก คนที่สองจะรอจนคนแรกจบ แล้วเห็นค่าล่าสุด

2. ล็อกแถวที่อ่านด้วย SELECT ... FOR UPDATE เมื่อต้องอ่านแล้วตัดสินใจด้วยตรรกะซับซ้อนก่อนเขียน

BEGIN;
SELECT qty FROM stock WHERE product_id = 1 FOR UPDATE;    -- ล็อกแถวนี้ คนอื่นที่จะแก้ต้องรอ
-- ... ตัดสินใจในแอป ...
UPDATE stock SET qty = qty - 1 WHERE product_id = 1;
COMMIT;

ข้อควรระวังคือ transaction ต้องสั้น ห้ามถือล็อกค้างไว้ระหว่างรอผู้ใช้กดปุ่มหรือเรียก API ภายนอก

3. Optimistic locking ด้วยคอลัมน์ version ไม่ล็อกอะไรตอนอ่าน แต่ตอนเขียนตรวจว่าแถวยังเป็นเวอร์ชันเดิมหรือไม่

UPDATE product
   SET price = 1300, version = version + 1
 WHERE product_id = 1 AND version = 7;     -- 0 แถว = มีคนแก้ไปก่อนแล้ว แอปต้องโหลดใหม่และให้ผู้ใช้ตัดสินใจ

เหมาะกับระบบที่ผู้ใช้เปิดฟอร์มแก้ข้อมูลทิ้งไว้นาน และโอกาสชนกันน้อย ซึ่งเป็นแนวเดียวกับ ETag กับ If-Match ใน REST API

Write skew และ Serializable

ตัวอย่างคลาสสิก โรงพยาบาลมีกฎว่าต้องมีแพทย์เวรอย่างน้อยหนึ่งคน แพทย์สองคนที่เวรอยู่ต่างกดขอลาพร้อมกัน แต่ละ transaction ตรวจว่า "ตอนนี้มีเวรอยู่ 2 คน เหลืออีกคนก็พอ" แล้วแก้แถวของตัวเอง ไม่มีแถวไหนถูกแก้ซ้ำกัน จึงไม่มี lock ชนกัน ผลคือไม่เหลือแพทย์เวรเลย

ลองใน PostgreSQL ที่ Repeatable Read ผลคือทั้งสอง transaction สำเร็จและตาราง doctor กลายเป็น on_call = false ทั้งคู่ แต่เมื่อเปลี่ยนเป็น Serializable ตัวหนึ่งสำเร็จและอีกตัวถูกปฏิเสธ

ERROR:  could not serialize access due to read/write dependencies among transactions
HINT:  The transaction might succeed if retried.

Serializable คือการรับประกันว่าผลเหมือนรันทีละ transaction ตามลำดับใดลำดับหนึ่ง แต่ฐานข้อมูลทำได้โดยปฏิเสธบาง transaction ดังนั้นแอปต้องมี retry loop ที่ลองทำ transaction ใหม่เมื่อเจอ error serialization failure (SQLSTATE 40001) และ deadlock (40P01 ใน PostgreSQL) เป็นเงื่อนไขปกติของระบบ ไม่ใช่ข้อยกเว้น ทางเลือกอื่นของ write skew คือล็อกแถวที่ใช้ตรวจเงื่อนไขทั้งหมดด้วย FOR UPDATE หรือเก็บตัวนับรวมไว้ในแถวเดียวที่ทุกคนต้องแก้ร่วมกัน

Deadlock

Deadlock เกิดเมื่อ transaction สองตัวถือ lock ที่อีกฝ่ายต้องการ และต่างรออีกฝ่ายอยู่ ไม่มีใครไปต่อได้

Session 1: UPDATE acct ... WHERE id = 1;   (ถือ lock แถว 1)
Session 2: UPDATE acct ... WHERE id = 2;   (ถือ lock แถว 2)
Session 1: UPDATE acct ... WHERE id = 2;   (รอ session 2)
Session 2: UPDATE acct ... WHERE id = 1;   (รอ session 1)  ← วงจรปิด

ERROR:  deadlock detected
DETAIL:  Process 435 waits for ShareLock on transaction 800; blocked by process 436.
         Process 436 waits for ShareLock on transaction 799; blocked by process 435.

ฐานข้อมูลตรวจเจอเอง (PostgreSQL ตรวจหลังรอเกิน deadlock_timeout ซึ่งค่าเริ่มต้นหนึ่งวินาที) แล้วเลือกยกเลิกหนึ่งตัวให้ ที่เหลือจึงไปต่อได้ ผลที่ได้จริงในการทดสอบคือฝั่งหนึ่งสำเร็จ อีกฝั่งโดนยกเลิก วิธีป้องกันตามลำดับที่ได้ผลที่สุด

  1. เรียงลำดับการล็อกให้เหมือนกันทุกที่ เช่น โอนเงินระหว่างสองบัญชีให้แก้บัญชีที่ id น้อยกว่าก่อนเสมอ ไม่ว่าใครเป็นผู้โอน
  2. ทำ transaction ให้สั้น ยิ่งถือ lock นานโอกาสชนยิ่งสูง ห้ามมีการเรียกภายนอกหรือรอผู้ใช้ข้างใน
  3. ล็อกให้ครบตั้งแต่ต้น ด้วย SELECT ... FOR UPDATE ORDER BY id ก่อนแก้
  4. ใส่ retry เมื่อโดนยกเลิก ให้รันทั้ง transaction ใหม่

Upsert แบบปลอดภัยเมื่อรันพร้อมกัน

รูปแบบ "ถ้ามีแล้วแก้ ถ้าไม่มีก็สร้าง" ที่เขียนเป็น SELECT ตามด้วย INSERT หรือ UPDATE เสี่ยงต่อ race condition เพราะสองคำขออาจเห็นว่ายังไม่มีแล้วต่างคน INSERT ผลคือชนกัน ให้ใช้คำสั่งที่ฐานข้อมูลทำเป็นขั้นเดียว

-- PostgreSQL / SQLite
INSERT INTO page_views (page_id, views) VALUES (10, 1)
ON CONFLICT (page_id) DO UPDATE SET views = page_views.views + 1;

-- MySQL
INSERT INTO page_views (page_id, views) VALUES (10, 1)
ON DUPLICATE KEY UPDATE views = views + 1;

-- SQL Server, Oracle, PostgreSQL 15 ขึ้นไป: MERGE (ใน SQL Server ต้องระวังเรื่อง concurrency ให้ใส่ HOLDLOCK)

คำสั่งแบบนี้ต้องพึ่ง unique constraint หรือ primary key บน page_id เพื่อให้ฐานข้อมูลรู้ว่าอะไรคือ "ซ้ำ"

ปัญหา transaction ที่เปิดค้าง

Transaction ที่เปิดแล้วไม่ปิดเป็นปัญหาที่เจอบ่อยในระบบจริง (เช่น แอป crash กลางทาง หรือ connection pool ที่ลืม commit) มันถือ lock ไว้ ทำให้คำสั่งอื่นต้องรอต่อแถว และใน PostgreSQL ทำให้ vacuum เก็บกวาดแถวตายไม่ได้ ตารางจึงบวมขึ้นเรื่อย ๆ ส่วน InnoDB ของ MySQL ต้องเก็บ undo log ของเวอร์ชันเก่าไว้ ตัวช่วยที่ควรตั้งคือ idle_in_transaction_session_timeout และ statement_timeout ใน PostgreSQL กับการดู pg_stat_activity เพื่อหา session ที่ state = 'idle in transaction' นาน ๆ

Savepoint

SAVEPOINT ให้ย้อนกลับเฉพาะบางส่วนของ transaction โดยไม่ต้องยกเลิกทั้งก้อน ใช้เมื่อบางขั้นอนุญาตให้ล้มได้ เช่น พยายามแทรกข้อมูลแล้วถ้าซ้ำก็ข้าม

BEGIN;
INSERT INTO orders (...) VALUES (...);
SAVEPOINT before_bonus;
INSERT INTO bonus_points (...) VALUES (...);     -- สมมติว่าล้มเพราะข้อมูลซ้ำ
ROLLBACK TO SAVEPOINT before_bonus;              -- ย้อนเฉพาะบรรทัดข้างบน ออเดอร์ยังอยู่
COMMIT;

ใน PostgreSQL ถ้าคำสั่งใดใน transaction ล้ม ตัว transaction ทั้งก้อนจะอยู่ในสถานะ aborted ไปต่อไม่ได้จนกว่าจะ ROLLBACK (หรือ ROLLBACK TO SAVEPOINT) ซึ่งต่างจาก DBMS บางตัวที่ยอมให้ทำต่อ ข้อนี้เป็นเหตุผลที่ ORM บางตัวใช้ savepoint ครอบคำสั่งเสี่ยงโดยอัตโนมัติ

เช็กลิสต์ตอนออกแบบ transactionทำให้สั้น ทำให้ครบในหน่วยเดียว (ไม่แบ่งหนึ่งงานทางธุรกิจเป็นหลาย transaction ที่ไม่มีอะไรรับประกันความต่อเนื่อง) ใส่เงื่อนไขไว้ใน UPDATE แทนอ่านแล้วค่อยเขียน ล็อกตามลำดับเดียวกัน เลือก isolation level ที่ต่ำที่สุดที่ปลอดภัยพอสำหรับงานนั้น และเตรียม retry ไว้เสมอ

COUNT(*) กับ COUNT(col)COUNT(*) นับทุกแถว COUNT(col) ข้ามแถวที่ col เป็น NULL และ COUNT(DISTINCT col) ข้ามทั้ง NULL และค่าซ้ำ
UNION กับ UNION ALLUNION ตัดแถวซ้ำ (ต้องเรียงหรือ hash) UNION ALL ต่อทุกแถว เร็วกว่าและเก็บแถวซ้ำไว้
เงื่อนไขใน ON กับ WHERE ของ LEFT JOINON ตัดสินว่าจับคู่ได้ไหม แถวฝั่งซ้ายยังอยู่ WHERE ที่อ้างคอลัมน์ฝั่งขวากรองแถว NULL ทิ้ง ทำให้กลายเป็น INNER JOIN
IN กับ EXISTSมักได้ plan เดียวกัน (semi-join) แต่ NOT IN กับ NOT EXISTS ต่างกันเมื่อมี NULL: NOT IN ที่ subquery มี NULL ได้ 0 แถว
ROW_NUMBER, RANK, DENSE_RANKROW_NUMBER ไม่ซ้ำเสมอ RANK ให้อันดับเท่ากันแล้วข้ามเลข DENSE_RANK ให้อันดับเท่ากันแล้วไม่ข้ามเลข
PRIMARY KEY กับ UNIQUEPK ห้ามเป็น NULL และมีได้ชุดเดียว UNIQUE มีได้หลายชุดและยอมให้มี NULL (ต่างกันตาม DBMS)
DELETE, TRUNCATE, DROPDELETE ลบแถวตามเงื่อนไข TRUNCATE ล้างทั้งตาราง DROP ลบตารางทั้งโครงสร้าง การย้อนกลับของ TRUNCATE ต่างกันตาม DBMS
EXPLAIN กับ EXPLAIN ANALYZEEXPLAIN แสดงแผนโดยประมาณและไม่รันคำสั่ง EXPLAIN ANALYZE รันจริง (แก้ข้อมูลจริงถ้าเป็น UPDATE/DELETE) และแสดงเวลาจริง
FUNCTION กับ PROCEDURE (PostgreSQL)function เรียกในนิพจน์ SQL และสั่ง COMMIT ไม่ได้ procedure เรียกด้วย CALL และควบคุม transaction ภายในได้
READ COMMITTED, REPEATABLE READ, SERIALIZABLEต่างกันที่ snapshot: แต่ละคำสั่งใช้ snapshot ใหม่, ทั้ง transaction ใช้ snapshot เดียว, และตรวจ write skew ด้วยการยกเลิกบาง transaction
Optimistic กับ pessimistic lockingoptimistic ไม่ล็อกตอนอ่านแล้วตรวจ version ตอนเขียน pessimistic ล็อกแถวไว้ตั้งแต่ตอนอ่านด้วย FOR UPDATE

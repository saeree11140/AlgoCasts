ลองดูตัวอย่างก่อนและหลังปรับ

ฟังก์ชันแรกด้านล่างคือแบบที่เจอบ่อยในโค้ดเก่า ผลลัพธ์ถูก แต่ไม่รู้ว่า c คืออะไร และ 0.9 กับ 0.8 คือส่วนลดหรือราคาที่เหลือ ฟังก์ชันที่สองพฤติกรรมเหมือนกัน แต่ชื่อบอกความหมายครบ และถ้าจะเพิ่มระดับสมาชิกใหม่ก็เพิ่มบรรทัดเดียวในตาราง ไม่ต้องไปแตะ if

// ก่อน: อ่านไม่ออกว่า c คืออะไร และ 0.9 มาจากไหน
function calc(a, b, c) {
  var r = 0
  if (c == 1) { r = a * b * 0.9 }
  else if (c == 2) { r = a * b * 0.8 }
  else { r = a * b }
  return r
}

// หลัง: ชื่อบอกความหมาย ส่วนลดอยู่ที่เดียว
const DISCOUNT_RATE_BY_TIER = { 1: 0.1, 2: 0.2 };

function calculateTotal(unitPrice, quantity, customerTier) {
  const discountRate = DISCOUNT_RATE_BY_TIER[customerTier] ?? 0;
  const subtotal = unitPrice * quantity;
  return subtotal * (1 - discountRate);
}

console.log(calc(100, 2, 1), calculateTotal(100, 2, 1));  // 180 180
console.log(calc(100, 3, 2), calculateTotal(100, 3, 2));  // 240 240
console.log(calc(100, 3, 9), calculateTotal(100, 3, 9));  // 300 300

จุดที่ควรสังเกตคือ ?? ใช้แทนค่าเมื่อเป็น null หรือ undefined เท่านั้น ถ้า tier ที่ไม่รู้จักได้ส่วนลด 0 ก็คือไม่ลด ซึ่งตรงกับพฤติกรรมเดิมของ else เรื่องนี้สำคัญตอนปรับโค้ด เพราะเป้าหมายคือหน้าตาโค้ดดีขึ้นโดยพฤติกรรมไม่เปลี่ยน

จัดรูปแบบให้สม่ำเสมอ และให้เครื่องมือทำแทนคน

เรื่อง indentation ว่าจะใช้ 2 หรือ 4 space หรือ tab ไม่มีคำตอบที่ถูกที่สุด สิ่งที่ผิดคือในไฟล์เดียวกันใช้ปนกัน ทีมส่วนใหญ่จึงไม่ถกเถียงกันเอง แต่ตั้งเครื่องมือให้ตัดสินแทน Prettier จัดรูปแบบอัตโนมัติ ESLint จับรูปแบบโค้ดที่เสี่ยง เช่น ใช้ == แทน === หรือประกาศตัวแปรแล้วไม่ได้ใช้ และ .editorconfig ทำให้ editor ของทุกคนตั้งค่าตรงกันตั้งแต่ต้น

// eslint.config.js (ESLint v9 ขึ้นไป ใช้ flat config และต้องติดตั้ง @eslint/js)
import js from "@eslint/js";

export default [
  js.configs.recommended,
  {
    rules: {
      eqeqeq: "error",       // บังคับ === แทน ==
      "no-var": "error",     // ใช้ let / const แทน var
      "prefer-const": "error" // ตัวแปรที่ไม่ถูกกำหนดค่าใหม่ ให้ใช้ const
    },
  },
];

ข้อสอบมักถามเรื่อง “standard-compliant” ในเชิงว่าโค้ดตามกติกาของภาษาและทีมหรือไม่ ตัวอย่างในฝั่ง JavaScript คือใช้ const เป็นค่าเริ่มต้นและใช้ let เมื่อต้องกำหนดค่าใหม่ ใช้ === และเขียนตามแนวทางที่ทีมใช้ เช่น Google JavaScript Style Guide ส่วนฝั่ง Python คือ PEP 8

ฟังก์ชันเล็ก ทำเรื่องเดียว และไม่ซ้อนลึก

ฟังก์ชันที่ยาวเป็นร้อยบรรทัดมักมี comment คั่นเป็นช่วง เช่น // validate, // calculate, // save ซึ่งเป็นสัญญาณว่าแต่ละช่วงควรเป็นฟังก์ชันของตัวเอง อีกเรื่องที่ช่วยให้อ่านง่ายขึ้นทันทีคือ early return จัดการกรณีผิดปกติก่อนแล้วออกจากฟังก์ชัน เส้นทางหลักจะได้อยู่ชิดซ้ายไม่ซ้อน if หลายชั้น

// ซ้อนลึก: ต้องไล่ดูว่า else นี้ของ if ตัวไหน
function getShippingFee(order) {
  if (order) {
    if (order.items.length > 0) {
      if (order.total >= 1000) {
        return 0;
      } else {
        return 50;
      }
    } else {
      return 0;
    }
  } else {
    throw new Error("order is required");
  }
}

const FREE_SHIPPING_THRESHOLD = 1000;
const STANDARD_SHIPPING_FEE = 50;

// early return: จัดการกรณีพิเศษก่อน แล้วเหลือเส้นทางหลักเส้นเดียว
function getShippingFeeFlat(order) {
  if (!order) throw new Error("order is required");
  if (order.items.length === 0) return 0;
  if (order.total >= FREE_SHIPPING_THRESHOLD) return 0;
  return STANDARD_SHIPPING_FEE;
}

const cart = { items: ["pen"], total: 250 };
console.log(getShippingFee(cart), getShippingFeeFlat(cart));                           // 50 50
console.log(getShippingFee({ items: [], total: 0 }), getShippingFeeFlat({ items: [], total: 0 })); // 0 0

Trade-off ที่ควรรู้

ชื่อยาวไม่ได้ดีกว่าเสมอไป ตัวแปรที่อยู่ในขอบเขตแคบ ๆ เช่น i ใน loop สั้น ๆ หรือ err ใน callback ใช้ชื่อสั้นได้เพราะบริบทชัดอยู่แล้ว ส่วนตัวแปรที่ใช้ข้ามหลายไฟล์ควรมีชื่อที่อธิบายตัวเองได้ การแตกฟังก์ชันก็มีจุดพอดี ถ้าแตกจนทุกฟังก์ชันมีสองบรรทัดและเรียกต่อกันเป็นสิบชั้น คนอ่านต้องกระโดดไปมาจนเสียเวลามากกว่าเดิม ให้ตัดสินจากว่าแต่ละฟังก์ชันตั้งชื่อได้ชัดและทดสอบแยกได้หรือไม่

สรุปสั้น ๆชื่อบอกว่าคืออะไรและทำอะไร boolean ตั้งเป็นเชิงบวก ใช้รูปแบบเดียวทั้งไฟล์และให้ Prettier กับ ESLint บังคับแทนคน ใช้ const กับ === ใช้ early return ลดการซ้อน แทนเลขลึกลับด้วยค่าคงที่ที่มีชื่อ และเขียน comment อธิบายเหตุผล ไม่ใช่เล่าซ้ำสิ่งที่โค้ดบอกอยู่แล้ว

function findMax(nums) {
  if (nums.length === 0) throw new RangeError("nums must not be empty");
  let max = nums[0];
  for (let i = 1; i < nums.length; i++) {
    if (nums[i] > max) max = nums[i];
  }
  return max;
}

Loop แบบไหนใช้เมื่อไร

เงื่อนไข: ลำดับที่เขียนมีผล

โจทย์ FizzBuzz เป็นตัวอย่างคลาสสิกของเรื่องนี้ เลขที่หารด้วย 15 ลงตัวก็หารด้วย 3 และ 5 ลงตัวด้วย ถ้าเช็ก % 3 ก่อน เลข 15 จะถูกจับเป็น Fizz และไม่มีทางไปถึงเงื่อนไข FizzBuzz กฎคือเขียนเงื่อนไขที่เฉพาะเจาะจงกว่าไว้ก่อนเสมอ

function fizzBuzz(n) {
  const result = [];
  for (let i = 1; i <= n; i++) {
    if (i % 15 === 0) result.push("FizzBuzz");
    else if (i % 3 === 0) result.push("Fizz");
    else if (i % 5 === 0) result.push("Buzz");
    else result.push(String(i));
  }
  return result;
}

function secondLargest(nums) {
  let first = -Infinity, second = -Infinity;
  for (const n of nums) {
    if (n > first) { second = first; first = n; }
    else if (n > second && n < first) second = n;
  }
  return second === -Infinity ? null : second;
}

console.log(secondLargest([4, 4]));         // null
console.log(secondLargest([-1, -5]));       // -5

Sort ขนาดเล็ก: ต้องเขียนเองได้และบอกขั้นตอนได้

งานจริงเรียกใช้ sort() ที่มีให้ก็พอ (ตั้งแต่ ES2019 รับประกันว่า stable คือค่าที่เท่ากันคงลำดับเดิม) แต่ข้อสอบยังชอบให้เขียนหรือไล่ขั้นตอนของ bubble sort, selection sort, insertion sort เพราะดูได้ว่าเข้าใจ loop ซ้อนกันและการสลับค่าหรือไม่ ทั้งสามแบบมีเวลาทำงานแบบ worst case เป็น O(n²) เหมาะกับข้อมูลหลักสิบถึงหลักร้อยเท่านั้น

function bubbleSort(input) {
  const a = [...input];                       // ไม่แก้ array ของผู้เรียก
  for (let pass = 0; pass < a.length - 1; pass++) {
    let swapped = false;
    for (let i = 0; i < a.length - 1 - pass; i++) {
      if (a[i] > a[i + 1]) {
        [a[i], a[i + 1]] = [a[i + 1], a[i]];
        swapped = true;
      }
    }
    if (!swapped) break;                      // รอบนี้ไม่มีการสลับ แปลว่าเรียงเสร็จแล้ว
  }
  return a;
}

function insertionSort(input) {
  const a = [...input];
  for (let i = 1; i < a.length; i++) {
    const current = a[i];
    let j = i - 1;
    while (j >= 0 && a[j] > current) { a[j + 1] = a[j]; j--; }
    a[j + 1] = current;
  }
  return a;
}

console.log(bubbleSort([5, 1, 4, 2, 8]));     // [ 1, 2, 4, 5, 8 ]
console.log(insertionSort([5, 1, 4, 2, 8]));  // [ 1, 2, 4, 5, 8 ]
console.log(bubbleSort([]), bubbleSort([7])); // [] [ 7 ]

ลองไล่ bubble sort กับ [5, 1, 4, 2, 8] รอบแรกค่า 5 ถูกดันไปจนสุด กลายเป็น [1, 4, 2, 5, 8] รอบสองค่า 4 ถูกดันไปอยู่หลัง 2 ได้ [1, 2, 4, 5, 8] รอบสามไม่มีการสลับเลย จึงหยุด เห็นได้ว่าหลังจบแต่ละรอบ ค่าที่ใหญ่ที่สุดของส่วนที่เหลือจะไปอยู่ท้ายสุด นี่คือสิ่งที่ถูกต้องเสมอทุกรอบ ซึ่งเรียกว่า loop invariant และเป็นเครื่องมือที่ดีในการพิสูจน์ว่า loop ถูกต้อง

ค้นหา: เมื่อไรควรใช้ binary search

ถ้าข้อมูลไม่เรียงต้องใช้ linear search คือไล่ดูทีละตัว ใช้ includes, indexOf หรือ find ได้เลย ถ้าข้อมูลเรียงอยู่แล้ว binary search ตัดช่วงที่ค้นหาลงครึ่งหนึ่งทุกรอบ ทำให้ใช้เวลา O(log n) แต่ต้องเขียนให้ถูกเรื่องขอบเขต low, high ซึ่งเป็นจุดที่ off-by-one เกิดบ่อยที่สุด

function binarySearch(sorted, target) {
  let low = 0, high = sorted.length - 1;
  while (low <= high) {
    const mid = low + Math.floor((high - low) / 2);
    if (sorted[mid] === target) return mid;
    if (sorted[mid] < target) low = mid + 1;
    else high = mid - 1;
  }
  return -1;
}
const data = [2, 5, 8, 12, 16, 23, 38];
console.log(binarySearch(data, 23), binarySearch(data, 2), binarySearch(data, 7), binarySearch([], 1));
// 5 0 -1 -1

สูตร low + (high - low) / 2 ให้ผลเท่ากับ (low + high) / 2 ใน JavaScript แต่เป็นนิสัยที่ดีเพราะในภาษาที่ใช้ integer ขนาดคงที่อย่าง Java หรือ C++ การบวก low + high ตรง ๆ อาจล้นได้

ข้อผิดพลาดที่พบบ่อย

คนมักลืมใส่ comparator ให้ sort() เมื่อเรียงตัวเลข ซึ่งเรียงออกมาเป็นแบบ string ดังที่เห็นในหัวข้อก่อน อีกอย่างคือแก้ array ระหว่างที่วนอ่านมันอยู่ เช่น เรียก splice ใน forEach ทำให้ index เลื่อนและข้ามสมาชิกบางตัวไป ถ้าต้องลบบางตัวให้ใช้ filter สร้าง array ใหม่ หรือวนจากท้ายมาหน้า ส่วนเรื่อง for...in กับ array และ switch ที่ไม่มี break ก็ยังเป็นกับดักที่ข้อสอบชอบหยิบมาใช้

Trade-offs ของวิธีเขียน

การใช้ map/filter/reduce ต่อกันอ่านง่ายเมื่อแต่ละขั้นสั้น แต่ทุกขั้นสร้าง array ใหม่หนึ่งชุด ถ้าข้อมูลมีไม่กี่ร้อยตัวผลต่างแทบไม่มีความหมาย ให้เลือกแบบที่อ่านง่ายกว่า ส่วน reduce ที่ยัดหลายเรื่องไว้ในฟังก์ชันเดียวมักอ่านยากกว่า for...of ธรรมดา ถ้าเกิดปัญหาด้านความเร็วจริง ค่อยวัดแล้วแก้เฉพาะจุดที่ช้า ซึ่งจะอธิบายในหัวข้อ Performance

สรุปสั้น ๆเงื่อนไขที่เฉพาะเจาะจงกว่าต้องมาก่อน จำกัดขอบเขต loop ให้ชัดว่ารวมหรือไม่รวมปลายทาง ใส่ comparator ทุกครั้งที่ sort ตัวเลข sort ขนาดเล็กแบบ O(n²) ต้องไล่ขั้นตอนได้ binary search ใช้ได้เมื่อข้อมูลเรียงแล้วเท่านั้น และต้องคิดกรณีขอบเสมอ เช่น array ว่าง ค่าซ้ำ และค่าติดลบ

Strategy: เปลี่ยนสาย if/else เป็นตารางของพฤติกรรม

กลับมาที่โค้ดส่วนลด แทนที่จะเช็กชนิดแล้วแตกกิ่ง ให้เก็บแต่ละวิธีคิดเป็นฟังก์ชันหนึ่งตัวในตาราง การเพิ่มส่วนลดใหม่คือเพิ่มหนึ่งบรรทัด ไม่ต้องแตะโค้ดที่ทำงานอยู่แล้ว ข้อควรระวังเฉพาะ JavaScript คือถ้าชื่อชนิดมาจากข้อมูลภายนอก การอ่าน strategies[type] ตรง ๆ อาจได้ property ที่ object ทุกตัวมีติดมา อย่าง constructor ดังนั้นให้เช็กด้วย Object.hasOwn (ES2022) หรือใช้ Map

const discountStrategies = {
  none: () => 0,
  member: (subtotal) => subtotal * 0.05,
  coupon: (subtotal, { amount }) => Math.min(amount, subtotal),
};

function priceAfterDiscount(subtotal, type, options = {}) {
  if (!Object.hasOwn(discountStrategies, type)) {
    throw new Error(`unknown discount type: ${type}`);
  }
  return subtotal - discountStrategies[type](subtotal, options);
}

console.log(priceAfterDiscount(1000, "none"));                      // 1000
console.log(priceAfterDiscount(1000, "member"));                    // 950
console.log(priceAfterDiscount(1000, "coupon", { amount: 1200 }));  // 0
try { priceAfterDiscount(1000, "constructor"); } catch (e) { console.log(e.message); }
// unknown discount type: constructor

Factory กับ Dependency Injection: ให้คนอื่นส่งของที่ต้องใช้เข้ามา

ถ้า OrderService สร้างตัวส่งอีเมลและอ่านเวลาปัจจุบันเองข้างใน การเขียน test ที่ไม่ส่งอีเมลจริงและได้เวลาที่แน่นอนทำได้ยากมาก วิธีแก้คือให้ผู้เรียกส่ง dependency เข้ามา ซึ่งเรียกว่า dependency injection ส่วน factory คือฟังก์ชันที่ประกอบของพวกนี้ให้ในที่เดียว เพื่อให้จุดอื่นของโปรแกรมไม่ต้องรู้ว่าใช้ class ไหนอยู่

class OrderService {

    this.notifier = notifier;
    this.clock = clock;

  place(order) {
    this.notifier.send(`order ${order.id} at ${this.clock().toISOString()}`);
  }
}

// ใน test: ใช้ของปลอมที่ควบคุมได้ ไม่ส่งอีเมลจริง และเวลาคงที่

Observer: คนทำงานไม่ต้องรู้ว่าใครสนใจ

เมื่อออเดอร์ถูกจ่ายเงิน ต้องส่งอีเมล ตัดสต็อก และบันทึกสถิติ ถ้าฟังก์ชันจ่ายเงินเรียกทั้งสามอย่างเอง มันจะผูกกับทุกเรื่องนี้ไปตลอด Observer ให้ฟังก์ชันจ่ายเงินประกาศแค่ว่า “paid แล้ว” ใครสนใจก็มาลงทะเบียนรับเอง Node.js มี EventEmitter ในตัว ตัวอย่างด้านล่างเขียนแบบย่อเพื่อดูหลักการ สังเกตว่า on คืนฟังก์ชันยกเลิกกลับมา เพราะ listener ที่ลงทะเบียนแล้วไม่เคยถอนคือสาเหตุยอดนิยมของ memory leak

function createEmitter() {
  const handlers = new Map();
  return {
    on(event, handler) {
      if (!handlers.has(event)) handlers.set(event, new Set());
      handlers.get(event).add(handler);
      return () => handlers.get(event).delete(handler);   // คืนตัวยกเลิกการติดตาม
    },
    emit(event, payload) {
      for (const handler of handlers.get(event) ?? []) handler(payload);
    },
  };
}

const orderEvents = createEmitter();
const stopEmail = orderEvents.on("paid", (o) => console.log("email:", o.id));
orderEvents.on("paid", (o) => console.log("stock:", o.id));

orderEvents.emit("paid", { id: 1 });   // email: 1  /  stock: 1
stopEmail();
orderEvents.emit("paid", { id: 2 });   // stock: 2

Decorator: เพิ่มความสามารถโดยห่อฟังก์ชันเดิม

เมื่อต้องเพิ่ม retry, logging หรือ cache ให้ฟังก์ชันที่มีอยู่ การแก้ในตัวฟังก์ชันทำให้ปนเรื่องกัน ทางเลือกคือสร้างฟังก์ชันที่รับฟังก์ชันเดิมแล้วคืนฟังก์ชันใหม่ที่ทำงานเหมือนกันแต่มีส่วนเพิ่ม คำว่า decorator ในที่นี้หมายถึงการห่อแบบนี้ ไม่ใช่ syntax @decorator ซึ่งการรองรับขึ้นกับ runtime และ TypeScript ที่ใช้ และมีข้อควรระวังตอนใช้งานจริงว่า retry เหมาะกับ operation ที่เรียกซ้ำได้โดยไม่ทำให้เกิดผลซ้ำเท่านั้น เช่น การอ่านข้อมูล ไม่ใช่การตัดเงิน

function withRetry(fn, { retries = 2 } = {}) {
  return async (...args) => {
    let lastError;
    for (let attempt = 0; attempt <= retries; attempt++) {
      try { return await fn(...args); }
      catch (err) { lastError = err; }
    }
    throw lastError;
  };
}

let calls = 0;
const flakyRead = async () => {
  calls++;
  if (calls < 3) throw new Error("temporary");
  return "ok";
};

withRetry(flakyRead)().then((result) => console.log(result, calls));   // ok 3

Composition ก่อน Inheritance

Inheritance ผูกลูกเข้ากับพ่อแม่แน่นกว่าที่เห็น เมื่อมีลำดับ Employee → Manager → Director แล้วอยู่ ๆ มีข้อกำหนดว่าผู้รับเหมาก็อนุมัติค่าใช้จ่ายได้ แต่ไม่ใช่พนักงาน ลำดับชั้นเดิมก็ไม่รองรับ Composition ประกอบความสามารถเล็ก ๆ เข้าด้วยกันตามที่ต้องการ แต่ละชิ้นทดสอบแยกได้และสลับได้

const canApprove = (limit) => ({ approve: (amount) => amount <= limit });
const canSubmit = () => ({ submit: (expense) => ({ ...expense, status: "submitted" }) });

const createManager = (name) => ({ name, ...canSubmit(), ...canApprove(50000) });
const createContractor = (name) => ({ name, ...canApprove(10000) });

แล้ว “high-performance” เกี่ยวอย่างไรกับ pattern

Pattern ส่วนใหญ่เพิ่มชั้นของการเรียกฟังก์ชัน ซึ่งโดยทั่วไปมีต้นทุนน้อยมากเมื่อเทียบกับงานที่ช้าจริงอย่าง network, database หรือการอ่านไฟล์ ดังนั้นอย่าตัดการแยกส่วนทิ้งเพราะกลัวช้าโดยไม่ได้วัด แต่ก็มีจุดที่ควรระวัง เช่น ในลูปที่ทำงานหลายล้านรอบ การสร้าง closure หรือ object ใหม่ทุกรอบสร้างภาระให้ garbage collector และ listener จำนวนมากที่ไม่เคยถูกถอนทำให้หน่วยความจำโตขึ้นเรื่อย ๆ ส่วน pattern ที่ออกแบบมาเพื่อความเร็วโดยตรงคือ memoization (จำผลลัพธ์ที่เคยคำนวณ) และ caching ซึ่งจะลงรายละเอียดในหัวข้อ Performance

ข้อผิดพลาดที่เจอบ่อย

ข้อแรกคือใส่ pattern ก่อนมีปัญหา มีเงื่อนไขแค่สองกิ่งแล้วสร้างระบบ strategy พร้อม factory เพิ่มแต่ความซับซ้อน หลักที่ทีมจำนวนมากใช้คือรอจนเห็นรูปแบบซ้ำสามครั้งก่อนค่อยแยก ข้อที่สองคือ Singleton ที่กลายเป็นตัวแปร global ซ่อนอยู่ ใน Node.js โมดูลถูก cache อยู่แล้ว ทำให้ export const db = ... ได้ผลคล้าย singleton โดยไม่ต้องเขียนอะไรพิเศษ แต่ทำให้ test ยากเพราะสถานะค้างข้ามเทสต์ ข้อที่สามคือสร้าง class ชื่อ Manager, Helper, Utils ที่รวมทุกเรื่องไว้ด้วยกัน และข้อสุดท้ายคือ inheritance ที่ลึกหลายชั้นจนไม่มีใครรู้ว่าเมธอดนี้มาจากไหน

Refactor คือการเปลี่ยนโครงสร้างภายในของโค้ดให้อ่านง่ายและแก้ง่ายขึ้น โดยพฤติกรรมที่ผู้ใช้หรือโค้ดอื่นมองเห็นต้องเหมือนเดิมทุกประการ ถ้าพฤติกรรมเปลี่ยน มันไม่ใช่ refactor แล้ว ไม่ว่าจะเปลี่ยนไปในทางที่ดีกว่าแค่ไหน เพราะฉะนั้นการแก้ bug, การเพิ่มฟีเจอร์ และการ refactor ควรอยู่คนละ commit เพื่อให้ตรวจย้อนหลังได้ง่าย และถ้ามีอะไรผิดก็ย้อนกลับเฉพาะส่วนได้

ขั้นแรก: ให้มี test ครอบก่อนแตะโค้ด

โค้ดเก่าที่ไม่มี test ต้องเริ่มด้วยการเขียน characterization test คือ test ที่จดว่าโค้ดตอนนี้ทำอะไร ไม่ใช่ว่าควรทำอะไร เอา input หลายแบบรวมถึงแบบแปลก ๆ ไปรัน แล้วบันทึกผลที่ออกมาเป็นค่าที่คาดหวัง ถ้าผลที่ได้ดูผิดก็จดไว้ก่อนว่าผิด แล้วไปแก้ในรอบหลัง จากนั้นค่อย refactor ทีละก้าวเล็ก ๆ และรัน test ทุกก้าว

สัญญาณว่าโค้ดควรถูก refactor

สัญญาณที่พบบ่อยคือ ฟังก์ชันยาวที่มี comment คั่นเป็นช่วง ๆ โค้ดที่คัดลอกไปวางซ้ำหลายที่ ตัวเลขดิบที่ไม่มีชื่อ เงื่อนไขซ้อนลึก ฟังก์ชันที่รับพารามิเตอร์ยาวเกินห้าหกตัว class ที่รู้ไปทุกเรื่อง และการแก้เรื่องเดียวที่ต้องไล่แก้สิบไฟล์ (shotgun surgery) เทคนิคที่ใช้แก้ก็มีชื่อเรียกชัดเจน ได้แก่ Extract Function แยกท่อนโค้ดออกเป็นฟังก์ชัน, Rename ตั้งชื่อใหม่ให้ตรง, Guard Clause ออกก่อนเมื่อเงื่อนไขไม่ผ่าน, Replace Conditional with Lookup แทนสาย if ด้วยตาราง, Introduce Parameter Object รวมพารามิเตอร์เป็น object และลบ dead code ที่ไม่มีใครเรียกแล้ว

ตัวอย่างทำจริง: ฟังก์ชัน legacy หนึ่งตัว

ฟังก์ชันด้านล่างคำนวณยอดออเดอร์ มีตัวแปรชื่อสั้น ใช้ == มีสายเงื่อนไขซ้ำ ๆ และตัวเลขที่ไม่มีที่มา ผมเขียนเวอร์ชันแรกของการ refactor ให้อ่านง่ายขึ้น แล้วเอาไปเทียบกับของเดิมด้วยชุด input ที่ตั้งใจใส่ตัวแปลก ๆ ลงไปด้วย ผลที่ได้คือความต่างที่มองไม่เห็นด้วยตา

ข้อผิดพลาดที่เจอบ่อย

Refactor ทั้งที่ไม่มี test เป็นอันดับหนึ่ง รองลงมาคือรวมการ refactor กับการเพิ่มฟีเจอร์ไว้ใน PR เดียว ทำให้ reviewer แยกไม่ออกว่าบรรทัดไหนเปลี่ยนพฤติกรรม ตามมาด้วยการเปลี่ยนชื่อ function ที่โมดูลอื่นหรือทีมอื่นเรียกอยู่โดยไม่ประกาศล่วงหน้า และการ “ถือโอกาส” ปรับพฤติกรรมเล็ก ๆ ระหว่างทาง ซึ่งเป็นสิ่งที่ตัวอย่างข้างบนเตือนไว้แล้ว

    ...e,

  }));
}

let steps = 0;
function attachNamesFast() {
  const nameById = new Map();
  for (const d of departments) { steps++; nameById.set(d.id, d.name); }
  return employees.map((e) => { steps++; return { ...e, deptName: nameById.get(e.deptId) }; });
}

const slow = attachNamesSlow();
const fast = attachNamesFast();
console.log(JSON.stringify(slow) === JSON.stringify(fast));   // true  (ผลเหมือนกัน)
console.log(comparisons, steps);                              // 51000 2050

ผลลัพธ์เหมือนกันทุกประการ แต่งานที่ทำลดจาก 51,000 ครั้งเหลือ 2,050 ครั้ง และช่องว่างจะกว้างขึ้นเรื่อย ๆ เมื่อข้อมูลโต เพราะแบบแรกโตเป็นผลคูณของสองขนาด แบบหลังโตเป็นผลบวก (ในภาษาโปรแกรมสากลเขียนว่า O(n·m) เทียบกับ O(n + m)) ตอนใช้ Map กับ key ที่มาจากภายนอก Map ยังปลอดภัยกว่า object ธรรมดาเรื่อง key ชน constructor ที่เจอในหัวข้อก่อนด้วย

ตัวอย่าง 2: Memoization จำผลที่เคยคิดแล้ว

ฟังก์ชันที่ให้ผลเหมือนเดิมเสมอเมื่อ input เหมือนเดิม (pure function) จำผลลัพธ์ไว้ได้ Fibonacci แบบเรียกตัวเองซ้ำเป็นตัวอย่างคลาสสิก เพราะคำนวณค่าเดิมซ้ำมหาศาล

let naiveCalls = 0;
function fibNaive(n) {
  naiveCalls++;
  return n < 2 ? n : fibNaive(n - 1) + fibNaive(n - 2);
}

function memoize(fn) {
  const cache = new Map();
  return (n) => {
    if (!cache.has(n)) cache.set(n, fn(n));
    return cache.get(n);
  };
}

let memoCalls = 0;
const fibMemo = memoize((n) => {
  memoCalls++;
  return n < 2 ? n : fibMemo(n - 1) + fibMemo(n - 2);
});

console.log(fibNaive(30), naiveCalls);   // 832040 2692537
console.log(fibMemo(30), memoCalls);     // 832040 31

ในงานจริง cache ต้องมีเพดานและวิธีหมดอายุ ถ้าไม่มี cache ที่เก็บผลทุก input ที่เคยเห็นจะโตไม่หยุดและกลายเป็น memory leak ต้องกำหนดขนาดสูงสุดหรืออายุ (เช่น LRU หรือ TTL) และถ้าข้อมูลต้นทางเปลี่ยนต้องมีวิธีทำให้ cache เก่าใช้ไม่ได้ (invalidate) ซึ่งเป็นเรื่องที่ยากกว่าตัวการเก็บเสียอีก และเก็บได้เฉพาะผลของฟังก์ชันที่ไม่ขึ้นกับสถานะภายนอก

ตัวอย่าง 3: ทำงานที่ไม่พึ่งกันให้พร้อมกัน และลดความถี่ของงานที่ถี่เกิน

ถ้าต้องดึงข้อมูลสามอย่างที่ไม่เกี่ยวกัน การ await ทีละตัวทำให้เวลารวมเป็นผลบวกของทั้งสาม ส่วน Promise.all ให้ทั้งสามเริ่มพร้อมกัน เวลารวมเท่ากับตัวที่ช้าที่สุด ถ้าผลของตัวหนึ่งเป็น input ของอีกตัวก็ต้องรอเรียงกันตามเดิม ส่วน debounce ใช้กับ event ที่ยิงถี่ เช่น พิมพ์ในช่องค้นหา รอให้หยุดพิมพ์ครู่หนึ่งแล้วค่อยทำงานครั้งเดียว

const wait = (ms) => new Promise((resolve) => setTimeout(resolve, ms));
async function fetchOne(id) { await wait(100); return id; }

async function sequential(ids) {
  const out = [];
  for (const id of ids) out.push(await fetchOne(id));
  return out;
}
const parallel = (ids) => Promise.all(ids.map(fetchOne));

async function timeIt(label, fn) {
  const start = performance.now();
  await fn([1, 2, 3]);
  console.log(label, "≈", Math.round((performance.now() - start) / 100), "หน่วย");
}

function debounce(fn, waitMs) {
  let timer;
  return (...args) => {
    clearTimeout(timer);
    timer = setTimeout(() => fn(...args), waitMs);
  };
}

(async () => {
  await timeIt("ทีละตัว ", sequential);   // ทีละตัว  ≈ 3 หน่วย
  await timeIt("พร้อมกัน", parallel);     // พร้อมกัน ≈ 1 หน่วย

  const search = debounce((q) => console.log("search:", q), 50);
  search("a"); search("ab"); search("abc");   // ผ่านไป 50 ms จะพิมพ์ครั้งเดียว: search: abc
})();

ข้อผิดพลาดที่เจอบ่อย

ข้อแรกคือแก้โดยไม่ได้วัด และไม่ได้วัดซ้ำหลังแก้ ทำให้ไม่รู้ว่าแก้แล้วดีขึ้นจริงหรือเปล่า ข้อที่สองคือเชื่อผล micro-benchmark ที่ไม่สะท้อนงานจริง ข้อที่สามคือเพิ่ม cache โดยไม่คิดเรื่องความถูกต้องของข้อมูล ผู้ใช้เห็นข้อมูลเก่าและทีมใช้เวลาไล่บั๊กนานกว่าเวลาที่ cache ช่วยประหยัด ข้อที่สี่คือทำให้ทุกอย่างพร้อมกันโดยไม่จำกัดจำนวน จน server ปลายทางหรือฐานข้อมูลรับไม่ไหว ข้อสุดท้ายคือแลกความอ่านง่ายกับการเร็วขึ้นที่ไม่มีใครสังเกตเห็นได้

Trade-offs

เกือบทุกการ optimize แลกกับอะไรสักอย่าง cache แลกความเร็วกับความสดของข้อมูลและหน่วยความจำ การทำงานพร้อมกันแลกเวลากับภาระที่ปลายทาง การ precompute แลกเวลาตอนใช้งานกับพื้นที่เก็บและเวลาตอนเตรียม จึงควรตั้งเป้าก่อนว่าต้องเร็วแค่ไหนถึงพอ (performance budget) เช่น หน้านี้ต้องตอบภายในกี่วินาทีที่ผู้ใช้เครื่องทั่วไป พอถึงเป้าแล้วก็หยุด ไม่ต้องไล่ลดเพิ่มทุกมิลลิวินาที

สรุปสั้น ๆวัดด้วย profiler ก่อน แก้ที่คอขวดจริง เริ่มจากอัลกอริทึมและ I/O ไม่ใช่ระดับบรรทัด loop ซ้อนที่ค้นหาซ้ำเปลี่ยนเป็น Map ได้ memoize เฉพาะ pure function และต้องมีเพดานกับ invalidation งานที่ไม่พึ่งกันใช้ Promise.all แต่ต้องจำกัดจำนวน และวัดซ้ำหลังแก้เพื่อพิสูจน์ว่าดีขึ้นจริง

let balance = 100;
const wait = (ms) => new Promise((resolve) => setTimeout(resolve, ms));

  await wait(10);                   // รอ (เช่น เรียก API ตรวจสิทธิ์)
  balance = current - amount;       // เขียนทับด้วยค่าที่อ่านไว้ก่อนหน้า
}

// ตัวช่วย: ทำงานทีละตัวตามลำดับคิว
function createMutex() {
  let tail = Promise.resolve();
  return (task) => {
    const result = tail.then(task);
    tail = result.catch(() => {});  // งานที่พังไม่ทำให้คิวค้าง
    return result;
  };
}
const lock = createMutex();
async function withdrawSafe(amount) {
  return lock(async () => {
    const current = balance;
    await wait(10);
    balance = current - amount;
  });
}

(async () => {
  await Promise.all([withdrawBuggy(30), withdrawBuggy(30)]);
  console.log("buggy:", balance);          // buggy: 70   (ควรเป็น 40)

  balance = 100;
  await Promise.all([withdrawSafe(30), withdrawSafe(30)]);

จากโค้ดช้าไปสู่โค้ดเร็วอย่างเป็นขั้นตอน

เทคนิค algorithm ที่ใช้ optimize ไม่ได้ต้องจำเป็นร้อยแบบ วิธีคิดที่ใช้ซ้ำคือเริ่มจากวิธีตรงไปตรงมาที่ถูกต้องก่อน แล้วถามว่าตรงไหนที่ทำงานซ้ำ แล้วหาโครงสร้างข้อมูลที่ตัดงานซ้ำนั้น ลองดูโจทย์ Two Sum ที่เป็นจุดเริ่มของหลายคน ให้หา index ของสองตัวใน array ที่บวกกันได้ค่าเป้าหมาย

// รอบ 1: ถูกต้อง แต่ทดลองทุกคู่ — ใช้เวลาเป็นกำลังสองของขนาด array
function twoSumBrute(nums, target) {
  for (let i = 0; i < nums.length; i++) {
    for (let j = i + 1; j < nums.length; j++) {
      if (nums[i] + nums[j] === target) return [i, j];
    }
  }
  return null;
}

// รอบ 2: ตัวที่ 2 ที่ต้องการคือ target - ตัวแรก ถามว่า "เคยเห็นค่านี้แล้วหรือยัง" ด้วย Map
function twoSum(nums, target) {
  const seen = new Map();                       // ค่า -> index
  for (let i = 0; i < nums.length; i++) {
    const need = target - nums[i];
    if (seen.has(need)) return [seen.get(need), i];
    seen.set(nums[i], i);
  }
  return null;
}

console.log(twoSumBrute([2, 7, 11, 15], 9), twoSum([2, 7, 11, 15], 9));   // [ 0, 1 ] [ 0, 1 ]
console.log(twoSum([3, 3], 6), twoSum([1, 2], 10));                        // [ 0, 1 ] null

จุดที่เปลี่ยนคือ ข้อมูลที่ต้องการหาถูกแปลงเป็นคำถามที่ตอบได้ในครั้งเดียว ด้วยการจ่ายค่าหน่วยความจำเพิ่มเพื่อแลกเวลา ซึ่งเป็นการแลกเปลี่ยนที่พบในแทบทุกการ optimize เทคนิคที่เจอบ่อยอีกสองแบบคือ prefix sum กับ sliding window ใช้เมื่อโจทย์ถามเรื่องช่วงต่อเนื่องของ array

// Prefix sum: เตรียมผลรวมสะสมครั้งเดียว แล้วถามผลรวมช่วงใดก็ได้ในเวลาคงที่
function buildPrefix(nums) {
  const prefix = [0];
  for (const n of nums) prefix.push(prefix[prefix.length - 1] + n);
  return prefix;
}
const rangeSum = (prefix, left, right) => prefix[right + 1] - prefix[left];   // รวม left..right

const prefix = buildPrefix([4, 1, 7, 3, 9]);
console.log(rangeSum(prefix, 1, 3));   // 11  (1 + 7 + 3)

// Sliding window: ผลรวมสูงสุดของ k ตัวติดกัน ขยับหน้าต่างทีละตำแหน่งแทนการรวมใหม่
function maxSumOfK(nums, k) {
  if (k <= 0 || k > nums.length) return null;
  let windowSum = 0;
  for (let i = 0; i < k; i++) windowSum += nums[i];
  let best = windowSum;
  for (let i = k; i < nums.length; i++) {
    windowSum += nums[i] - nums[i - k];   // รับตัวใหม่เข้า ปล่อยตัวเก่าออก
    best = Math.max(best, windowSum);
  }
  return best;
}
console.log(maxSumOfK([2, 1, 5, 1, 3, 2], 3), maxSumOfK([1], 2));   // 9 null

เมื่อโจทย์ถามว่า “ค่าน้อยที่สุดที่ทำให้เงื่อนไขเป็นจริง” และเงื่อนไขนั้นเป็นแบบที่ถ้าจริงที่ค่าหนึ่ง ก็จริงที่ค่าที่มากกว่าด้วย สามารถใช้ binary search บนคำตอบได้เลย ไม่ใช่แค่บน array เช่น หา “ความเร็วต่ำสุดที่ทำงานเสร็จภายในเวลาที่กำหนด” โดยลองความเร็วตรงกลางแล้วตัดครึ่งช่วงที่เหลือ ส่วนเรื่อง greedy กับ dynamic programming จะลงรายละเอียดในหัวข้อ DSA

ข้อแรกคือแก้ race condition ด้วยการเพิ่ม setTimeout ให้รอ ซึ่งอาจผ่านบนเครื่องเราแต่พังเมื่อเครื่องช้าลง การรอแบบเดาเวลาไม่ใช่การแก้ ข้อที่สองคือกลืน error เงียบ ๆ ใน catch ทำให้ log ไม่เหลือร่องรอย ข้อที่สามคือ optimize algorithm โดยไม่เช็กว่าเงื่อนไขของวิธีนั้นเป็นจริงกับข้อมูล เช่น ใช้ binary search กับข้อมูลที่ยังไม่เรียง หรือ sliding window กับ array ที่มีค่าติดลบในโจทย์ที่ต้องการผลรวมไม่น้อยกว่าค่าหนึ่ง ซึ่งเทคนิคนี้ใช้ไม่ได้

Trade-offs

วิธีที่ใช้หน่วยความจำเพิ่มเพื่อให้เร็วขึ้นอย่าง Map หรือ prefix array ไม่เหมาะเมื่อข้อมูลมหาศาลจนไม่พอใส่หน่วยความจำ และวิธีที่ซับซ้อนกว่าไม่คุ้มเมื่อข้อมูลมีแค่หลักสิบ ที่ที่อ่านง่ายกว่าก็ควรเลือก ส่วนการแก้ความถูกต้องของข้อมูลที่ใช้ร่วมกัน ให้เลือกตามบริบทว่าต้องการความแม่นยำเต็มที่ (ล็อกหรือ transaction) หรือยอมรับผลลัพธ์ที่สอดคล้องกันภายหลัง (eventual consistency)

สรุปสั้น ๆใช้การแบ่งครึ่งหาจุดเริ่มของบั๊ก (git bisect) ทำให้บั๊กที่เกิดบางครั้งเกิดซ้ำได้ก่อนแก้ ปัญหา read-await-write แก้ด้วยคิวหรือคำสั่ง atomic ของฐานข้อมูล ไม่ใช่การรอเดาเวลา และการ optimize เริ่มจากวิธีที่ถูกต้องแล้วหาว่าส่วนไหนทำซ้ำ ก่อนเลือก Map, prefix sum, sliding window หรือ binary search ให้ตรงกับเงื่อนไขของข้อมูล

ระดับ Advance

ตัดสินใจเรื่องสถาปัตยกรรมที่ใช้หลายภาษา เลือก paradigm ให้เหมาะกับปัญหาแต่ละส่วน กำหนดมาตรฐานการเขียนโค้ดที่ทั้งองค์กรใช้ และนำการตัดสินใจเรื่อง optimization กับกลยุทธ์ด้านภาษาโดยมีเหตุผลรองรับ

ข้อสอบระดับนี้มักไม่มีคำตอบที่ถูกเพียงข้อเดียวในเชิงเทคโนโลยี แต่ดูว่าให้เหตุผลและยอมรับ trade-off ได้ครบหรือไม่

ระบบประมวลผลออเดอร์ชิ้นเดียวมีงานที่ธรรมชาติต่างกันอยู่ข้างใน ตัวบัญชีและกฎธุรกิจที่ต้องรักษาความถูกต้องของข้อมูลเหมาะกับการห่อไว้เป็น object งานแปลงข้อมูลจากรายการออเดอร์เป็นรายงานเหมาะกับฟังก์ชันที่ต่อกันเป็นท่อ ส่วนการดึงข้อมูลจากหลายแหล่งพร้อมกันต้องคิดเรื่องลำดับเวลา การเชี่ยวชาญหลาย paradigm ไม่ได้แปลว่าต้องผสมให้ครบทุกแบบ แต่คือเลือกให้เหมาะกับปัญหาแต่ละส่วนได้ และบอกเหตุผลได้

OOP: เมื่อมีสถานะที่ต้องรักษากฎ

หัวใจของ OOP ไม่ใช่การมี class แต่คือ encapsulation ซ่อนสถานะภายใน และเปิดให้เปลี่ยนผ่านเมธอดที่ตรวจกฎก่อนเสมอ เพื่อให้ object ไม่มีวันอยู่ในสถานะที่ผิด ตัวอย่างบัญชีเงินฝากข้างล่างใช้ private field (#, ES2022) ทำให้โค้ดข้างนอกแก้ยอดตรง ๆ ไม่ได้ ส่วน polymorphism ทำให้ผู้เรียก monthlyClose() ไม่ต้องรู้ว่ากำลังคุยกับบัญชีชนิดไหน

class Account {
  #balance;
  constructor(opening = 0) {
    if (opening < 0) throw new RangeError("opening balance must be >= 0");
    this.#balance = opening;
  }
  get balance() { return this.#balance; }
  deposit(amount) {
    if (!(amount > 0)) throw new RangeError("amount must be > 0");
    this.#balance += amount;
  }
  withdraw(amount) {
    if (!(amount > 0)) throw new RangeError("amount must be > 0");
    if (amount > this.#balance) throw new Error("insufficient funds");
    this.#balance -= amount;
  }
  monthlyClose() {}                                  // ตัวแม่ไม่ทำอะไร
}

class SavingsAccount extends Account {
  constructor(opening, rate) { super(opening); this.rate = rate; }
  monthlyClose() { this.deposit(this.balance * this.rate); }
}

class CheckingAccount extends Account {
  monthlyClose() { if (this.balance >= 20) this.withdraw(20); }   // ค่าธรรมเนียม
}

const accounts = [new SavingsAccount(1000, 0.01), new CheckingAccount(100)];
accounts.forEach((a) => a.monthlyClose());
console.log(accounts.map((a) => a.balance));          // [ 1010, 80 ]
try { accounts[1].withdraw(500); } catch (e) { console.log(e.message); }   // insufficient funds

OOP เสียเปรียบเมื่อใช้เป็นแค่ถุงเก็บข้อมูลที่มีเมธอดจำนวนมากไม่ผูกกับสถานะ (บางทีเรียกว่า anemic หรือ god object) และเมื่อสร้างลำดับชั้น inheritance ลึกเกินไปจนเปลี่ยนตัวแม่แล้วลูกพังหลายที่

Functional: เมื่อเป็นงานแปลงข้อมูลและอยากทดสอบง่าย

ฟังก์ชันแบบ pure รับ input แล้วคืน output โดยไม่แก้ของที่อยู่นอกตัวเองและไม่พึ่งสถานะที่ซ่อนอยู่ จึงทดสอบง่ายและเรียกพร้อมกันได้โดยไม่ต้องกลัวชนกัน เมื่อต่อฟังก์ชันเล็ก ๆ เป็นท่อก็อ่านเป็นลำดับขั้นของธุรกิจได้ตรง ๆ อีกเรื่องที่เกี่ยวกันคือ immutability การแก้ข้อมูลให้สร้างสำเนาใหม่แทนการแก้ของเดิม เพราะข้อมูลที่ไม่เปลี่ยนไม่ต้องกลัวว่าส่วนอื่นของโปรแกรมแอบแก้ระหว่างทาง

const pipe = (...fns) => (input) => fns.reduce((value, fn) => fn(value), input);

const orders = [
  { id: 1, status: "paid", total: 500 },
  { id: 2, status: "cancelled", total: 900 },
  { id: 3, status: "paid", total: 1500 },
];

const paidOnly = (list) => list.filter((o) => o.status === "paid");
const totals = (list) => list.map((o) => o.total);
const sum = (nums) => nums.reduce((a, b) => a + b, 0);
const revenue = pipe(paidOnly, totals, sum);

console.log(revenue(orders));                                   // 2000

const updated = orders.map((o) => (o.id === 2 ? { ...o, status: "refunded" } : o));
console.log(orders[1].status, updated[1].status);               // cancelled refunded

// Object.freeze แช่แข็งแค่ชั้นแรก
const config = Object.freeze({ retry: { max: 3 } });
config.retry.max = 99;
console.log(config.retry.max);                                  // 99  (ชั้นในยังแก้ได้)
(function () {
  "use strict";
  try { config.name = "x"; } catch (e) { console.log(e.constructor.name); }   // TypeError
})();

ต้นทุนของ functional คือการสร้างสำเนาใหม่เมื่อข้อมูลใหญ่ ต้องคิดว่าสำเนาแบบตื้น (spread) พอไหม หรือต้องใช้ structuredClone สำหรับสำเนาลึก (Node.js 17 ขึ้นไป) และถ้าห่วงโซ่ reduce หรือการเขียนแบบ point-free ซับซ้อนเกินไป ทีมอ่านไม่ออก ก็ไม่ได้ประโยชน์อะไร

Concurrent: คิดเรื่องลำดับเวลาโดยตรง

ต้องแยกสองคำให้ออกก่อน concurrency คือโครงสร้างที่ให้หลายงานดำเนินไปคาบเกี่ยวกัน ส่วน parallelism คือหลายงานรันในเวลาเดียวกันจริง ๆ บนหลาย core JavaScript ใช้ event loop เธรดเดียว ทำให้ I/O หลายอย่างคาบเกี่ยวกันได้โดยไม่ต้องมี lock แต่งาน CPU หนักที่เขียนแบบ synchronous จะบล็อกทุกอย่าง ถ้าต้องการ parallelism จริงใน Node.js ใช้ worker_threads ที่สื่อสารกันด้วยการส่งข้อความ ฝั่ง browser ใช้ Web Workers

// จำกัดจำนวนงานที่ทำพร้อมกัน แต่ยังเก็บผลตามลำดับเดิมของ input
async function mapLimit(items, limit, worker) {
  const results = new Array(items.length);
  let next = 0, running = 0, peak = 0;
  async function lane() {
    while (next < items.length) {
      const index = next++;
      running++; peak = Math.max(peak, running);
      try { results[index] = await worker(items[index], index); }
      finally { running--; }
    }
  }
  await Promise.all(Array.from({ length: Math.min(limit, items.length) }, lane));
  return { results, peak };
}

const wait = (ms) => new Promise((resolve) => setTimeout(resolve, ms));

(async () => {
  const ids = [1, 2, 3, 4, 5, 6, 7, 8];
  const { results, peak } = await mapLimit(ids, 3, async (id) => { await wait(20); return id * 2; });
  console.log(results.join(","), "peak:", peak);   // 2,4,6,8,10,12,14,16 peak: 3

  // งาน CPU หนัก: ย้ายไปทำใน worker thread
  const { Worker } = require("node:worker_threads");
  const code = `
    const { parentPort, workerData } = require("node:worker_threads");
    let sum = 0;
    for (let i = 0; i < workerData.n; i++) sum += i;
    parentPort.postMessage(sum);
  `;
  const worker = new Worker(code, { eval: true, workerData: { n: 10_000_000 } });
  worker.on("message", (sum) => console.log("sum from worker:", sum));   // 49999995000000
})();

mapLimit ข้างบนเป็นแบบ fail-fast คือถ้า worker ตัวใด throw error Promise.all จะ reject ทันที แต่ lane อื่นยังทำงานต่อไปจนจบ ถ้าต้องการให้ทุกงานจบก่อนแล้วค่อยสรุปว่าตัวไหนพัง ให้เก็บผลแบบ settled (Promise.allSettled) หรือใช้ AbortController เพื่อยกเลิกงานที่เหลือ ในงานจริงหลายทีมเลือกใช้ library ที่ทดสอบมาแล้วแทนเขียนเอง

ภาษาอื่นแก้โจทย์เดียวกันด้วยโมเดลที่ต่างกัน Java ใช้ thread ร่วมกับ lock และ concurrent collection ซึ่งให้ parallelism จริงแต่ต้องระวัง race และ deadlock Go ใช้ goroutine กับ channel ตามแนวคิด CSP ที่สื่อสารกันด้วยการส่งค่า Erlang และ Elixir ใช้ actor ที่แต่ละตัวมีสถานะเป็นของตัวเองและคุยกันด้วยข้อความ ส่วน Python มี asyncio สำหรับงาน I/O และสำหรับงานที่ใช้ CPU ต้องดูว่า interpreter และ version ที่ใช้อนุญาตให้รันเธรดคู่ขนานแค่ไหน ซึ่งเปลี่ยนไปตามรุ่น แนวคิดที่ควรติดตัวคือ “ใครเป็นเจ้าของสถานะ และสถานะนั้นถูกแก้พร้อมกันได้หรือไม่”

เลือก paradigm ให้เหมาะกับปัญหา

ลักษณะปัญหาแนวที่เหมาะเหตุผลข้อควรระวัง
เอนทิตีที่มีสถานะและกฎต้องรักษา เช่น บัญชี ออเดอร์OOP แบบ encapsulationกฎอยู่ที่เดียวกับข้อมูล ไม่มีทางข้ามกฎอย่าสร้าง inheritance ลึก
แปลงหรือสรุปข้อมูลเป็นขั้น ๆFunctional (pure + pipe)ทดสอบง่าย ไม่มีผลข้างเคียงสำเนาข้อมูลใหญ่มีต้นทุน
เรียก I/O หลายตัวที่ไม่พึ่งกันConcurrent แบบ asyncรอพร้อมกัน ลดเวลารวมจำกัดจำนวนงาน และจัดการ error ของแต่ละงาน
คำนวณหนักบน CPUParallel ด้วย worker / processไม่บล็อก event loopต้องส่งข้อมูลข้ามเธรด มีต้นทุนคัดลอก
ระบบมีหลายส่วนที่เปลี่ยนด้วยเหตุผลต่างกันผสม: แกนแบบ functional ห่อด้วยชั้น I/Oส่วนคิดทดสอบง่าย ส่วนต่อโลกภายนอกบางลงต้องมีขอบเขตชัดว่าอะไรอยู่ชั้นไหน

ข้อผิดพลาดที่เจอบ่อย

ใน concurrent ที่เจอบ่อยสุดคือ Promise.all กับรายการยาวหลายพันตัวโดยไม่จำกัด ปลายทางรับไม่ไหวแล้วเกิด error ต่อเนื่อง ตามมาด้วย promise ที่ reject โดยไม่มีใครรับ (unhandled rejection) และการเรียกงาน CPU หนักใน handler ของ server ทำให้ request อื่นค้างทั้งหมด ส่วนฝั่ง OOP กับ functional ก็มักพลาดในรูปแบบตรงข้ามกัน คือฝั่งหนึ่งยัดสถานะที่แก้ได้ทุกที่ไว้ใน object ที่ใครก็เข้าถึงได้ อีกฝั่งสร้างสำเนาทุกขั้นจนช้าโดยไม่จำเป็น

Trade-offs

การเลือก paradigm คือการเลือกว่าจะจ่ายอะไร OOP จ่ายด้วยความผูกพันของลำดับชั้น functional จ่ายด้วยการสร้างข้อมูลใหม่และต้องคุ้นกับวิธีคิด concurrent จ่ายด้วยความซับซ้อนของการคิดเรื่องเวลาและการดีบั๊ก ทีมที่ไม่คุ้นกับ paradigm ไหนมาก่อนจะเสียต้นทุนการเรียนรู้ด้วย การตัดสินใจจึงควรดูทั้งตัวปัญหาและทักษะของทีม ไม่ใช่ดูเฉพาะว่าแบบไหนสง่างามกว่า

สรุปสั้น ๆOOP ดีเมื่อมีสถานะที่ต้องรักษากฎ functional ดีเมื่อเป็นการแปลงข้อมูลและต้องการทดสอบง่าย concurrent ต้องแยก concurrency (คาบเกี่ยว) ออกจาก parallelism (พร้อมกันจริง) ใน JavaScript งาน CPU หนักไปอยู่ใน worker และงานที่ทำพร้อมกันต้องมีเพดาน แล้วคุยกันที่คำถาม “ใครเป็นเจ้าของสถานะ”

เหตุผลที่ฟังขึ้นมีไม่กี่ข้อ ข้อแรกคือ ecosystem ของงานบางประเภทอยู่ในภาษาหนึ่งเป็นหลัก เช่น งาน machine learning ส่วนใหญ่มี library พร้อมใช้ใน Python ข้อสองคือคุณสมบัติของ runtime เช่น ส่วนที่ต้องการความเร็วและการควบคุมหน่วยความจำอาจเหมาะกับภาษาที่คอมไพล์เป็น native ข้อสามคือขอบเขตของทีม ทีมที่รับผิดชอบคนละส่วนเลือกเครื่องมือที่ตัวเองเชี่ยวชาญ ข้อสุดท้ายคือมรดกของระบบเดิมที่ย้ายไม่ได้ในเวลาอันสั้น ส่วนต้นทุนก็ชัดเจนเช่นกัน แต่ละภาษาเพิ่มภาระเรื่องการสรรหาคน เครื่องมือ build การดูแลช่องโหว่ของ dependency การ monitor และความรู้ที่คนในทีมต้องสลับไปมา

ภาษาต่างกันคุยกันได้อย่างไร

มีสี่แนวที่ใช้กันทั่วไป แนวแรกคือเรียกกันผ่านเครือข่ายด้วย HTTP และ JSON โดยมี OpenAPI เป็นสัญญา หรือใช้ gRPC ที่นิยามข้อความด้วย Protocol Buffers แล้วสร้างโค้ดฝั่ง client และ server ให้หลายภาษาจากไฟล์เดียว แนวที่สองคือส่งผ่านคิวหรือ event stream ซึ่งแยกเวลาการทำงานของสองฝั่งออกจากกัน ต้องมี schema ที่ตกลงกันและมักมี schema registry เก็บเวอร์ชัน แนวที่สามคือเรียกกันในโปรเซสเดียวผ่าน native addon, FFI หรือ WebAssembly ได้ความเร็วในการเรียกสูงสุด แต่ถ้าโค้ดฝั่ง native พัง ก็ลากทั้งโปรเซสลงไปด้วย และการ build ซับซ้อนขึ้น แนวสุดท้ายคือรันเป็นโปรเซสแยกแล้วคุยกันผ่าน stdin/stdout หรือ sidecar ซึ่งง่ายแต่เสียเวลาสร้างโปรเซสและจัดการ error ลำบาก

แนวที่ควรหลีกเลี่ยงคือให้หลายระบบต่างภาษาอ่านเขียนตารางเดียวกันในฐานข้อมูลโดยตรง เพราะสัญญากลายเป็นโครงสร้างตารางที่ทุกฝ่ายต้องรู้ เปลี่ยนคอลัมน์หนึ่งคอลัมน์แล้วทุกภาษาที่ใช้ตารางนั้นพังพร้อมกัน

ข้อมูลที่ข้ามภาษามักเพี้ยนตรงไหน

ปัญหาของระบบหลายภาษาส่วนมากไม่ใช่ตัวโปรโตคอล แต่เป็นความต่างเล็ก ๆ ของวิธีที่แต่ละภาษาแทนข้อมูลชนิดเดียวกัน ตัวอย่างที่เจอบ่อยที่สุดคือเลขจำนวนเต็มขนาดใหญ่ ภาษาอย่าง Java และ Go มี int 64 บิต แต่ JavaScript ใช้เลขทศนิยมแบบ 64 บิตที่เก็บจำนวนเต็มได้แม่นยำถึง Number.MAX_SAFE_INTEGER เท่านั้น (253 − 1) เมื่อ JSON.parse เจอเลขเกินช่วงนี้ ค่าจะถูกปัดเงียบ ๆ โดยไม่มี error

console.log(Number.MAX_SAFE_INTEGER);                           // 9007199254740991
console.log(JSON.parse('{"id": 9007199254740993}').id);         // 9007199254740992  (ผิดไปหนึ่ง)
console.log(JSON.parse('{"id": "9007199254740993"}').id);       // 9007199254740993  (เป็น string ค่าครบ ส่งเป็น string ปลอดภัยกว่า)
console.log(BigInt("9007199254740993") + 1n);                   // 9007199254740994n

// ความยาวของ string เดียวกันต่างกันตามภาษา: JS นับหน่วย UTF-16
console.log("😀".length, [..."😀"].length);                      // 2 1

ในฝั่งภาษาอื่น ข้อความ “😀” ตัวเดียวมีความยาว 1 เมื่อนับเป็น code point ใน Python 3 แต่ฟังก์ชัน len ของ Go นับเป็นไบต์ได้ 4 ตารางข้างล่างสรุปเรื่องที่ควรตกลงในสัญญาตั้งแต่ต้น

เงินทศนิยมแบบ float ให้ผลคลาดเคลื่อนส่งเป็นหน่วยเล็กที่สุดแบบจำนวนเต็ม (สตางค์) หรือ decimal เป็น string พร้อมระบุสกุลเงิน

null, ไม่มีฟิลด์, ค่าว่างบางภาษาแยกสามอย่างนี้ไม่ได้ระบุในสัญญาว่าความหมายแต่ละแบบคืออะไร
Enumฝั่งหนึ่งเพิ่มค่าใหม่ อีกฝั่งไม่รู้จักให้ผู้รับมีค่า “unknown” และไม่พังเมื่อเจอค่าที่ไม่คุ้น

ตรวจข้อมูลที่ขอบ และอย่าไว้ใจฝั่งตรงข้าม

ข้อมูลที่มาจาก service อื่น ต่อให้เป็นของทีมเรา ก็ควรถือเป็น input ที่ไม่น่าไว้ใจ เพราะเวอร์ชันของอีกฝั่งอาจต่างจากที่คิด หลักที่ใช้กันคือ tolerant reader ผู้รับอ่านเฉพาะฟิลด์ที่ตัวเองต้องใช้ ฟิลด์แปลกที่เพิ่มมาทีหลังไม่ทำให้พัง แต่ฟิลด์ที่จำเป็นต้องตรวจให้ครบและปฏิเสธข้อความที่ผิดรูปแบบทันที พร้อมมี version ติดไปกับข้อความเพื่อให้รู้ว่าควรอ่านด้วยกติกาไหน

function parseOrderEvent(raw) {
  const data = typeof raw === "string" ? JSON.parse(raw) : raw;
  if (data === null || typeof data !== "object") throw new TypeError("event must be an object");

  const { schemaVersion, orderId, totalMinor, currency = "THB" } = data;   // ฟิลด์อื่นถูกมองข้าม
  if (schemaVersion !== 1) throw new RangeError(`unsupported schemaVersion: ${schemaVersion}`);
  if (typeof orderId !== "string" || orderId === "") throw new TypeError("orderId must be a non-empty string");
  if (!Number.isSafeInteger(totalMinor) || totalMinor < 0) throw new TypeError("totalMinor must be a safe non-negative integer");
  return { orderId, totalMinor, currency };
}

const attempts = [
  '{"schemaVersion":1,"orderId":"A100","totalMinor":25050,"newFieldFromOtherTeam":true}',
  '{"schemaVersion":1,"totalMinor":100}',
  '{"schemaVersion":1,"orderId":"A101","totalMinor":9007199254740993}',
  '{"schemaVersion":2,"orderId":"A102","totalMinor":100}',
];
for (const raw of attempts) {
  try { console.log("ok  :", parseOrderEvent(raw)); }
  catch (e) { console.log("fail:", e.name, "-", e.message); }
}

สัญญาที่เขียนเป็นไฟล์ ไม่ใช่เอกสารที่ลืมอัปเดต

สัญญาที่ดีเป็นไฟล์ที่เครื่องอ่านได้และอยู่ใน repository เช่น OpenAPI สำหรับ REST หรือ .proto สำหรับ gRPC เพื่อให้ CI ตรวจได้ว่าการแก้ครั้งนี้ทำให้ client เดิมพังหรือไม่ ใน Protocol Buffers ข้อที่ต้องจำคือ เลขประจำฟิลด์คือสิ่งที่ถูกใช้ตอนเข้ารหัสจริง ไม่ใช่ชื่อฟิลด์ ดังนั้นห้ามนำเลขที่เคยใช้แล้วกลับมาใช้กับฟิลด์ใหม่ และควร reserved เลขของฟิลด์ที่ลบไปแล้ว

// order.proto — ตัวอย่างประกอบ
syntax = "proto3";

message OrderPlaced {
  string order_id    = 1;
  int64  total_minor = 2;     // หน่วยเล็กที่สุด เช่น สตางค์
  string currency    = 3;
  reserved 4;                  // เคยใช้ฟิลด์ที่ถูกลบไปแล้ว ห้ามนำเลขนี้ไปใช้ซ้ำ
  string channel     = 5;
}

ส่วนเรื่องการติดตามคำขอข้ามภาษา มาตรฐานที่ใช้กันคือ OpenTelemetry สำหรับเก็บ trace และ metric และ W3C Trace Context ที่กำหนด header traceparent ให้ทุก service ส่งต่อ ID เดียวกัน ทำให้เห็นเส้นทางของคำขอหนึ่งตั้งแต่หน้าเว็บจนถึงฐานข้อมูล ไม่ว่าแต่ละช่วงเขียนด้วยภาษาอะไร

ตัดสินใจอย่างไรว่าจะเพิ่มภาษาใหม่หรือไม่

คำถามที่ควรตอบก่อนอนุมัติมีประมาณนี้ ปัญหานี้ภาษาที่มีอยู่แก้ได้ไหมถ้าแก้ได้แล้วช้าหรือยากแค่ไหน มีข้อมูลจากการวัดไหม หรือเป็นแค่ความเชื่อ ทีมมีคนที่ดูแลภาษานี้ได้ในระยะยาวกี่คน ถ้าคนเดียวลาออกจะเกิดอะไร ภาษานี้มีเส้นทาง deploy, monitoring, การสแกนช่องโหว่ และ library ที่ทีมเชื่อถือแล้วหรือยัง ต้องดูแล LTS หรือการอัปเดต runtime เพิ่มอีกเท่าไร และเมื่อสองฝั่งต้องคุยกัน สัญญาและเครื่องมือ generate code รองรับครบไหม องค์กรจำนวนมากแก้ด้วยการมี “paved road” คือกำหนดภาษาหลักสองสามตัวที่ platform team จัดเครื่องมือ template และ pipeline ให้ครบ ใครอยากใช้ภาษานอกรายการต้องเสนอเหตุผลและรับภาระดูแลเอง

ข้อผิดพลาดที่เจอบ่อย

ที่พบบ่อยคือแยกเป็น service หลายตัวแล้ว แต่แต่ละตัวยังต้อง deploy พร้อมกันเสมอ จึงเป็น distributed monolith ที่ได้ต้นทุนของระบบกระจายโดยไม่ได้ประโยชน์ ต่อมาคือไม่มี contract test ทำให้รู้ว่าสัญญาพังตอน production ไม่ใช่ตอน CI การเขียนกฎธุรกิจเดียวกันซ้ำในสองภาษาจนค่อย ๆ เพี้ยนจากกัน และการเรียกข้ามภาษาถี่ ๆ ในลักษณะ chatty ทำให้ latency สะสมโดยไม่เห็นในโค้ดฝั่งใดฝั่งหนึ่ง

Trade-offs

ใช้ภาษาเดียวทั้งระบบทำให้แบ่งปันโค้ด คน และเครื่องมือได้ง่ายสุด แต่ต้องยอมรับข้อจำกัดของภาษานั้นในทุกส่วน ใช้หลายภาษาได้เครื่องมือที่เหมาะกับงานแต่ละประเภท แต่ภาระเรื่องสัญญา การ monitor และคนเพิ่มขึ้นตามจำนวนภาษา และการเรียกข้ามภาษาในโปรเซสเดียวให้ความเร็วสูงแต่ผูกกับรายละเอียดของ build และ runtime มากที่สุด ไม่มีคำตอบเดียวสำหรับทุกบริบท ต้องตัดสินจากปัญหา ทีม และอายุของระบบ

สรุปสั้น ๆเส้นแบ่งระหว่างภาษาควรเป็นสัญญาที่เขียนเป็นไฟล์และตรวจใน CI ไม่ใช่โค้ดที่ใช้ร่วมกัน ระวังข้อมูลที่เพี้ยนข้ามภาษา โดยเฉพาะเลขใหญ่ เงิน เวลา และ Unicode ตรวจข้อมูลที่ขอบแบบ tolerant reader ส่ง trace context ต่อให้ครบ และเพิ่มภาษาใหม่เมื่อมีเหตุผลที่วัดได้และมีคนดูแลระยะยาว

มาตรฐานที่เป็นเอกสารอย่างเดียวจะค่อย ๆ ถูกลืม ถ้าอยากให้ใช้จริงต้องฝังเข้าไปในเครื่องมือที่ทุกคนใช้อยู่แล้ว ได้แก่ formatter ที่จัดรูปแบบอัตโนมัติ linter ที่จับรูปแบบโค้ดเสี่ยง pre-commit hook ที่รันตรวจก่อน commit CI ที่ปฏิเสธ PR ที่ไม่ผ่านการตรวจ และ template ของโปรเจกต์ใหม่ที่ตั้งค่าทุกอย่างมาให้ตั้งแต่ต้น วิธีนี้ทำให้ reviewer ไม่ต้องเสียเวลาคอมเมนต์เรื่องรูปแบบ และไปโฟกัสเรื่องที่คนต้องตัดสินจริง คือตรรกะ การออกแบบ และความเสี่ยง

# .github/workflows/ci.yml — ตัวอย่างขั้นตรวจที่ทุก repo ใช้ร่วมกันได้
name: ci
on: [pull_request]
jobs:
  quality:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: 22 }
      - run: npm ci
      - run: npm run lint            # ESLint: รูปแบบและโค้ดเสี่ยง
      - run: npm run format:check    # Prettier: รูปแบบต้องตรงกัน
      - run: npm test                # unit test
      - run: npm audit --audit-level=high   # ช่องโหว่ของ dependency

ไฟล์ข้างบนเป็นตัวอย่างเพื่ออธิบายหลักการ เวอร์ชันของ action และคำสั่งต้องปรับให้ตรงกับเครื่องมือที่องค์กรใช้จริง และควรตรวจเวอร์ชันล่าสุดจากเอกสารของ GitHub Actions ก่อนนำไปใช้

บันทึกการตัดสินใจ เพื่อให้คนใหม่รู้ว่าทำไม

กฎที่ไม่มีเหตุผลกำกับถูกท้าทายบ่อยและถูกยกเลิกบ่อยโดยไม่มีใครรู้ว่าเคยมีเรื่องอะไรเกิดขึ้น แนวทางที่หลายทีมใช้คือ Architecture Decision Record (ADR) เป็นเอกสารสั้น ๆ หนึ่งหน้าต่อหนึ่งการตัดสินใจ มีบริบท ตัวเลือกที่พิจารณา สิ่งที่เลือก และผลที่ตามมา เก็บไว้ใน repository ข้างโค้ด พอเงื่อนไขเปลี่ยนก็เขียน ADR ใหม่ที่แทนที่ตัวเก่า ไม่ต้องลบประวัติ

ADR-0007: ใช้รูปแบบ error response เดียวกันทุก API
สถานะ: Accepted (2026-03-02)
บริบท:   แต่ละทีมตอบ error คนละรูปแบบ client ต้องเขียนตัวแปลงหลายชุด
ตัวเลือก: (1) ปล่อยตามทีม (2) กำหนดรูปแบบกลางเป็น JSON ที่มี code, message, traceId
การตัดสินใจ: เลือก (2) และให้ library กลางสร้าง response ให้
ผลที่ตามมา: ทีมที่มี client เก่าต้องย้ายภายใน 2 ไตรมาส ดูแล library กลางเพิ่มหนึ่งตัว

กำหนดแล้ว ต้องมีทางเปลี่ยนและทางยกเว้น

มาตรฐานที่แข็งตัวเป็นปัญหาเท่ากับไม่มีมาตรฐาน ควรมีกระบวนการเสนอแก้กฎที่ชัดเจน เช่น เปิด pull request ไปที่ repository ของมาตรฐาน มีเจ้าของที่รับผิดชอบ และมีช่องทางขอยกเว้นอย่างมีเหตุผล พร้อมบันทึกว่ายกเว้นเรื่องอะไรถึงเมื่อไร เมื่อเปลี่ยนกฎที่กระทบโค้ดเดิมจำนวนมาก ให้ออกเครื่องมือช่วยย้ายไปด้วย เช่น codemod ที่แก้โค้ดตามกฎใหม่อัตโนมัติ และตั้ง rule เป็นแค่คำเตือนก่อนช่วงหนึ่งแล้วค่อยเปลี่ยนเป็นปฏิเสธ เพื่อไม่ให้ทุกทีมถูกบล็อกพร้อมกัน

Code review ในฐานะมาตรฐาน

Review ที่ดีมีเป้าหมายให้โค้ดโดยรวมของระบบดีขึ้นอย่างต่อเนื่อง ไม่ใช่ให้ผู้เขียนผ่านด่าน แนวปฏิบัติที่หลายองค์กรใช้ เช่น Google Engineering Practices เผยแพร่ไว้เป็นสาธารณะ ใจความสำคัญคือ PR ควรเล็กพอให้ตรวจละเอียดได้ reviewer ตอบกลับภายในเวลาที่ตกลงกัน แยกคอมเมนต์ที่ต้องแก้ ออกจากข้อเสนอแนะที่เลือกทำหรือไม่ก็ได้ และเมื่อเถียงกันไม่จบ ให้ย้อนไปดูมาตรฐานที่เขียนไว้ ถ้ากฎไม่ครอบคลุมเรื่องนั้นก็เสนอเพิ่มกฎ ไม่ต้องเถียงรอบหน้า

วัดผลอย่างระมัดระวัง

อยากรู้ว่ามาตรฐานได้ผลไหม ให้ดูสิ่งที่เกี่ยวข้องโดยตรง เช่น เวลาที่ PR รอ review สัดส่วนคอมเมนต์เรื่องรูปแบบใน review จำนวนครั้งที่ CI ล้มเพราะเรื่องที่ตรวจอัตโนมัติได้ และเวลาที่คนใหม่ใช้จนส่งงานชิ้นแรกได้ ตัวเลขเหล่านี้เป็นสัญญาณ ไม่ใช่เป้าหมายในตัวเอง ถ้านำไปใช้ตัดสินคนหรือตั้งโควตา คนจะปรับพฤติกรรมให้ตัวเลขดูดีโดยไม่ได้ทำให้งานดีขึ้นจริง เช่น แตก PR เป็นชิ้นเล็กไร้ความหมายเพื่อให้ผ่านเกณฑ์ขนาด

ข้อผิดพลาดที่เจอบ่อย

กำหนดมาตรฐานใหญ่ทีเดียวทั้งหมดโดยไม่ถามคนที่ต้องใช้ ทำให้ถูกต่อต้านและถูกเลี่ยง เขียนเป็นเอกสารยาวที่ไม่มีเครื่องมือตรวจ ใช้กฎกับโค้ดเดิมทั้งหมดทันทีจน CI แดงทั้งองค์กร ตั้งกฎเพราะความชอบของคนเขียนมาตรฐานโดยไม่มีเหตุผลรองรับ และไม่มีเจ้าของดูแล ทำให้ไม่มีใครอัปเดตเมื่อเครื่องมือหรือภาษาเปลี่ยนเวอร์ชัน

Trade-offs

มาตรฐานที่เข้มทำให้โค้ดทั้งองค์กรหน้าตาเหมือนกันและย้ายคนข้ามทีมง่าย แต่ลดอิสระของทีมที่มีบริบทเฉพาะและเพิ่มภาระการดูแลกฎ มาตรฐานที่หลวมให้ทีมคล่องตัวแต่ต้องจ่ายด้วยความไม่สม่ำเสมอ ทางสายกลางที่หลายองค์กรเลือกคือ กฎส่วนที่เกี่ยวกับความปลอดภัยและความถูกต้องบังคับแน่น ส่วนรูปแบบและโครงสร้างให้ค่าเริ่มต้นที่ดีแล้วเปิดช่องให้ทีมปรับด้วยเหตุผล

สรุปสั้น ๆเลือกกำหนดเฉพาะเรื่องที่ประหยัดเวลาทั้งองค์กรหรือลดความเสี่ยงจริง ฝังเข้าเครื่องมือ (formatter, linter, CI, template) แทนการพึ่งเอกสาร บันทึกเหตุผลด้วย ADR มีทางเสนอแก้และขอยกเว้น ใช้ codemod ช่วยย้ายโค้ดเดิม และใช้ตัวเลขเป็นสัญญาณ ไม่ใช่เป้าหมายที่ใช้ตัดสินคน

เป้าที่ดีวัดด้วย percentile ไม่ใช่ค่าเฉลี่ย

ค่าเฉลี่ยซ่อนผู้ใช้ที่ได้ประสบการณ์แย่ที่สุดไว้ ถ้าคำขอ 90% ตอบใน 100 ms และอีก 10% ตอบใน 2 วินาที ค่าเฉลี่ยดูเหมือนยอมรับได้ แต่ผู้ใช้หนึ่งในสิบคนเจอหน้าค้าง จึงนิยมตั้งเป้าที่ p95 หรือ p99 ซึ่งหมายถึงค่าที่ 95% หรือ 99% ของคำขอเร็วกว่าหรือเท่ากับ ในตัวอย่างข้างล่างข้อมูลเป็นตัวเลขที่สมมติขึ้นเพื่ออธิบาย ไม่ใช่ผลวัดจากระบบจริง

// latency สมมติ 100 ค่า: 90 ค่าที่ 100 ms และ 10 ค่าที่ 2000 ms
const latencies = [...Array(90).fill(100), ...Array(10).fill(2000)];

function percentile(values, p) {            // nearest-rank
  const sorted = [...values].sort((a, b) => a - b);
  const rank = Math.ceil((p / 100) * sorted.length);
  return sorted[Math.max(rank, 1) - 1];
}
const average = latencies.reduce((a, b) => a + b, 0) / latencies.length;

console.log("average:", average);                       // average: 290
console.log("p50:", percentile(latencies, 50));         // p50: 100
console.log("p95:", percentile(latencies, 95));         // p95: 2000

เร่งส่วนที่ไม่ใช่คอขวดได้ผลน้อยกว่าที่คิด

กฎของ Amdahl อธิบายว่าถ้าส่วนที่ปรับปรุงคิดเป็นสัดส่วน p ของเวลาทั้งหมด และทำให้ส่วนนั้นเร็วขึ้น s เท่า ความเร็วโดยรวมที่ได้คือ 1 / ((1 − p) + p / s) สูตรนี้บอกว่าต่อให้ส่วนนั้นเร็วขึ้นแบบไม่จำกัด ก็ไม่มีทางเกิน 1 / (1 − p) เท่า ตัวเลขในตัวอย่างเป็นการคำนวณทางคณิตศาสตร์ ไม่ใช่ผลวัดจากภาษาใดภาษาหนึ่ง

const overallSpeedup = (fractionOptimized, localSpeedup) =>
  1 / ((1 - fractionOptimized) + fractionOptimized / localSpeedup);

console.log(overallSpeedup(0.4, 10).toFixed(2));         // 1.56  ส่วนที่กินเวลา 40% เร็วขึ้น 10 เท่า
console.log(overallSpeedup(0.4, Infinity).toFixed(2));   // 1.67  เพดานสูงสุดถ้าส่วนนั้นเหลือเวลาเป็นศูนย์
console.log(overallSpeedup(0.9, 10).toFixed(2));         // 5.26  ส่วนที่กินเวลา 90% เร็วขึ้น 10 เท่า

ตรงนี้คือเหตุผลที่ต้อง profile ก่อน ถ้าส่วนที่จะเขียนใหม่กินเวลาแค่ 40% ของคำขอ ต่อให้ภาษาใหม่เร็วกว่าสิบเท่า ภาพรวมก็เร็วขึ้นประมาณ 1.5 เท่า ซึ่งอาจไม่คุ้มกับต้นทุนการเขียนใหม่ ทดสอบ และดูแลสองภาษา

Rewrite หรือ Refactor

การเขียนใหม่ทั้งระบบมักถูกมองว่าตัดปัญหาได้เด็ดขาด แต่มีความเสี่ยงที่เจอซ้ำ ๆ ระบบเก่ามีพฤติกรรมแฝงที่ไม่มีใครจดไว้และทีมใหม่ไม่รู้ ระหว่างเขียนต้องดูแลระบบเก่าควบคู่ไปด้วยทำให้ทำฟีเจอร์ใหม่ช้าลงสองต่อ และกำหนดเสร็จมักเลื่อน ทางเลือกที่ความเสี่ยงต่ำกว่าคือ ค่อย ๆ แทนที่ทีละส่วนด้วย strangler pattern ที่เห็นในหัวข้อ Refactor หรือแยกเฉพาะส่วนร้อนออกมาเป็นโมดูลหรือ service ที่มีสัญญาชัด แล้วเขียนใหม่เฉพาะส่วนนั้น เหตุผลที่พอฟังขึ้นสำหรับเขียนใหม่ทั้งหมด เช่น runtime หรือ framework ที่ใช้อยู่หมดการสนับสนุนและย้ายไม่ได้ ข้อจำกัดพื้นฐานของแพลตฟอร์มที่ปรับแล้วก็ยังไม่ถึงเป้า หรือหาคนดูแลเทคโนโลยีเดิมไม่ได้อีกแล้ว และต้องมีหลักฐานที่ตรวจสอบได้ประกอบ

Big-O บอกว่าจำนวนงานโตตามขนาด input อย่างไร ไม่ได้บอกเวลาเป็นวินาที โค้ด O(n) กับ O(n²) ที่ n = 10 ใช้เวลาต่างกันไม่ถึงที่ใครจะสังเกต แต่ที่ n = 100,000 ความต่างคือหนึ่งแสนเท่า จุดสำคัญของการอ่านโค้ดให้ออกคือ loop หนึ่งชั้นที่วนตามขนาดข้อมูลได้ O(n) loop ซ้อนกันสองชั้นได้ O(n²) การตัดช่วงค้นหาลงครึ่งหนึ่งทุกรอบได้ O(log n) และการเรียงลำดับได้ O(n log n) ส่วนการลองทุกชุดย่อยได้ O(2n) และการลองทุกลำดับได้ O(n!)

กลุ่มนี้เกิดบ่อยเพราะแต่ละบรรทัดดูเรียบง่าย ตัวอย่างแรกคือการต่อ array ด้วย spread ใน reduce ทุกรอบสร้าง array ใหม่และคัดลอกของเดิมทั้งหมด ตัวอย่างที่สองคือใช้ shift ดึงหัวคิวใน loop ซึ่งต้องขยับสมาชิกที่เหลือทุกครั้ง และตัวอย่างที่สามคือเรียก includes บน array ภายใน loop ผมลองรันสองแบบแรกบนเครื่องตัวเองด้วย n = 20,000 และ 200,000 ตามลำดับ ผลต่างขนาดหลายร้อยถึงหลายพันเท่า ตัวเลขจริงขึ้นกับเครื่องและเวอร์ชัน Node.js แต่ทิศทางของความต่างขยายตามขนาดข้อมูลเสมอ

// ช้า: ทุกรอบคัดลอกสมาชิกที่สะสมมาทั้งหมด → O(n²)
const slowCopy = (nums) => nums.reduce((acc, x) => [...acc, x], []);

// เร็ว: ใส่ท้าย array เดิม → O(n)
const fastCopy = (nums) => { const out = []; for (const x of nums) out.push(x); return out; };

// คิวด้วย shift: ทุกครั้งขยับสมาชิกที่เหลือ → O(n²) โดยรวม
function drainWithShift(items) { const q = [...items]; let count = 0; while (q.length) { q.shift(); count++; } return count; }

// คิวด้วยตัวชี้หัว: ไม่ขยับสมาชิก → O(n)
function drainWithHead(items) { let head = 0, count = 0; while (head < items.length) { head++; count++; } return count; }

// ตรวจซ้ำด้วย includes ใน loop → O(n²)  ,  ใช้ Set → O(n)
const hasDuplicateSlow = (nums) => nums.some((x, i) => nums.indexOf(x) !== i);
const hasDuplicateFast = (nums) => new Set(nums).size !== nums.length;

const data = Array.from({ length: 2000 }, (_, i) => i);
console.log(slowCopy(data).length, fastCopy(data).length);                 // 2000 2000
console.log(drainWithShift(data), drainWithHead(data));                    // 2000 2000
console.log(hasDuplicateSlow([1, 2, 3, 2]), hasDuplicateFast([1, 2, 3, 2]));   // true true
console.log(hasDuplicateSlow(data), hasDuplicateFast(data));                   // false false

อีกกับดักที่ไม่เกี่ยวกับเวลาโดยตรงคือการส่ง array ขนาดใหญ่เข้าฟังก์ชันด้วย spread เช่น Math.max(...arr) ซึ่งแปลง array เป็นอาร์กิวเมนต์ทีละตัว ถ้า array ใหญ่มากจะเกิด RangeError: Maximum call stack size exceeded ผมทดสอบกับ array 200,000 ตัวบน Node.js 22 แล้วเจอ error นี้ ขีดจำกัดที่แน่นอนขึ้นกับ engine และเวอร์ชัน ให้ใช้ reduce หรือ loop ธรรมดาแทนเมื่อข้อมูลอาจใหญ่ ตามที่เห็นในหัวข้อ Debugging ก่อนหน้านี้ ว่าปัญหาแบบนี้มักไม่โผล่ตอนทดสอบด้วยข้อมูลน้อย

เลือก Object, Map หรือ Set

ถ้าเป็นคีย์ที่รู้ล่วงหน้าและมีรูปร่างตายตัว เช่น { name, age } ใช้ object ถ้าใช้เป็นพจนานุกรมที่คีย์มาจากข้อมูลภายนอก ใช้ Map ซึ่งไม่ชนกับ constructor หรือ __proto__ รองรับคีย์เป็นอะไรก็ได้ และนับจำนวนได้ตรงด้วย size ถ้าต้องการแค่ “มีหรือไม่มี” หรือกำจัดค่าซ้ำ ใช้ Set

const unique = [...new Set([3, 1, 3, 2, 1])];
console.log(unique);                                  // [ 3, 1, 2 ]  (คงลำดับที่เจอครั้งแรก)

const asObject = { 2: "b", 1: "a", z: "z" };
console.log(Object.keys(asObject));                   // [ '1', '2', 'z' ]  (key ตัวเลขถูกเรียงขึ้นก่อน)
const asMap = new Map([[2, "b"], [1, "a"], ["z", "z"]]);
console.log([...asMap.keys()]);                       // [ 2, 1, 'z' ]  (คงลำดับที่ใส่)

console.log(new Set([NaN, NaN]).size, [NaN].includes(NaN), [NaN].indexOf(NaN));   // 1 true -1

บรรทัดสุดท้ายเป็นความต่างเล็ก ๆ ที่ข้อสอบชอบ Set และ includes ถือว่า NaN เท่ากับ NaN แต่ indexOf ใช้ === ซึ่ง NaN ไม่เท่ากับตัวเอง

Stack (LIFO) ใช้เมื่อสิ่งที่เพิ่งเข้ามาคือสิ่งที่ต้องจัดการก่อน เช่น call stack ของโปรแกรม ปุ่ม undo การเดินกลับของ DFS และการจับคู่วงเล็บ ใน JavaScript ใช้ array กับ push และ pop ได้เลย ทั้งสองเป็น O(1) แบบ amortized โจทย์ตรวจวงเล็บคือ ถ้าเจอวงเล็บเปิดก็ใส่ stack เจอวงเล็บปิดก็ดึงตัวบนสุดมาเทียบว่าเป็นคู่กันไหม และตอนจบ stack ต้องว่าง

function isValid(s) {
  const pairs = { ")": "(", "]": "[", "}": "{" };
  const stack = [];
  for (const ch of s) {
    if (ch === "(" || ch === "[" || ch === "{") {
      stack.push(ch);
    } else if (Object.hasOwn(pairs, ch)) {
      if (stack.pop() !== pairs[ch]) return false;   // stack ว่างจะได้ undefined ซึ่งไม่เท่ากับวงเล็บเปิดใด ๆ
    }
  }
  return stack.length === 0;
}

console.log(["()[]{}", "(]", "([)]", "((", "", ")", "{[()]}"].map(isValid));
// [ true, false, false, false, true, false, true ]

Queue: เข้าก่อน ออกก่อน

Queue (FIFO) ใช้กับงานที่ต้องทำตามลำดับมาถึง และเป็นหัวใจของ BFS ข้อควรระวังเฉพาะ JavaScript คือใช้ shift ดึงหัวคิวไม่ได้ดีนักเมื่อข้อมูลเยอะ เพราะโดยทั่วไปต้องขยับสมาชิกที่เหลือทุกครั้ง ทางออกที่ง่ายคือเก็บตัวชี้ตำแหน่งหัวคิวไว้ แล้วตัดส่วนที่ใช้แล้วทิ้งเป็นระยะ

class Queue {
  #items = [];
  #head = 0;
  enqueue(value) { this.#items.push(value); }
  dequeue() {
    if (this.isEmpty()) return undefined;
    const value = this.#items[this.#head];
    this.#items[this.#head] = undefined;          // ปล่อยการอ้างอิงให้ garbage collector เก็บได้
    this.#head++;
    if (this.#head > 32 && this.#head * 2 >= this.#items.length) {
      this.#items = this.#items.slice(this.#head); // ตัดส่วนที่ใช้แล้วทิ้ง เป็นระยะ
      this.#head = 0;
    }
    return value;
  }
  get size() { return this.#items.length - this.#head; }
  isEmpty() { return this.size === 0; }
}

for (let i = 1; i <= 100; i++) q.enqueue(i);
const out = [];
while (!q.isEmpty()) out.push(q.dequeue());
console.log(out.slice(0, 5), out.length, q.size, q.dequeue());   // [ 1, 2, 3, 4, 5 ] 100 0 undefined

Linked List: เส้นทางที่ต้องเดินตามตัวชี้

Linked list เก็บข้อมูลเป็นโหนดที่แต่ละตัวชี้ไปยังตัวถัดไป ใส่ที่หัวเป็น O(1) แต่หาตำแหน่งที่ n ต้องเดินจากหัวทีละตัว เป็น O(n) ในงานจริง array ของ JavaScript ใช้ได้ดีกว่าในเกือบทุกกรณีเพราะ engine ปรับมาให้เร็ว แต่โจทย์สอบชอบ linked list เพราะต้องคุมตัวชี้ให้ถูก เทคนิคที่ต้องรู้คือ ตัวชี้สองตัวเดินด้วยความเร็วต่างกัน (slow เดินทีละก้าว fast เดินทีละสองก้าว) ใช้หาโหนดกลางและตรวจว่ามีวงวนหรือไม่ ถ้า fast ไปถึงปลายได้ ไม่มีวงวน แต่ถ้ามีวงวนสุดท้าย fast จะไล่ทัน slow

class ListNode {
  constructor(value, next = null) { this.value = value; this.next = next; }
}
const fromArray = (values) => {
  let head = null;
  for (let i = values.length - 1; i >= 0; i--) head = new ListNode(values[i], head);
  return head;
};
const toArray = (head) => {
  const out = [];
  for (let node = head; node; node = node.next) out.push(node.value);
  return out;
};

function reverseList(head) {
  let prev = null, current = head;
  while (current) {
    const next = current.next;      // เก็บตัวถัดไปไว้ก่อน ไม่งั้นหลุดหลังเปลี่ยนตัวชี้
    current.next = prev;
    prev = current;
    current = next;
  }
  return prev;
}

function middleNode(head) {         // ถ้ามีจำนวนคู่ คืนตัวกลางตัวที่สอง
  let slow = head, fast = head;
  while (fast && fast.next) { slow = slow.next; fast = fast.next.next; }
  return slow;
}

function hasCycle(head) {
  let slow = head, fast = head;
  while (fast && fast.next) {
    slow = slow.next; fast = fast.next.next;
    if (slow === fast) return true;
  }
  return false;
}

console.log(hasCycle(looped), hasCycle(fromArray([1, 2, 3])), hasCycle(null));     // true false false

Hash Table: ใช้ key หา value ในเวลาเกือบคงที่

Hash table แปลง key เป็นตัวเลขด้วยฟังก์ชัน hash แล้วใช้เลขนั้นชี้ไปที่ช่อง (bucket) ที่เก็บข้อมูล ถ้าสอง key ได้ช่องเดียวกัน (collision) ต้องมีวิธีรับมือ แบบหนึ่งคือให้แต่ละช่องเก็บรายการหลายตัว (chaining) และเมื่อข้อมูลหนาแน่นเกินจะขยายจำนวนช่องแล้วกระจายใหม่ ทำให้การค้นเฉลี่ยยังใกล้ O(1) แต่ในกรณีแย่สุดที่ทุก key ชนช่องเดียว จะกลายเป็น O(n) ใน JavaScript ใช้ Map กับ Set ที่มีให้แล้ว ตัวอย่างข้างล่างเขียนเองเพื่อให้เห็นกลไก ไม่ได้มีไว้ใช้แทนของจริง

class HashTable {
  constructor(size = 8) {
    this.buckets = Array.from({ length: size }, () => []);
    this.count = 0;
  }
  #indexOf(key) {
    let hash = 5381;
    for (const ch of String(key)) hash = (hash * 33 + ch.codePointAt(0)) >>> 0;
    return hash % this.buckets.length;
  }
  set(key, value) {
    const bucket = this.buckets[this.#indexOf(key)];
    const entry = bucket.find((e) => e[0] === key);
    if (entry) { entry[1] = value; return; }
    bucket.push([key, value]);
    this.count++;
    if (this.count > this.buckets.length * 0.75) this.#resize();
  }
  get(key) {
    const entry = this.buckets[this.#indexOf(key)].find((e) => e[0] === key);
    return entry ? entry[1] : undefined;
  }
  #resize() {
    const entries = this.buckets.flat();
    this.buckets = Array.from({ length: this.buckets.length * 2 }, () => []);
    this.count = 0;
    for (const [k, v] of entries) this.set(k, v);
  }
}

for (let i = 0; i < 20; i++) table.set(`key${i}`, i * i);
table.set("key3", "updated");
console.log(table.get("key3"), table.get("key19"), table.get("missing"), table.count, table.buckets.length);
// updated 361 undefined 20 32

งานที่ hash table ช่วยให้เร็วขึ้นมีรูปแบบซ้ำ ๆ คือนับความถี่ จัดกลุ่ม ตรวจว่าเคยเจอแล้วหรือยัง และหาคู่ที่ต้องการ (อย่างโจทย์ Two Sum ในหัวข้อก่อนหน้า) ตัวอย่างการจัดกลุ่มคำที่สลับตัวอักษรกันได้ (anagram) ใช้คีย์เป็นคำที่เรียงตัวอักษรแล้ว คำที่เป็น anagram กันจะได้คีย์เดียวกัน

function groupAnagrams(words) {
  const groups = new Map();
  for (const word of words) {
    const key = [...word].sort().join("");
    if (!groups.has(key)) groups.set(key, []);
    groups.get(key).push(word);
  }
  return [...groups.values()];
}

// [["eat","tea","ate"],["tan","nat"],["bat"]]

ตัวอย่างรวม: LRU Cache

โจทย์ที่ออกบ่อยทั้งในข้อสอบและสัมภาษณ์คือ cache ที่มีที่จำกัด เมื่อเต็มให้ลบรายการที่ไม่ถูกใช้นานที่สุดออก (Least Recently Used) ต้องทำ get และ put ให้เร็ว ใน JavaScript มีทางลัดเพราะ Map จำลำดับที่ใส่เข้าไป วิธีคือทุกครั้งที่ใช้ key ใดให้ลบแล้วใส่ใหม่ เพื่อย้ายมาไว้ท้ายสุด ส่วนตัวที่ไม่ได้ใช้นานสุดจะอยู่หัวของ Map เสมอ ในภาษาที่ map ไม่จำลำดับ ต้องใช้ hash table ร่วมกับ doubly linked list ซึ่งเป็นรูปแบบที่โจทย์เดิมตั้งใจให้เขียน

class LRUCache {
  #capacity;
  #map = new Map();
  constructor(capacity) { this.#capacity = capacity; }

  get(key) {
    if (!this.#map.has(key)) return -1;
    const value = this.#map.get(key);
    this.#map.delete(key);
    this.#map.set(key, value);                 // ย้ายไปท้ายสุด = ใช้ล่าสุด
    return value;
  }

  put(key, value) {
    if (this.#map.has(key)) this.#map.delete(key);
    this.#map.set(key, value);
    if (this.#map.size > this.#capacity) {
      const oldestKey = this.#map.keys().next().value;   // หัวของ Map = ไม่ถูกใช้นานสุด
      this.#map.delete(oldestKey);
    }
  }
}

const cache = new LRUCache(2);
cache.put(1, 1); cache.put(2, 2);
console.log(cache.get(1));       // 1
cache.put(3, 3);                 // ไล่ key 2 ออก
console.log(cache.get(2));       // -1
cache.put(4, 4);                 // ไล่ key 1 ออก
console.log(cache.get(1), cache.get(3), cache.get(4));   // -1 3 4

Tree และ Binary Search Tree

Tree คือ graph ที่ไม่มีวงวนและมีโหนดราก (root) หนึ่งตัว แต่ละโหนดมีลูกได้หลายตัว ส่วน binary tree จำกัดให้มีลูกไม่เกินสองตัว คือซ้ายและขวา Binary Search Tree (BST) เพิ่มกฎว่าทุกค่าในฝั่งซ้ายน้อยกว่าโหนดนั้นและทุกค่าฝั่งขวามากกว่า ทำให้ค้นหาโดยตัดทิ้งครึ่งหนึ่งทุกชั้นได้ ซึ่งเป็น O(h) เมื่อ h คือความสูงของต้นไม้ ถ้าต้นไม้สมดุลจะเป็น O(log n) แต่ถ้าใส่ข้อมูลเรียงลำดับเข้าไปตรง ๆ ต้นไม้จะเอียงเป็นเส้นตรงและกลายเป็น O(n) จึงมี tree ที่ปรับสมดุลให้เองอย่าง AVL และ Red-Black ที่ฐานข้อมูลและไลบรารีหลายตัวใช้

การเดินผ่านทุกโหนดมีสี่แบบที่ต้องแยกให้ออก Pre-order เยี่ยมโหนดก่อนแล้วไปซ้ายและขวา In-order ซ้าย โหนด ขวา ซึ่งบน BST จะได้ค่าเรียงจากน้อยไปมาก Post-order ซ้าย ขวา แล้วโหนด เหมาะกับการลบหรือคำนวณจากลูกขึ้นมาหาพ่อ และ Level-order เดินทีละชั้นด้วย queue

class TreeNode {
  constructor(value) { this.value = value; this.left = null; this.right = null; }
}

class BST {
  root = null;
  insert(value) {
    const node = new TreeNode(value);
    if (!this.root) { this.root = node; return this; }
    let current = this.root;
    while (true) {
      if (value === current.value) return this;             // ไม่เก็บค่าซ้ำ
      const side = value < current.value ? "left" : "right";
      if (!current[side]) { current[side] = node; return this; }
      current = current[side];
    }
  }
  has(value) {
    let current = this.root;
    while (current) {
      if (value === current.value) return true;
      current = value < current.value ? current.left : current.right;
    }
    return false;
  }
}

const inOrder   = (n, out = []) => (n && (inOrder(n.left, out), out.push(n.value), inOrder(n.right, out)), out);
const preOrder  = (n, out = []) => (n && (out.push(n.value), preOrder(n.left, out), preOrder(n.right, out)), out);
const postOrder = (n, out = []) => (n && (postOrder(n.left, out), postOrder(n.right, out), out.push(n.value)), out);

function levelOrder(root) {
  if (!root) return [];
  const levels = [];
  let frontier = [root];
  while (frontier.length) {
    levels.push(frontier.map((n) => n.value));
    frontier = frontier.flatMap((n) => [n.left, n.right].filter(Boolean));
  }
  return levels;
}

const maxDepth = (n) => (n ? 1 + Math.max(maxDepth(n.left), maxDepth(n.right)) : 0);

console.log(preOrder(tree.root));      // [ 8, 3, 1, 6, 4, 10, 14 ]
console.log(postOrder(tree.root));     // [ 1, 4, 6, 3, 14, 10, 8 ]
console.log(JSON.stringify(levelOrder(tree.root)));   // [[8],[3,10],[1,6,14],[4]]
console.log(maxDepth(tree.root), tree.has(6), tree.has(7));   // 4 true false

DFS และ BFS

DFS (Depth-First Search) ลงลึกตามทางเดียวจนสุดแล้วค่อยย้อนกลับ ใช้ recursion หรือ stack เหมาะกับการหาว่าเชื่อมกันไหม นับกลุ่มที่เชื่อมกัน และตรวจวงวน BFS (Breadth-First Search) แผ่ไปทีละชั้นด้วย queue เหมาะกับการหาเส้นทางที่มีจำนวนเส้นเชื่อมน้อยที่สุดใน graph ที่ไม่มีน้ำหนัก ทั้งสองแบบใช้เวลา O(V + E) เมื่อ V คือจำนวนโหนด E คือจำนวนเส้นเชื่อม ข้อสำคัญคือต้องมี set ของโหนดที่เยี่ยมแล้ว ไม่งั้นเจอวงวนแล้วโปรแกรมวนไม่จบ

function buildGraph(edges, directed = false) {
  const graph = new Map();
  const ensure = (v) => { if (!graph.has(v)) graph.set(v, []); };
  for (const [a, b] of edges) {
    ensure(a); ensure(b);
    graph.get(a).push(b);
    if (!directed) graph.get(b).push(a);
  }
  return graph;
}

function bfsShortestPath(graph, start, goal) {
  if (!graph.has(start) || !graph.has(goal)) return -1;
  const dist = new Map([[start, 0]]);
  const queue = [start];
  for (let head = 0; head < queue.length; head++) {      // ใช้ตัวชี้หัวคิวแทน shift
    const node = queue[head];
    if (node === goal) return dist.get(node);
    for (const next of graph.get(node)) {
      if (!dist.has(next)) { dist.set(next, dist.get(node) + 1); queue.push(next); }
    }
  }
  return -1;
}

function countComponents(graph) {
  const seen = new Set();
  let count = 0;
  for (const start of graph.keys()) {
    if (seen.has(start)) continue;
    count++;
    const stack = [start];                                // DFS แบบใช้ stack
    seen.add(start);
    while (stack.length) {
      const node = stack.pop();
      for (const next of graph.get(node)) {
        if (!seen.has(next)) { seen.add(next); stack.push(next); }
      }
    }
  }
  return count;
}

const g = buildGraph([[1, 2], [2, 3], [3, 4], [1, 4], [5, 6]]);
console.log(bfsShortestPath(g, 1, 3), bfsShortestPath(g, 1, 6), bfsShortestPath(g, 1, 1));   // 2 -1 0
console.log(countComponents(g));                                                             // 2

// BFS บนตาราง: เดินสี่ทิศ 0 = ทางเดิน 1 = กำแพง คืนจำนวนก้าวน้อยสุดจากมุมซ้ายบนถึงมุมขวาล่าง
function shortestInGrid(grid) {
  const rows = grid.length, cols = grid[0].length;
  if (grid[0][0] === 1 || grid[rows - 1][cols - 1] === 1) return -1;
  const steps = Array.from({ length: rows }, () => Array(cols).fill(-1));
  steps[0][0] = 0;
  const queue = [[0, 0]];
  for (let head = 0; head < queue.length; head++) {
    const [r, c] = queue[head];
    for (const [dr, dc] of [[1, 0], [-1, 0], [0, 1], [0, -1]]) {
      const nr = r + dr, nc = c + dc;
      if (nr < 0 || nc < 0 || nr >= rows || nc >= cols) continue;
      if (grid[nr][nc] === 1 || steps[nr][nc] !== -1) continue;
      steps[nr][nc] = steps[r][c] + 1;
      queue.push([nr, nc]);
    }
  }
  return steps[rows - 1][cols - 1];
}
console.log(shortestInGrid([[0, 0, 0], [1, 1, 0], [0, 0, 0]]));   // 4
console.log(shortestInGrid([[0, 1], [1, 0]]));                    // -1

Topological Sort: เรียงงานที่พึ่งพากัน

ถ้า graph มีทิศทางและไม่มีวงวน (DAG) เราเรียงโหนดให้ทุกเส้นเชื่อมชี้จากหน้าไปหลังได้ ใช้กับเรื่องอย่างลำดับการ build ที่โมดูลหนึ่งต้องรอโมดูลที่มันพึ่งพาก่อน หรือลำดับวิชาที่ต้องเรียนก่อนหลัง วิธีของ Kahn คือนับว่าแต่ละโหนดมีเส้นเข้ากี่เส้น (in-degree) เริ่มจากโหนดที่ไม่มีเส้นเข้า ถอดมันออกแล้วลดจำนวนเส้นเข้าของเพื่อนบ้าน ถ้ามีโหนดไหนเหลือ 0 ก็ใส่คิวต่อ เมื่อจบแล้วถ้าจำนวนโหนดที่ได้ไม่ครบทุกตัว แสดงว่ามีวงวน และเรียงให้เสร็จไม่ได้

function topologicalOrder(count, prerequisites) {      // โหนด 0..count-1, [a, b] = ต้องทำ b ก่อน a
  const next = Array.from({ length: count }, () => []);
  const inDegree = Array(count).fill(0);
  for (const [a, b] of prerequisites) { next[b].push(a); inDegree[a]++; }

  const queue = [];
  for (let i = 0; i < count; i++) if (inDegree[i] === 0) queue.push(i);

  const order = [];
  for (let head = 0; head < queue.length; head++) {
    const node = queue[head];
    order.push(node);
    for (const n of next[node]) if (--inDegree[n] === 0) queue.push(n);
  }
  return order.length === count ? order : [];          // ไม่ครบ = มีวงวน
}

console.log(topologicalOrder(4, [[1, 0], [2, 0], [3, 1], [3, 2]]));   // [ 0, 1, 2, 3 ]
console.log(topologicalOrder(2, [[0, 1], [1, 0]]));                   // []

Merge Sort: แบ่งครึ่ง เรียง แล้วรวม

Merge sort แบ่ง array ครึ่งหนึ่งเรื่อย ๆ จนเหลือชิ้นเดียว แล้วรวมสองชิ้นที่เรียงแล้วเข้าด้วยกันด้วยการเทียบหัวของแต่ละชิ้นทีละตัว มีเวลา O(n log n) ในทุกกรณี เป็น stable แต่ใช้หน่วยความจำเพิ่ม O(n) ข้อดีคือคาดการณ์เวลาได้ ส่วนตัวขั้นตอน “รวม” เป็นฐานของโจทย์ยอดนิยม เช่น รวมลิสต์ที่เรียงแล้ว นับ inversion (คู่ที่อยู่ผิดลำดับ) ได้โดยเพิ่มตัวนับตอนรวม

function mergeSort(values) {
  if (values.length <= 1) return values;
  const mid = Math.floor(values.length / 2);
  const left = mergeSort(values.slice(0, mid));
  const right = mergeSort(values.slice(mid));

  const merged = [];
  let i = 0, j = 0;
  while (i < left.length && j < right.length) {
    // ใช้ <= เพื่อให้ตัวที่เท่ากัน ตัวซ้ายมาก่อน = stable
    merged.push(left[i] <= right[j] ? left[i++] : right[j++]);
  }
  while (i < left.length) merged.push(left[i++]);
  while (j < right.length) merged.push(right[j++]);
  return merged;
}
console.log(mergeSort([38, 27, 43, 3, 9, 82, 10]));   // [ 3, 9, 10, 27, 38, 43, 82 ]
console.log(mergeSort([]), mergeSort([5]), mergeSort([2, 2, 1]));   // [] [ 5 ] [ 1, 2, 2 ]

Quick Sort: เลือก pivot แล้วแบ่งสองฝั่ง

Quick sort เลือกค่าหนึ่งเป็น pivot แล้วจัดให้ค่าที่น้อยกว่าไปอยู่ซ้าย ที่มากกว่าไปอยู่ขวา แล้วทำซ้ำกับสองฝั่ง ทำงานใน array เดิมได้ ใช้หน่วยความจำเพิ่มน้อย เวลาเฉลี่ย O(n log n) แต่กรณีแย่สุดเป็น O(n²) เกิดเมื่อ pivot ที่เลือกแบ่งข้อมูลได้ไม่สมดุลทุกรอบ เช่น เลือกตัวแรกเป็น pivot เสมอแล้วข้อมูลเรียงมาแล้ว การสุ่ม pivot ช่วยลดโอกาสนี้ และ quick sort ไม่ stable ตัวอย่างข้างล่างทดสอบกับ array สุ่มหลายชุดเทียบกับ sort ที่มีให้ โดยใช้ตัวสร้างเลขสุ่มแบบกำหนด seed เพื่อให้ผลซ้ำได้

function quickSort(arr, random = Math.random) {
  const a = [...arr];
  (function sort(lo, hi) {
    if (lo >= hi) return;
    const pivotIndex = lo + Math.floor(random() * (hi - lo + 1));
    [a[pivotIndex], a[hi]] = [a[hi], a[pivotIndex]];        // ย้าย pivot ไปท้ายช่วง
    const pivot = a[hi];
    let store = lo;
    for (let i = lo; i < hi; i++) {
      if (a[i] < pivot) { [a[i], a[store]] = [a[store], a[i]]; store++; }
    }
    [a[store], a[hi]] = [a[hi], a[store]];                  // วาง pivot ในที่ที่ถูกต้อง
    sort(lo, store - 1);
    sort(store + 1, hi);
  })(0, a.length - 1);
  return a;
}

// ตัวสร้างเลขสุ่มแบบกำหนด seed (LCG) เพื่อให้ทดสอบซ้ำได้
function seeded(seed) { let s = seed; return () => (s = (s * 1664525 + 1013904223) % 4294967296) / 4294967296; }

const rnd = seeded(42);
let allOk = true;
for (let t = 0; t < 200; t++) {
  const size = Math.floor(rnd() * 60);
  const data = Array.from({ length: size }, () => Math.floor(rnd() * 20));   // มีค่าซ้ำเยอะ
  const expected = [...data].sort((x, y) => x - y);
  if (JSON.stringify(quickSort(data, rnd)) !== JSON.stringify(expected)) allOk = false;
}
console.log("quickSort ตรงกับ sort ทุกชุด:", allOk);   // true

Heap: ดึงค่าน้อยสุดหรือมากสุดได้เร็ว

Heap คือต้นไม้ไบนารีที่โหนดพ่อน้อยกว่าลูกเสมอ (min-heap) เก็บในรูป array ได้เลยโดยไม่ต้องมีตัวชี้ โหนดที่ index i มีพ่ออยู่ที่ (i − 1) >> 1 และลูกที่ 2i + 1 กับ 2i + 2 ใส่ค่าใหม่ไว้ท้าย array แล้วดันขึ้น (sift up) ดึงค่าบนสุดแล้วเอาตัวท้ายมาแทนและดันลง (sift down) ทั้งสองเป็น O(log n) ส่วนดูค่าบนสุดเป็น O(1) JavaScript ไม่มี heap ในตัว ข้อสอบจึงมักต้องเขียนเอง ตัวอย่างด้านล่างรับ comparator เพื่อใช้ได้ทั้ง min และ max

class Heap {
  #items = [];
  #before;                                   // (a, b) => true ถ้า a ควรอยู่ใกล้ยอดมากกว่า b
  constructor(before = (a, b) => a < b) { this.#before = before; }
  get size() { return this.#items.length; }
  peek() { return this.#items[0]; }
  push(value) {
    const a = this.#items;
    a.push(value);
    let i = a.length - 1;
    while (i > 0) {
      const parent = (i - 1) >> 1;
      if (!this.#before(a[i], a[parent])) break;
      [a[i], a[parent]] = [a[parent], a[i]];
      i = parent;
    }
  }
  pop() {
    const a = this.#items;
    if (a.length === 0) return undefined;
    const top = a[0];
    const last = a.pop();
    if (a.length > 0) {
      a[0] = last;
      let i = 0;
      while (true) {
        let best = i;
        const l = 2 * i + 1, r = 2 * i + 2;
        if (l < a.length && this.#before(a[l], a[best])) best = l;
        if (r < a.length && this.#before(a[r], a[best])) best = r;
        if (best === i) break;
        [a[i], a[best]] = [a[best], a[i]];
        i = best;
      }
    }
    return top;
  }
}

// Top-K: เก็บ min-heap ขนาด k ถ้ามีตัวใหม่ใหญ่กว่ายอด ให้แทนที่ → O(n log k) ไม่ต้องเรียงทั้งหมด
function topK(nums, k) {
  const h = new Heap();
  for (const x of nums) {
    if (h.size < k) h.push(x);
    else if (x > h.peek()) { h.pop(); h.push(x); }
  }
  const out = [];
  while (h.size) out.push(h.pop());
  return out.reverse();                      // มากไปน้อย
}

Dijkstra ด้วย heap

กลับไปที่ graph ที่เส้นเชื่อมมีน้ำหนัก (ตามที่ค้างไว้ในหัวข้อก่อน) Dijkstra ดึงโหนดที่ระยะสะสมน้อยสุดออกมาจาก heap ทุกครั้ง ถ้าระยะที่ดึงมาเก่ากว่าระยะที่รู้แล้วให้ข้าม (เพราะ heap ที่เขียนไว้ไม่มีการแก้ค่าของตัวที่อยู่ข้างใน จึงใส่รายการใหม่ซ้ำแทน) เวลารวม O((V + E) log V)

// ใช้คลาส Heap จากตัวอย่างด้านบน (ตัวเปรียบเทียบรับ [ระยะ, โหนด])
function dijkstra(edges, source) {
  const graph = new Map();
  for (const [from, to, weight] of edges) {
    if (!graph.has(from)) graph.set(from, []);
    if (!graph.has(to)) graph.set(to, []);
    graph.get(from).push([to, weight]);
  }
  const dist = new Map([[source, 0]]);
  const heap = new Heap((a, b) => a[0] < b[0]);           // [ระยะสะสม, โหนด]
  heap.push([0, source]);
  while (heap.size) {
    const [d, node] = heap.pop();
    if (d > dist.get(node)) continue;                     // รายการเก่า ข้าม
    for (const [next, weight] of graph.get(node)) {
      const candidate = d + weight;
      if (!dist.has(next) || candidate < dist.get(next)) {
        dist.set(next, candidate);
        heap.push([candidate, next]);
      }
    }
  }
  return Object.fromEntries(dist);
}

Binary Search บนคำตอบ

Binary search ไม่ได้ใช้ได้แค่กับ array ที่เรียงแล้ว ถ้าโจทย์ถามหา “ค่าน้อยที่สุดที่ทำให้เงื่อนไขหนึ่งเป็นจริง” และเงื่อนไขนั้นมีคุณสมบัติว่าจริงที่ค่าหนึ่งแล้วจะจริงต่อไปทุกค่าที่มากกว่า เราลองค่ากลาง ถ้าจริงก็ตัดฝั่งที่มากกว่าทิ้ง ถ้าไม่จริงก็ตัดฝั่งที่น้อยกว่า ตัวอย่างคือ รถบรรทุกต้องส่งพัสดุตามลำดับให้ครบภายใน D วัน ต้องมีความจุอย่างน้อยเท่าไร ความจุตั้งแต่ค่าที่ทำได้ขึ้นไปก็ทำได้ทั้งหมด จึงค้นด้วย binary search และตรวจแต่ละค่าด้วยการจำลองแบบง่าย ๆ ในเวลา O(n)

function lowerBound(sorted, target) {            // index แรกที่ค่า >= target (คืน length ถ้าไม่มี)
  let lo = 0, hi = sorted.length;
  while (lo < hi) {
    const mid = lo + ((hi - lo) >> 1);
    if (sorted[mid] < target) lo = mid + 1;
    else hi = mid;
  }
  return lo;
}

function minShipCapacity(weights, days) {
  const canShip = (capacity) => {
    let used = 1, load = 0;
    for (const w of weights) {
      if (load + w > capacity) { used++; load = 0; }
      load += w;
    }
    return used <= days;
  };
  let lo = Math.max(...weights);                         // ต้องบรรทุกชิ้นที่หนักสุดได้
  let hi = weights.reduce((a, b) => a + b, 0);           // ส่งทีเดียววันเดียวพอ
  while (lo < hi) {
    const mid = lo + ((hi - lo) >> 1);
    if (canShip(mid)) hi = mid; else lo = mid + 1;
  }
  return lo;
}

function bruteForce(weights, days) {
  for (let c = Math.max(...weights); ; c++) {
    let used = 1, load = 0;
    for (const w of weights) { if (load + w > c) { used++; load = 0; } load += w; }
    if (used <= days) return c;
  }
}

console.log(cases.map(([w, d]) => minShipCapacity(w, d)));                       // [ 15, 6, 3, 7 ]
console.log(cases.every(([w, d]) => minShipCapacity(w, d) === bruteForce(w, d)));  // true

ใช้ sort โดยไม่ใส่ comparator กับตัวเลข หรือใส่ comparator ที่คืน boolean ใช้ a - b กับค่าที่อาจเป็น Infinity หรือ NaN เรียงทั้ง array ทั้งที่ต้องการแค่ค่าสูงสุด k อันดับ และใช้ binary search กับเงื่อนไขที่ไม่ใช่แบบ “จริงแล้วจริงต่อเนื่อง” ซึ่งทำให้ตัดช่วงผิด ส่วนใน binary search เองที่ผิดบ่อยคือเลือกใช้ lo <= hi กับ lo < hi ไม่ตรงกับวิธีขยับ hi = mid หรือ hi = mid − 1 ทำให้วนไม่จบหรือข้ามคำตอบ

Trade-offs

ถ้าต้องการแค่ค่าสูงสุดหรือต่ำสุด สแกนรอบเดียว O(n) พอ ถ้าต้องการ top-k ที่ k เล็กกว่า n มาก heap ขนาด k ให้ O(n log k) ซึ่งดีกว่าการเรียงทั้งหมด O(n log n) แต่ถ้า k ใกล้ n การเรียงทั้งหมดก็อ่านง่ายกว่าและพอกัน ส่วน quickselect หาอันดับที่ k ได้ที่ O(n) โดยเฉลี่ยแต่เขียนยากกว่า merge sort ให้เวลาที่คาดเดาได้และ stable ส่วน quick sort ประหยัดหน่วยความจำแต่ไม่ stable และมีกรณีแย่ที่ O(n²) และถ้าข้อมูลเป็นจำนวนเต็มในช่วงแคบ การนับ (counting sort) ให้ O(n + k) ได้โดยไม่ต้องเทียบกันเลย

Dynamic Programming: จำคำตอบของปัญหาย่อยที่ซ้ำ

DP ใช้ได้เมื่อปัญหาแตกเป็นปัญหาย่อยที่ซ้อนทับกัน (overlapping subproblems) และคำตอบที่ดีที่สุดประกอบมาจากคำตอบที่ดีที่สุดของปัญหาย่อย (optimal substructure) ขั้นตอนคิดที่ใช้ซ้ำได้มีห้าข้อ นิยามว่า “สถานะ” หนึ่งตัวแทนอะไร เขียนสมการที่ต่อจากสถานะก่อนหน้าไปสถานะนี้ ตั้งค่าเริ่มต้น (base) ตัดสินใจลำดับการคำนวณ และบอกว่าคำตอบสุดท้ายอยู่ที่สถานะไหน มีสองรูปแบบคือ top-down (memoization) เขียน recursion ตามสมการแล้วจำผลไว้ และ bottom-up (tabulation) เติมตารางจากสถานะเล็กไปใหญ่ด้วย loop ซึ่งไม่เสี่ยง stack overflow และมักลดพื้นที่เก็บได้

// ไม่จำ: ตรงไปตรงมาแต่ซ้ำเยอะ → เวลาโตแบบเอ็กซ์โพเนนเชียล
function waysNaive(n) { return n <= 2 ? n : waysNaive(n - 1) + waysNaive(n - 2); }

// top-down: จำผลด้วย Map
function waysMemo(n, memo = new Map()) {
  if (n <= 2) return n;
  if (memo.has(n)) return memo.get(n);
  const result = waysMemo(n - 1, memo) + waysMemo(n - 2, memo);
  memo.set(n, result);
  return result;
}

// bottom-up และลดพื้นที่: เก็บแค่สองค่าล่าสุด → O(n) เวลา O(1) พื้นที่
function waysIterative(n) {
  if (n <= 2) return n;
  let prev = 1, curr = 2;
  for (let i = 3; i <= n; i++) [prev, curr] = [curr, prev + curr];
  return curr;
}

console.log(waysMemo(30), waysIterative(30));                  // 1346269 1346269

ตัวอย่างที่ต้องรู้: เหรียญ, ผลรวมสูงสุด, ลำดับย่อยร่วม

โจทย์ coin change ถามว่าต้องใช้เหรียญน้อยที่สุดกี่เหรียญจึงได้ยอดที่ต้องการ สถานะคือ dp[a] จำนวนเหรียญน้อยที่สุดสำหรับยอด a สมการคือ dp[a] = min(dp[a − coin] + 1) สำหรับทุกเหรียญที่ไม่เกินยอด ข้อสังเกตสำคัญคือ greedy (เลือกเหรียญใหญ่สุดก่อนเสมอ) ไม่ถูกต้องเสมอไป ถ้าเหรียญคือ 1, 3, 4 และยอด 6 greedy ได้ 4 + 1 + 1 สามเหรียญ แต่คำตอบที่ถูกคือ 3 + 3 สองเหรียญ Greedy ใช้ได้ต่อเมื่อพิสูจน์ได้ว่าการเลือกที่ดีที่สุดตรงหน้านำไปสู่คำตอบที่ดีที่สุดโดยรวม เช่นโจทย์เลือกกิจกรรมที่ไม่ทับกันให้ได้มากที่สุดด้วยการเลือกตัวที่จบเร็วที่สุดก่อน

function coinChange(coins, amount) {
  const dp = Array(amount + 1).fill(Infinity);
  dp[0] = 0;                                               // ยอด 0 ใช้ 0 เหรียญ
  for (let a = 1; a <= amount; a++) {
    for (const coin of coins) {
      if (coin <= a && dp[a - coin] + 1 < dp[a]) dp[a] = dp[a - coin] + 1;
    }
  }
  return dp[amount] === Infinity ? -1 : dp[amount];
}

function greedyCoins(coins, amount) {                      // เพื่อเทียบ: เลือกเหรียญใหญ่ก่อน
  let count = 0;
  for (const coin of [...coins].sort((a, b) => b - a)) {
    count += Math.floor(amount / coin);
    amount %= coin;
  }
  return amount === 0 ? count : -1;
}

// ผลรวมต่อเนื่องสูงสุด (Kadane): สถานะ = ผลรวมสูงสุดของช่วงที่ "จบที่" ตำแหน่งนี้
function maxSubarray(nums) {
  let endingHere = nums[0], best = nums[0];
  for (let i = 1; i < nums.length; i++) {
    endingHere = Math.max(nums[i], endingHere + nums[i]);  // เริ่มใหม่ หรือต่อจากเดิม
    best = Math.max(best, endingHere);
  }
  return best;
}
console.log(maxSubarray([-2, 1, -3, 4, -1, 2, 1, -5, 4]), maxSubarray([-3, -1, -2]));   // 6 -1

// ลำดับย่อยร่วมที่ยาวที่สุด (LCS): dp[i][j] = ความยาวของ LCS ของ a[0..i) กับ b[0..j)
function lcs(a, b) {
  const dp = Array.from({ length: a.length + 1 }, () => Array(b.length + 1).fill(0));
  for (let i = 1; i <= a.length; i++) {
    for (let j = 1; j <= b.length; j++) {
      dp[i][j] = a[i - 1] === b[j - 1] ? dp[i - 1][j - 1] + 1 : Math.max(dp[i - 1][j], dp[i][j - 1]);
    }
  }
  return dp[a.length][b.length];
}
console.log(lcs("abcde", "ace"), lcs("abc", "def"), lcs("", "x"));   // 3 0 0

Backtracking: ลองสร้างคำตอบทีละส่วน และถอยเมื่อเดินต่อไม่ได้

Backtracking เหมาะกับโจทย์ที่ต้องหาทุกคำตอบหรือคำตอบที่ตรงเงื่อนไข เช่น ทุกชุดย่อย ทุกลำดับสลับ หรือการวางราชินีบนกระดานหมากรุก แนวคิดคือเลือกทีละขั้น ลงลึกไปต่อ แล้วเมื่อจบหรือติดขัดก็ยกเลิกการเลือกล่าสุดแล้วลองทางอื่น (choose, explore, un-choose) หัวใจของความเร็วคือ pruning ตัดกิ่งที่เห็นแล้วว่าไม่มีทางถูกทิ้งตั้งแต่ต้น ตัวอย่าง generateParentheses ข้างล่างใส่วงเล็บปิดได้ก็ต่อเมื่อมีวงเล็บเปิดที่ยังไม่ถูกปิด จึงไม่เคยสร้างลำดับที่ผิดเลย

function subsets(nums) {
  const result = [];
  const path = [];
  (function walk(start) {
    result.push([...path]);                   // ต้องคัดลอก ไม่งั้นทุกตัวใน result ชี้ไป array เดียวกัน
    for (let i = start; i < nums.length; i++) {
      path.push(nums[i]);                     // เลือก
      walk(i + 1);                            // ลงลึก
      path.pop();                             // ยกเลิกการเลือก
    }
  })(0);
  return result;
}

function permutations(nums) {
  const result = [];
  const used = Array(nums.length).fill(false);
  const path = [];
  (function walk() {
    if (path.length === nums.length) { result.push([...path]); return; }
    for (let i = 0; i < nums.length; i++) {
      if (used[i]) continue;
      used[i] = true; path.push(nums[i]);
      walk();
      path.pop(); used[i] = false;
    }
  })();
  return result;
}

function generateParentheses(n) {
  const result = [];
  (function walk(current, open, close) {
    if (current.length === 2 * n) { result.push(current); return; }
    if (open < n) walk(current + "(", open + 1, close);
    if (close < open) walk(current + ")", open, close + 1);     // prune: ปิดได้เมื่อมีเปิดค้างอยู่
  })("", 0, 0);
  return result;
}

Two Pointers: สองตัวชี้เดินเข้าหากันหรือไปทางเดียวกัน

ใช้เมื่อข้อมูลเรียงอยู่แล้ว หรือโจทย์ถามเรื่องคู่หรือช่วงที่ขยับได้ทั้งสองปลาย ตัวอย่างคือหาคู่ที่บวกกันได้ค่าเป้าหมายใน array ที่เรียงแล้ว ถ้าผลรวมน้อยไปให้ขยับตัวซ้ายขึ้น ถ้ามากไปให้ขยับตัวขวาลง ทำให้ได้ O(n) โดยไม่ต้องใช้พื้นที่เพิ่ม ตรวจ palindrome ก็เป็นรูปแบบเดียวกัน คือเทียบตัวหัวกับตัวท้ายแล้วเดินเข้าหากัน

function pairWithSum(sorted, target) {
  let left = 0, right = sorted.length - 1;
  while (left < right) {
    const sum = sorted[left] + sorted[right];
    if (sum === target) return [left, right];
    if (sum < target) left++; else right--;
  }
  return null;
}
console.log(pairWithSum([1, 3, 4, 6, 8, 11], 10), pairWithSum([1, 2], 7));   // [ 2, 3 ] null

function isCleanPalindrome(text) {               // ข้ามตัวที่ไม่ใช่ตัวอักษรและตัวเลข ไม่สนตัวพิมพ์เล็กใหญ่
  const isAlnum = (ch) => /[a-z0-9]/i.test(ch);
  let left = 0, right = text.length - 1;
  while (left < right) {
    if (!isAlnum(text[left])) { left++; continue; }
    if (!isAlnum(text[right])) { right--; continue; }
    if (text[left].toLowerCase() !== text[right].toLowerCase()) return false;
    left++; right--;
  }
  return true;
}
console.log(isCleanPalindrome("A man, a plan, a canal: Panama"), isCleanPalindrome("race a car"), isCleanPalindrome(""));

Sliding Window: หน้าต่างที่ขยายและหดตามเงื่อนไข

รูปแบบนี้ใช้กับโจทย์เรื่องช่วงต่อเนื่อง หน้าต่างมีสองขอบคือซ้ายกับขวา ขยับขอบขวาเพื่อเพิ่มข้อมูลเข้ามา ถ้าเงื่อนไขเสียก็ขยับขอบซ้ายเพื่อลดจนกลับมาถูก แต่ละตำแหน่งถูกเยี่ยมไม่เกินสองครั้ง จึงเป็น O(n) ตัวอย่างคือหาช่วงข้อความที่ไม่มีอักษรซ้ำที่ยาวที่สุด และหาช่วงที่สั้นที่สุดที่ผลรวมไม่น้อยกว่าเป้าหมาย (ใช้ได้เมื่อตัวเลขเป็นบวกทั้งหมด เพราะการขยายหน้าต่างทำให้ผลรวมเพิ่มเสมอ)

function longestUniqueSubstring(text) {
  const lastSeen = new Map();                    // ตัวอักษร -> ตำแหน่งล่าสุดที่เจอ
  let start = 0, best = 0;
  for (let i = 0; i < text.length; i++) {
    const ch = text[i];
    if (lastSeen.has(ch) && lastSeen.get(ch) >= start) start = lastSeen.get(ch) + 1;   // ข้ามไปหลังตัวที่ซ้ำ
    lastSeen.set(ch, i);
    best = Math.max(best, i - start + 1);
  }
  return best;
}
console.log(["abcabcbb", "bbbbb", "pwwkew", "", "dvdf"].map(longestUniqueSubstring));   // [ 3, 1, 3, 0, 3 ]

function minSubarrayLen(target, nums) {           // nums เป็นบวกทั้งหมด
  let left = 0, sum = 0, best = Infinity;
  for (let right = 0; right < nums.length; right++) {
    sum += nums[right];
    while (sum >= target) {
      best = Math.min(best, right - left + 1);
      sum -= nums[left++];
    }
  }
  return best === Infinity ? 0 : best;
}
console.log(minSubarrayLen(7, [2, 3, 1, 2, 4, 3]), minSubarrayLen(100, [1, 2, 3]));   // 2 0

Prefix Sum กับ Hash Map: นับช่วงที่ผลรวมเท่ากับ k

ถ้าโจทย์ถามจำนวนช่วงต่อเนื่องที่ผลรวมเท่ากับ k และตัวเลขอาจติดลบ sliding window ใช้ไม่ได้ เพราะการขยายหน้าต่างไม่ได้ทำให้ผลรวมเพิ่มเสมอไป ให้คิดใหม่ว่าผลรวมช่วง (i, j] เท่ากับ prefix[j] − prefix[i] เราต้องการ prefix[j] − prefix[i] = k คือ prefix[i] = prefix[j] − k ระหว่างเดิน ให้เก็บว่าเคยเจอผลรวมสะสมแต่ละค่ากี่ครั้งใน Map แล้วถามว่ามีค่า prefix − k กี่ตัว ตัวอย่างข้างล่างเทียบกับวิธี brute force บนข้อมูลสุ่มเพื่อยืนยันว่าถูก

function countSubarraysWithSum(nums, k) {
  const seen = new Map([[0, 1]]);                // ผลรวมสะสม 0 เกิดขึ้นแล้วหนึ่งครั้ง (ช่วงว่างตอนเริ่ม)
  let prefix = 0, count = 0;
  for (const n of nums) {
    prefix += n;
    count += seen.get(prefix - k) ?? 0;
    seen.set(prefix, (seen.get(prefix) ?? 0) + 1);
  }
  return count;
}

function brute(nums, k) {
  let count = 0;
  for (let i = 0; i < nums.length; i++) {
    let sum = 0;
    for (let j = i; j < nums.length; j++) { sum += nums[j]; if (sum === k) count++; }
  }
  return count;
}

Monotonic Stack: หาตัวถัดไปที่มากกว่า

โจทย์อย่าง “อีกกี่วันกว่าอุณหภูมิจะสูงกว่าวันนี้” หรือ “ตัวเลขที่มากกว่าตัวถัดไปทางขวาคืออะไร” ถ้าทำ loop ซ้อนจะเป็น O(n²) แต่ถ้าเก็บ stack ของ index ที่ยังรอคำตอบไว้ โดยค่าตามนั้นลดลงเสมอ เมื่อเจอค่าที่ใหญ่กว่ายอด stack ก็ดึงตัวที่รอออกมาตอบทีละตัว แต่ละ index ถูกใส่และดึงอย่างละครั้งเดียว จึงเป็น O(n)

function daysUntilWarmer(temps) {
  const answer = Array(temps.length).fill(0);
  const waiting = [];                            // เก็บ index ที่ยังไม่เจอวันที่อุ่นกว่า
  for (let i = 0; i < temps.length; i++) {
    while (waiting.length && temps[i] > temps[waiting[waiting.length - 1]]) {
      const j = waiting.pop();
      answer[j] = i - j;
    }
    waiting.push(i);
  }
  return answer;
}
console.log(daysUntilWarmer([73, 74, 75, 71, 69, 72, 76, 73]));   // [ 1, 1, 4, 2, 1, 1, 0, 0 ]

function mergeIntervals(intervals) {
  const sorted = intervals.map(([a, b]) => [a, b]).sort((x, y) => x[0] - y[0]);   // คัดลอกก่อนเรียง ไม่แก้ input
  const merged = [];
  for (const [start, end] of sorted) {
    const last = merged[merged.length - 1];
    if (last && start <= last[1]) last[1] = Math.max(last[1], end);
    else merged.push([start, end]);
  }
  return merged;
}

console.log(JSON.stringify(mergeIntervals([[5, 6], [1, 10], [2, 3]])));              // [[1,10]]

รูปแบบข้อสอบที่เจอบ่อย

ข้อสอบที่ใช้ HackerRank มักผสมสามแบบ แบบแรกคือข้อเลือกตอบ (multiple choice) ทั้งตอบข้อเดียวและเลือกหลายข้อ ถามเรื่องอ่านโค้ดแล้วทายผล ความซับซ้อนของเวลา และแนวคิดของภาษา แบบที่สองคือโจทย์เขียนโค้ด (coding challenge) ที่เขียนฟังก์ชันให้ผ่านเทสต์เคสที่ซ่อนอยู่ และแบบที่สามคือ debugging ที่ให้โค้ดมาแล้วแก้ให้ผ่านเทสต์ ในใบรับรอง (certification) ของ HackerRank เองที่ตรวจไว้ เช่น Problem Solving (Basic/Intermediate) และ JavaScript (Basic) ระบุว่ามีโจทย์เขียนโค้ดสองข้อในเวลา 90 นาที แต่ข้อสอบที่บริษัททำเองจะกำหนดจำนวนข้อและเวลาต่างออกไป

วิธีที่ระบบให้คะแนน

โดยทั่วไปคะแนนของโจทย์เขียนโค้ดมาจากสัดส่วนเทสต์เคสที่ผ่าน และเทสต์เคสมักมีทั้งชุดเล็กที่เห็นได้ กับชุดใหญ่ที่ซ่อนไว้ ชุดใหญ่นี้เองที่ทำให้วิธี O(n²) ที่ผ่านตัวอย่างหมดแล้วยังโดน “Time limit exceeded” จึงต้องเทียบ complexity กับขนาด input สูงสุดเสมอ ส่วนโจทย์ที่ผ่านบางเทสต์ก็ยังได้คะแนนบางส่วน การส่งหลายรอบมักทำได้ แต่ต้องอ่านกติกาของข้อสอบนั้นอีกครั้ง เพราะบางผู้จัดจำกัดจำนวนครั้งหรือคิดคะแนนจากครั้งสุดท้าย

อ่าน input ใน Node.js

โจทย์ส่วนใหญ่ของ HackerRank ให้ template มาแล้วสำหรับอ่าน input และเรียกฟังก์ชันที่เราต้องเขียน จึงไม่ต้องเขียนส่วนอ่านเอง แต่ถ้าโจทย์ให้เขียนโปรแกรมเต็ม ๆ รูปแบบด้านล่างใช้ได้กับ Node.js ทั่วไป อ่านทั้งหมดครั้งเดียวแล้วแยกเป็นบรรทัด ซึ่งพอสำหรับข้อมูลระดับหลักแสนบรรทัด ส่วน trim() ช่วยกันเรื่องอักขระ \r ที่ติดมาเมื่อไฟล์ input มีการขึ้นบรรทัดแบบ Windows

// บรรทัดแรก: n   บรรทัดที่สอง: ตัวเลข n ตัวคั่นด้วยช่องว่าง
const lines = require("fs").readFileSync(0, "utf8").split("\n");
let at = 0;
const next = () => lines[at++].trim();

const n = parseInt(next(), 10);
const nums = next().split(" ").map(Number);
console.log(nums.reduce((sum, x) => sum + x, 0));

ถ้าจะใช้ readline ซึ่งเป็น API มาตรฐานของ Node.js ก็ทำได้เช่นกัน โดยเก็บบรรทัดจาก event line แล้วประมวลผลใน event close อย่าลืมว่า input ทั้งหมดเป็นข้อความ ต้องแปลงเป็นตัวเลขเองเสมอ และการพิมพ์ผลด้วย console.log ทีละบรรทัดหลายแสนครั้งช้ากว่าการรวมเป็นข้อความเดียวแล้วพิมพ์ครั้งเดียว

const out = [];
for (let i = 0; i < 5; i++) out.push(i * i);
console.log(out.join("\n"));     // พิมพ์ครั้งเดียว แทนการเรียก console.log ห้าครั้ง

แบ่งเวลาอย่างไร

วิธีที่ใช้ได้กับข้อสอบสองข้อต่อ 90 นาที คือใช้ห้านาทีแรกอ่านทั้งสองข้อแล้วเลือกข้อที่มั่นใจกว่าทำก่อน ทำวิธีที่ถูกต้องให้ผ่านตัวอย่างก่อน แล้วค่อยปรับเรื่องความเร็ว เหลือเวลาประมาณสิบนาทีสุดท้ายไว้ตรวจกรณีขอบและส่งคำตอบ ถ้าติดข้อใดเกินสามสิบนาทีโดยไม่คืบหน้า ให้ส่งสิ่งที่ผ่านบางเทสต์ไว้ก่อนแล้วไปข้อถัดไป เพราะคะแนนบางส่วนยังมีค่า

ข้อควรระวังเฉพาะ JavaScriptฟังก์ชัน sort() ที่ไม่ใส่ comparator จะเรียงตามข้อความ จึงได้ [10, 9, 1] ไม่ใช่ [1, 9, 10] ตัวเลข 64 บิตที่เกิน 9007199254740991 ต้องใช้ BigInt และ recursion ลึกเป็นหมื่นชั้นมีโอกาส stack overflow จึงควรเปลี่ยนเป็น loop กับ stack ของตัวเอง

Coding Lab: เขียนโค้ดแล้วรันเทสต์ในหน้านี้

โค้ดที่เขียนรันในเบราว์เซอร์ของคุณเอง (ใน Web Worker แยกจากหน้าหลัก) ไม่ถูกส่งไปที่ไหน แต่ละข้อตั้ง timeout ไว้ 4 วินาที เพราะเทสต์ขนาดใหญ่ถูกใส่ไว้เพื่อให้วิธีที่ช้าเกินไปล้มจริง ๆ เหมือนบนแพลตฟอร์ม ต้องประกาศฟังก์ชันหรือ class ตามชื่อที่โจทย์กำหนด ส่วนเทสต์ซ่อนบนแพลตฟอร์มจริงจะบอกแค่ผ่านหรือไม่ผ่าน แต่ในหน้านี้ตารางจะแสดงค่าที่คาดหวังเทียบกับค่าที่ได้ของทุกเคส

ถ้าเบราว์เซอร์ไม่รองรับ Web Worker ตัวรันจะทำงานในหน้าหลักแทน ซึ่ง loop ที่ไม่จบจะทำให้หน้าค้างจนต้องรีโหลด

Refactor กับ RewriteRefactor ปรับโครงสร้างโดยพฤติกรรมภายนอกเหมือนเดิม มีเทสต์คอยยืนยัน ส่วน Rewrite เขียนใหม่ทั้งชิ้น พฤติกรรมอาจต่างและความเสี่ยงสูงกว่า
Concurrency กับ ParallelismConcurrency คือจัดการหลายงานที่คืบหน้าสลับกันได้ ส่วน Parallelism คือหลายงานทำงานพร้อมกันจริงบนหลาย core Node.js แบบ single thread ทำ concurrency ด้วย event loop ส่วน worker thread ทำ parallelism
Promise.all กับ Promise.allSettledall ล้มทันทีเมื่อมีตัวหนึ่ง reject ส่วน allSettled รอทุกตัวแล้วรายงานผลแต่ละตัว
== กับ ===== แปลงชนิดก่อนเทียบ ส่วน === ไม่แปลง จึงควรใช้ === เป็นค่าเริ่มต้น
Map กับ ObjectMap รับ key ได้ทุกชนิด จำลำดับการใส่ และไม่ชนกับ prototype ส่วน Object เหมาะกับข้อมูลโครงสร้างคงที่และ JSON
slice กับ spliceslice คัดลอกบางส่วนโดยไม่แก้ array เดิม ส่วน splice แก้ array เดิม

BFS กับ DFSBFS เดินทีละชั้น ได้เส้นทางที่มีจำนวนขั้นน้อยสุดใน graph ที่ไม่มีน้ำหนัก ส่วน DFS ลงลึกก่อน เหมาะกับการสำรวจทุกกรณีและ backtracking

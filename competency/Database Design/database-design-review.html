NoSQL ไม่ได้แปลว่า “ไม่มี SQL”

คำนี้ย่อมาจาก “Not only SQL” หมายถึงฐานข้อมูลที่ไม่ได้เก็บข้อมูลเป็นตารางที่เชื่อมกันแบบ relational database เหตุผลที่คนเลือกใช้มักมีสามข้อ คือโครงสร้างข้อมูลเปลี่ยนบ่อย ต้องการความเร็วสูงมากสำหรับการอ่านเขียนแบบง่าย ๆ หรือต้องขยายออกไปหลายเครื่องได้ง่าย ข้อสอบชอบถามว่าโจทย์แบบไหนควรใช้ NoSQL ประเภทไหน ตารางนี้จึงเป็นสิ่งแรกที่ควรจำให้ได้

ประเภทเก็บข้อมูลแบบไหนตัวอย่างเหมาะกับ
Key-Valuekey หนึ่งตัวชี้ไปที่ค่าหนึ่งค่า ระบบไม่สนว่าข้างในเป็นอะไรRedis, Amazon DynamoDB, Memcachedsession, ตะกร้าสินค้า, cache, ตัวนับ, rate limit
Documentเอกสาร JSON หรือ BSON ที่ซ้อนกันได้ ค้นด้วยฟิลด์ข้างในได้MongoDB, Couchbase, Firestoreแคตตาล็อกสินค้าที่แต่ละชิ้นมีคุณสมบัติไม่เหมือนกัน, โปรไฟล์ผู้ใช้
Wide-columnแถวที่แต่ละแถวมีคอลัมน์ไม่เท่ากัน แบ่งตาม partition keyApache Cassandra, HBase, ScyllaDBข้อมูลเขียนเยอะมาก เช่น log, IoT, ประวัติการใช้งาน
Graphnode กับ edge ที่เชื่อมกันNeo4j, Amazon Neptuneเครือข่ายเพื่อน, ระบบแนะนำสินค้า, ตรวจจับการโกง

Key-Value ด้วย Redis

Redis เก็บข้อมูลไว้ในหน่วยความจำ จึงอ่านเขียนได้เร็วกว่าฐานข้อมูลที่ต้องไปอ่านจากดิสก์มาก (ตัวเลขจริงขึ้นกับเครือข่ายและขนาดข้อมูล) เวลาตั้งชื่อ key ให้ใช้รูปแบบ ประเภท:รหัส เช่น session:abc123 จะได้ค้นและลบเป็นกลุ่มได้ง่าย จุดเด่นของ Redis คือมีชนิดข้อมูลให้เลือกหลายแบบ ไม่ได้มีแค่ string และตั้งเวลาหมดอายุให้แต่ละ key ได้

# เก็บ session ไว้ 30 นาที (1800 วินาที) หมดเวลาแล้ว Redis ลบให้เอง
SET session:abc123 '{"userId":42,"role":"customer"}' EX 1800
GET session:abc123
TTL session:abc123                 # เหลือเวลากี่วินาที

# ตัวนับยอดเข้าชมสินค้า INCR ทำงานแบบ atomic ไม่ต้องอ่านมาบวกเอง
INCR views:product:1001

# ตะกร้าสินค้าเก็บเป็น Hash: field คือรหัสสินค้า value คือจำนวน
HSET cart:42 P1001 2 P2002 1
HINCRBY cart:42 P1001 1
HGETALL cart:42

# Sorted Set ทำ leaderboard สินค้าขายดี
ZINCRBY bestseller 3 P1001
ZREVRANGE bestseller 0 9 WITHSCORES   # 10 อันดับแรก
ข้อมูลใน Redis อาจหายเมื่อไฟดับหรือ restartRedis เก็บข้อมูลไว้ในหน่วยความจำ ถ้าไม่ได้ตั้งค่าให้บันทึกลงดิสก์ (persistence) ข้อมูลจะหายทั้งหมดเมื่อเครื่อง restart การบันทึกลงดิสก์มีสองแบบ แบบแรกคือ RDB ถ่าย snapshot เก็บไว้เป็นระยะ ถ้าไฟดับจะเสียข้อมูลที่เขียนหลัง snapshot ครั้งล่าสุด แบบที่สองคือ AOF บันทึกทุกคำสั่งเขียน เสียข้อมูลน้อยกว่าแต่ไฟล์ใหญ่กว่า ข้อมูลที่หายไม่ได้ เช่น ออร์เดอร์หรือยอดเงิน จึงไม่ควรใช้ Redis เป็นที่เก็บหลัก

Document ด้วย MongoDB

ใน MongoDB ข้อมูลหนึ่งชิ้นเรียกว่าเอกสาร (document) และถูกเก็บรวมไว้ใน collection เอกสารเทียบได้กับแถวในตาราง ต่างกันตรงที่แต่ละชิ้นไม่จำเป็นต้องมีฟิลด์เหมือนกัน เช่น เสื้อมีไซซ์และสี ส่วนโน้ตบุ๊กมี CPU กับ RAM ก็เก็บใน collection เดียวกันได้โดยไม่ต้องสร้างคอลัมน์ว่างจำนวนมาก ทุกเอกสารมีฟิลด์ _id เป็น primary key เสมอ

db.products.insertOne({
  sku: "TS-001", name: "เสื้อยืดคอกลม", price: 290,
  attrs: { color: "navy", sizes: ["S", "M", "L"] },
  tags: ["cotton", "unisex"]
})

// ค้นด้วยฟิลด์ที่ซ้อนอยู่ข้างใน และเลือกเฉพาะฟิลด์ที่ต้องการ (projection)
db.products.find({ "attrs.color": "navy", price: { $lte: 300 } }, { name: 1, price: 1 })

// แก้บางฟิลด์ด้วย $set และเพิ่มค่าด้วย $inc
db.products.updateOne({ sku: "TS-001" }, { $set: { price: 259 }, $inc: { stock: -1 } })

// สร้าง index ไม่งั้นทุกการค้นจะไล่อ่านทั้ง collection
db.products.createIndex({ sku: 1 }, { unique: true })

ฝังไว้ด้วยกัน (embed) หรือแยกแล้วอ้างถึง (reference)

นี่คือการตัดสินใจที่สำคัญที่สุดของการออกแบบ document และเป็นคำถามที่เจอบ่อย หลักคิดง่าย ๆ คือ ข้อมูลที่อ่านพร้อมกันเสมอและมีจำนวนจำกัด ให้ฝังไว้ในเอกสารเดียว ส่วนข้อมูลที่เพิ่มขึ้นเรื่อย ๆ ไม่มีขีดจำกัด หรือถูกใช้จากหลายที่ ให้แยกเป็น collection แล้วเก็บแค่ id อ้างถึง

สถานการณ์ควรทำเหตุผล
ที่อยู่จัดส่ง 1–3 ที่ของลูกค้าEmbedอ่านพร้อมโปรไฟล์ทุกครั้ง จำนวนน้อย
รายการสินค้าในออร์เดอร์Embed พร้อม snapshot ราคาออร์เดอร์ต้องจำราคาตอนซื้อ แม้ราคาสินค้าจะเปลี่ยนทีหลัง
รีวิวของสินค้าขายดีที่มีเป็นหมื่นReference (collection reviews)จำนวนเพิ่มขึ้นได้ไม่จำกัด และเอกสารหนึ่งชิ้นของ MongoDB ใหญ่ได้ไม่เกิน 16 MB
ข้อมูลผู้ขายที่สินค้าหลายพันชิ้นใช้ร่วมกันReferenceถ้า embed ต้องแก้หลายพันที่เวลาผู้ขายเปลี่ยนชื่อร้าน
จำสั้น ๆใช้ NoSQL ก็ยังต้องออกแบบข้อมูล แต่ให้เริ่มจากถามว่าข้อมูลจะถูกอ่านและเขียนแบบไหน (access pattern) แล้วจัดรูปข้อมูลให้ query ที่ใช้บ่อยดึงได้ในครั้งเดียว

CUSTOMER ||--o{ ORDERS      : places      ลูกค้า 1 คน มีออร์เดอร์ 0 ถึงหลายใบ
ORDERS   ||--|{ ORDER_ITEM  : contains    ออร์เดอร์ 1 ใบ มีอย่างน้อย 1 รายการ
PRODUCT  ||--o{ ORDER_ITEM  : "sold in"   สินค้า 1 ชิ้น อยู่ใน 0 ถึงหลายรายการ

สังเกตว่า ORDERS กับ PRODUCT เป็นความสัมพันธ์แบบหลายต่อหลาย (M:N) ออร์เดอร์หนึ่งใบมีสินค้าหลายชิ้น และสินค้าหนึ่งชิ้นอยู่ในหลายออร์เดอร์ ฐานข้อมูลเชิงสัมพันธ์เก็บ M:N ตรง ๆ ไม่ได้ เราจึงต้องมีตารางกลาง (junction หรือ associative table) อย่าง ORDER_ITEM มาเชื่อมระหว่างสองตาราง ตารางกลางนี้ยังใช้เก็บข้อมูลที่เป็นของความสัมพันธ์นั้นได้ด้วย เช่น จำนวนที่ซื้อและราคาตอนซื้อ

แปลง ER เป็น Schema

CREATE TABLE customer (
  customer_id  BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  email        VARCHAR(255) NOT NULL UNIQUE,
  full_name    VARCHAR(200) NOT NULL,
  created_at   TIMESTAMPTZ  NOT NULL DEFAULT now()
);

CREATE TABLE product (
  product_id   BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  sku          VARCHAR(40)  NOT NULL UNIQUE,
  name         VARCHAR(200) NOT NULL,
  price        NUMERIC(12,2) NOT NULL CHECK (price >= 0)   -- เงินใช้ NUMERIC ห้ามใช้ FLOAT
);

CREATE TABLE orders (
  order_id     BIGINT GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
  customer_id  BIGINT NOT NULL REFERENCES customer(customer_id),
  status       VARCHAR(20) NOT NULL DEFAULT 'PENDING'
               CHECK (status IN ('PENDING','PAID','SHIPPED','CANCELLED')),
  created_at   TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE order_item (
  order_id     BIGINT NOT NULL REFERENCES orders(order_id) ON DELETE CASCADE,
  product_id   BIGINT NOT NULL REFERENCES product(product_id),
  quantity     INT    NOT NULL CHECK (quantity > 0),
  unit_price   NUMERIC(12,2) NOT NULL,          -- ราคาตอนซื้อ ไม่ดึงจาก product
  PRIMARY KEY (order_id, product_id)            -- composite key
);

บรรทัด ON DELETE CASCADE หมายความว่าถ้าลบออร์เดอร์ รายการในออร์เดอร์จะถูกลบตามไปด้วย ซึ่งสมเหตุสมผลเพราะ order_item เป็น weak entity แต่ฝั่ง product เราไม่ใส่ cascade เพราะไม่อยากให้การลบสินค้าไปลบประวัติการขาย ถ้าพยายามลบสินค้าที่เคยถูกสั่งซื้อไปแล้ว ฐานข้อมูลจะไม่ยอมให้ลบ (ค่าเริ่มต้นคือ NO ACTION หรือ RESTRICT)

ชนิดข้อมูลที่ชอบออกสอบเงินให้ใช้ DECIMAL/NUMERIC เพราะ FLOAT เก็บทศนิยมเป็นเลขฐานสองแบบค่าประมาณ ผลของ 0.1 + 0.2 จึงไม่เท่ากับ 0.3 พอดี วันเวลาให้ใช้ TIMESTAMPTZ หรือเก็บเป็นเวลา UTC ส่วน CHAR(n) เติมช่องว่างให้ครบความยาวเสมอ ขณะที่ VARCHAR(n) เก็บตามความยาวจริง

ก่อนเลือกวิธีสำรอง ต้องถามเจ้าของระบบให้ได้สองคำตอบ RPO (Recovery Point Objective) คือยอมเสียข้อมูลย้อนหลังได้มากสุดเท่าไร ถ้าตอบว่า 1 ชั่วโมง เราต้องสำรองอย่างน้อยทุกชั่วโมง RTO (Recovery Time Objective) คือระบบหยุดทำงานได้นานที่สุดเท่าไรก่อนต้องกลับมาใช้งานได้ สมมติว่าธุรกิจตอบว่า 15 นาที การกู้ไฟล์ขนาด 2 TB ที่ใช้เวลาสามชั่วโมงก็ไม่ตอบโจทย์ จำง่าย ๆ ว่า RPO คือปริมาณข้อมูลที่ยอมให้หาย ส่วน RTO คือเวลาที่ใช้กู้ระบบกลับมา

คำสั่ง backup และ restore ที่ควรเขียนเป็น

# PostgreSQL: -Fc คือ custom format บีบอัดและ restore เลือกบางตารางได้
pg_dump -h db.internal -U backup_user -d shop -Fc -f shop_2026-11-13.dump
pg_restore -d shop_restore --clean --if-exists shop_2026-11-13.dump
pg_dumpall --globals-only > roles.sql        # role และสิทธิ์ ไม่ได้อยู่ใน pg_dump
pg_basebackup -D /backup/base -Ft -z -P       # physical backup ทั้ง cluster

# MySQL: --single-transaction ได้ snapshot ที่สอดคล้องกันโดยไม่ล็อกตาราง InnoDB
mysqldump --single-transaction --routines --triggers shop > shop.sql
mysql shop_restore < shop.sql

# MongoDB
mongodump --uri="mongodb://backup@db:27017/shop" --gzip --archive=shop.gz
mongorestore --gzip --archive=shop.gz --nsFrom="shop.*" --nsTo="shop_restore.*"

# SQL Server
BACKUP DATABASE Shop TO DISK = 'D:\backup\Shop_full.bak' WITH COMPRESSION, CHECKSUM;
BACKUP LOG Shop TO DISK = 'D:\backup\Shop_log.trn';

# SQLite: ใช้คำสั่ง .backup ไม่ใช่ cp ไฟล์ขณะมีคนเขียน
sqlite3 shop.db ".backup 'shop_backup.db'"

ปัญหาที่ตามมามีสามแบบ เรียกรวมว่า anomaly

  • Update anomaly ลูกค้าเปลี่ยนชื่อ ต้องแก้ทุกแถว ถ้าแก้ไม่ครบชื่อจะไม่ตรงกัน (เหมือนแถว 5001 กับ 5002)
  • Insert anomaly อยากเพิ่มสินค้าใหม่ที่ยังไม่มีใครซื้อ ทำไม่ได้เพราะต้องมีออร์เดอร์ก่อน
  • Delete anomaly ลบออร์เดอร์เดียวของลูกค้าคนหนึ่ง ข้อมูลลูกค้าคนนั้นหายไปด้วย

Normalization คือการแตกตารางตามกฎทีละขั้น เพื่อให้ข้อเท็จจริงหนึ่งเรื่องถูกเก็บไว้ที่เดียว

Functional dependency คือพื้นฐานของทุก normal form

ถ้ารู้ค่า A แล้วบอกค่า B ได้แน่นอนเสมอ จะเขียนว่า A → B เช่น customer_email → customer_name หรือ (order_id, product_id) → quantity normal form ทุกระดับคือการตรวจว่าแต่ละคอลัมน์ขึ้นอยู่กับอะไร

การ normalize ทำให้ข้อมูลถูกต้อง แต่ query ต้อง join หลายตาราง ระบบที่มีการอ่านหนักมากบางครั้งจึงยอมเก็บข้อมูลซ้ำบางส่วน เช่น เก็บ order_total ไว้ในตาราง orders แทนการ SUM ทุกครั้ง หรือเก็บ review_count ไว้ใน product การ denormalize ควรทำหลังจากวัดแล้วว่าช้าจริง และต้องมีวิธีทำให้ค่าที่เก็บซ้ำตรงกันเสมอ เช่น อัปเดตใน transaction เดียวกัน ใช้ trigger หรือใช้ materialized view ส่วนงานวิเคราะห์ข้อมูลอย่าง data warehouse นิยม denormalize เป็น star schema ซึ่งจะเจออีกครั้งในหัวข้อ 13

Cache-aside (lazy loading)

แอปถาม Redis ก่อน ถ้าไม่เจอ (cache miss) จึงอ่านจากฐานข้อมูล แล้วเขียนผลลง cache พร้อมตั้ง TTL ส่วนตอนแก้ข้อมูล ให้เขียนฐานข้อมูลก่อน แล้วลบ key นั้นใน cache ทิ้ง (ไม่ใช่เขียนค่าใหม่ลง cache) คนที่อ่านครั้งถัดไปจะได้ค่าล่าสุดจากฐานข้อมูลเอง

async function getProduct(id) {
  const key = `product:${id}`;
  const hit = await redis.get(key);
  if (hit) return JSON.parse(hit);                       // cache hit

  const row = await pg.query('SELECT * FROM product WHERE product_id = $1', [id]);
  if (row.rowCount) await redis.set(key, JSON.stringify(row.rows[0]), 'EX', 300 + rand(60));
  return row.rows[0] ?? null;                            // สุ่ม TTL กันหมดอายุพร้อมกัน
}

async function updatePrice(id, price) {
  await pg.query('UPDATE product SET price = $1 WHERE product_id = $2', [price, id]);
  await redis.del(`product:${id}`);                      // ลบ ไม่ใช่ set ค่าใหม่
}

ปัญหา dual write

ถ้าโค้ดเขียน PostgreSQL แล้วเขียน MongoDB หรือส่งข้อความเข้า Kafka ต่อทันที สักวันขั้นที่สองจะล้มเหลว เช่น เครือข่ายหลุดหรือ service ล่ม แล้วข้อมูลสองที่จะไม่ตรงกันโดยไม่มีใครรู้ตัว เพราะไม่มี transaction ที่ครอบสองระบบได้ วิธีแก้ที่ใช้กันจริงมีสองแบบ

วิธีทำงานอย่างไรข้อควรรู้
Transactional Outboxบันทึกข้อมูลจริงพร้อมกับแถวในตาราง outbox ใน transaction เดียวกัน แล้วให้อีกโปรเซสหนึ่งอ่าน outbox และส่งต่อไปยังระบบอื่นส่งอย่างน้อยหนึ่งครั้ง (at-least-once) ข้อความจึงอาจซ้ำ ผู้รับต้องรับซ้ำได้โดยผลไม่เปลี่ยน (idempotent)
Change Data Capture (CDC)อ่าน log ของฐานข้อมูล (WAL, binlog) แล้วแปลงเป็น event เช่น Debeziumไม่ต้องแก้โค้ดแอป แต่ event จะผูกกับโครงสร้างตาราง ถ้าตารางเปลี่ยน ฝั่งผู้รับก็ได้รับผลกระทบ
BEGIN;
  INSERT INTO orders (...) VALUES (...);
  INSERT INTO outbox (aggregate_id, event_type, payload)
  VALUES (5001, 'OrderPlaced', '{"orderId":5001,"total":580}');
COMMIT;   -- สำเร็จหรือล้มเหลวไปด้วยกัน ไม่มีกรณีออร์เดอร์มีแต่ event หาย

บางครั้งฐานข้อมูลเดียวก็พอ

ถ้าแค่ต้องการเก็บคุณสมบัติสินค้าที่ไม่ตายตัว PostgreSQL มีคอลัมน์ JSONB ที่ค้นและทำ index (GIN) ได้ MySQL ก็มีชนิด JSON การใช้ JSONB ทำให้ยังใช้ transaction และ join ได้ในฐานข้อมูลเดียว ไม่ต้องดูแลฐานข้อมูลเพิ่มอีกตัว เพราะทุกฐานข้อมูลที่เพิ่มเข้ามาหมายถึงงาน backup, monitoring และความปลอดภัยที่เพิ่มขึ้น รวมถึงต้องมีคนที่ดูแลเป็น

SELECT name FROM product WHERE attrs @> '{"color":"navy"}';
CREATE INDEX idx_product_attrs ON product USING GIN (attrs);
ข้อมูลในแต่ละระบบจะตรงกันช้ากว่ากันเล็กน้อยเมื่อข้อมูลไหลจากฐานข้อมูลหลักไปยัง search index หรือ cache จะมีช่วงสั้น ๆ ที่ข้อมูลยังไม่ตรงกันเสมอ (eventual consistency) จึงต้องออกแบบหน้าจอให้รองรับ เช่น หลังลูกค้าแก้ที่อยู่ ให้แสดงค่าที่เพิ่งบันทึกจากฐานข้อมูลหลัก ไม่ใช่อ่านจาก index

อย่าเดา ให้ดู query plan ก่อน

เวลามีคนบอกว่า “หน้าประวัติการสั่งซื้อช้า” ขั้นแรกคือหา query ที่ช้าจริง (slow query log หรือ pg_stat_statements) แล้วรัน EXPLAIN ANALYZE ซึ่งจะรัน query จริงและบอกว่าแต่ละขั้นใช้เวลาเท่าไร ระวังอย่ารัน EXPLAIN ANALYZE กับ UPDATE หรือ DELETE บน production เพราะมันแก้ข้อมูลจริง ถ้าจำเป็นให้ครอบด้วย BEGIN แล้ว ROLLBACK

EXPLAIN ANALYZE
SELECT order_id, created_at, status FROM orders
WHERE customer_id = 42 ORDER BY created_at DESC LIMIT 20;

-- ก่อนมี index (ตัวเลขในตัวอย่างนี้สมมติขึ้นเพื่ออธิบาย ไม่ใช่ผลวัดจริง)

          Filter: (customer_id = 42)

-- หลังมี index

สิ่งที่เห็นใน planแปลว่า
Seq Scan / Full Table Scanอ่านทั้งตาราง ไม่ได้แปลว่าผิดเสมอ ถ้าตารางเล็กหรือต้องดึงแถวส่วนใหญ่ การอ่านทั้งตารางจะเร็วกว่า
Index Scanใช้ index หาตำแหน่งแล้วไปอ่านแถวจริงในตาราง
Index Only Scanข้อมูลที่ต้องการอยู่ใน index ครบ ไม่ต้องกลับไปอ่านตาราง (covering index)
Bitmap Heap Scanรวบรวมตำแหน่งจาก index ก่อน แล้วอ่านตารางเรียงตามตำแหน่ง ใช้เมื่อผลลัพธ์มีจำนวนแถวปานกลาง
Nested Loopวนทีละแถวของตารางหนึ่งไปหาคู่ในอีกตาราง เหมาะเมื่อตารางนอกมีแถวน้อยและตารางในมี index
Hash Joinสร้าง hash table จากตารางเล็ก ดีกับการ join ข้อมูลจำนวนมากด้วยเงื่อนไขเท่ากับ
Merge Joinข้อมูลทั้งสองฝั่งเรียงลำดับมาแล้ว จึงไล่จับคู่ไปพร้อมกันได้
rows (ประมาณ) ห่างจาก actual rows มากสถิติของตารางเก่าแล้ว ให้รัน ANALYZE เพื่อให้ optimizer เลือก plan ได้ถูก

หลักการทำ index ที่ใช้ได้จริง

  • Composite index ใช้ได้จากซ้ายไปขวา (leftmost prefix) index (customer_id, created_at) ช่วย query ที่กรอง customer_id อย่างเดียว หรือกรอง customer_id แล้วเรียง created_at แต่ไม่ช่วย query ที่กรองแค่ created_at
  • วางคอลัมน์เงื่อนไขเท่ากับไว้ก่อน แล้วตามด้วยช่วงหรือการเรียง เช่น WHERE status = 'PAID' AND created_at > ... ควรเป็น (status, created_at)
  • เขียนเงื่อนไขให้ใช้ index ได้ (sargable) WHERE YEAR(created_at) = 2026 หรือ WHERE LOWER(email) = ... ทำให้ index ปกติใช้ไม่ได้ ให้เขียนเป็นช่วง created_at >= '2026-01-01' AND created_at < '2027-01-01' หรือสร้าง expression index ON customer (LOWER(email))
  • LIKE 'abc%' ใช้ B-tree ได้ แต่ LIKE '%abc' ใช้ไม่ได้ ต้องใช้ full-text search หรือ trigram index
  • Partial index ทำ index เฉพาะแถวที่ค้นบ่อย เช่น CREATE INDEX ... ON orders (created_at) WHERE status = 'PENDING'
  • Covering index ใส่คอลัมน์ที่ SELECT ไว้ใน index ด้วย INCLUDE (status) จะได้ Index Only Scan
  • คอลัมน์ที่มีค่าซ้ำมาก (selectivity ต่ำ) อย่าง gender หรือ boolean ทำ index เดี่ยว ๆ แทบไม่ช่วยอะไร
  • index มีต้นทุน ทุก INSERT, UPDATE, DELETE ต้องอัปเดต index ด้วย และ index ยังกินพื้นที่ดิสก์กับ RAM ตัวที่ไม่มีใครใช้ควรลบทิ้ง
  • อย่าลืม index ที่คอลัมน์ foreign key เพราะ PostgreSQL ไม่สร้างให้อัตโนมัติ ทำให้ join และการลบแถวแม่ช้า
ชนิด indexใช้เมื่อ
B-treeค่าเริ่มต้น ใช้กับ =, <, >, BETWEEN, ORDER BY
Hashเฉพาะเงื่อนไขเท่ากับ
GINค่าที่มีหลายส่วน เช่น JSONB, array, full-text
GiST / SP-GiSTข้อมูลเชิงพื้นที่ ช่วงเวลา หรือ nearest neighbor
BRINตารางใหญ่มากที่ข้อมูลถูกเขียนเรียงกันอยู่แล้ว เช่น log ที่เรียงตามเวลา ได้ index ขนาดเล็กมาก
Clustered (SQL Server, InnoDB PK)ตัวตารางเรียงตาม index นี้ มีได้ตัวเดียวต่อตาราง

สาเหตุอื่นที่แก้ด้วย index ไม่ได้

  • N+1 query ดึงออร์เดอร์ 50 ใบ แล้ววน query ลูกค้าทีละใบอีก 50 ครั้ง ให้ใช้ JOIN หรือ WHERE customer_id IN (...) ครั้งเดียว
  • SELECT * ดึงคอลัมน์ที่ไม่ใช้ เปลือง network และทำให้ใช้ covering index ไม่ได้
  • OFFSET ในหน้าลึก ๆ OFFSET 100000 ฐานข้อมูลยังต้องอ่านและทิ้ง 100,000 แถว ใช้ keyset pagination แทน WHERE (created_at, order_id) < (:last_created, :last_id) ORDER BY created_at DESC, order_id DESC LIMIT 20

Connection pool

การเปิด connection ใหม่ไปยังฐานข้อมูลมีต้นทุนสูง ต้อง handshake และยืนยันตัวตน และใน PostgreSQL ยังต้องสร้าง process ใหม่ทุกครั้ง แอปจึงใช้ pool ที่เปิด connection ค้างไว้ แล้วให้แต่ละ request ยืมไปใช้และคืนเมื่อเสร็จ ค่าที่ต้องปรับมีไม่กี่ตัว

ค่าความหมายข้อควรระวัง
maximumPoolSizeจำนวน connection สูงสุดของแอปหนึ่งตัวยิ่งมากไม่ได้แปลว่ายิ่งเร็ว ฐานข้อมูลมี CPU และดิสก์จำกัด ถ้า connection มากเกินไปจะแย่งทรัพยากรกันจนช้าลงทั้งระบบ
minimumIdleจำนวนที่เปิดรอไว้ถ้าโหลดค่อนข้างคงที่ ให้ตั้งเท่ากับค่าสูงสุด จะได้ไม่ต้องเปิดใหม่ตอน traffic พุ่ง
connectionTimeoutรอยืม connection ได้นานเท่าไรตั้งให้สั้นพอที่จะแจ้ง error ได้เร็ว ไม่ปล่อยให้ request ค้างนาน
idleTimeout / maxLifetimeปิด connection ที่ไม่ได้ใช้หรือเก่าเกินไปmaxLifetime ต้องสั้นกว่า timeout ของ firewall หรือฐานข้อมูล

สูตรตั้งต้นที่หน้า About Pool Sizing ของ HikariCP อ้างถึงคือ connections = (จำนวน core × 2) + จำนวนดิสก์ที่หมุนพร้อมกัน โดย core คือของเซิร์ฟเวอร์ฐานข้อมูล ถ้ามี 4 core กับดิสก์หนึ่งชุดจะได้ (4 × 2) + 1 = 9 ให้ถือเป็นจุดเริ่มต้นแล้วปรับตามผลวัด อย่าลืมคูณจำนวน instance ของแอป ถ้ามี 20 pod และแต่ละ pod ตั้ง 20 จะได้ 400 connection ซึ่งเกินค่า max_connections เริ่มต้นของ PostgreSQL (100) ไปมาก กรณีนี้ใช้ pooler กลางอย่าง PgBouncer ในโหมด transaction pooling ช่วยได้

อาการเมื่อ connection ใน pool ถูกใช้จนหมดrequest ค้างแล้ว timeout ทั้งที่ CPU ของฐานข้อมูลยังว่าง สาเหตุที่พบบ่อยคือโค้ดยืม connection แล้วไม่คืน (ลืม release ใน finally) หรือถือ transaction ค้างไว้ระหว่างเรียก API ภายนอก ให้เปิด leak detection และอย่าเรียก service อื่นระหว่างที่ transaction เปิดอยู่

  3. ทดสอบกับข้อมูลขนาดใกล้ของจริง เพราะ ALTER ที่ใช้เวลาไม่กี่วินาทีบนเครื่อง dev อาจล็อกตารางนานเป็นนาทีหรือมากกว่านั้นบน production ขึ้นกับขนาดตารางและเวอร์ชันของฐานข้อมูล
  4. แยก migration ที่เปลี่ยนโครงสร้างออกจากการย้ายข้อมูลจำนวนมาก การ backfill ล้านแถวควรทำเป็นรอบเล็ก ๆ
  5. ควรเตรียมแผนถอยกลับไว้ก่อน จะเป็น down migration หรือ forward fix ก็ได้ และต้องรู้ว่า DROP COLUMN ย้อนกลับไม่ได้ถ้าไม่มี backup

กลยุทธ์ฐานข้อมูลระดับองค์กร: Scalability, Availability และ Consistency

Advance

Architect enterprise database strategies for scalability, availability, and consistency

ควรขยายระบบแบบไหนก่อน

Scale up (vertical) คือเพิ่ม CPU, RAM และดิสก์ให้เครื่องเดิม ง่ายที่สุดและไม่ต้องแก้แอป แต่ขยายได้ถึงจุดหนึ่งเท่านั้น และถ้าเครื่องนั้นล่ม ทั้งระบบก็ล่มตาม Scale out (horizontal) คือเพิ่มเครื่อง ทำได้สองทาง คือเพิ่ม read replica ให้ช่วยรับงานอ่าน กับแบ่งข้อมูลไปหลายเครื่อง (sharding) ลำดับที่แนะนำสำหรับระบบส่วนใหญ่คือ tune query และ index ก่อน ตามด้วย scale up เพิ่ม cache เพิ่ม read replica ทำ partition และค่อยพิจารณา sharding เป็นทางสุดท้าย เพราะ sharding เพิ่มความซับซ้อนมากที่สุด

CAP บอกว่าเมื่อเครือข่ายแยกเป็นสองฝั่ง (Partition) ระบบกระจายต้องเลือกระหว่าง Consistency คือทุกคนเห็นค่าล่าสุดตรงกัน หรือ Availability คือทุก request ได้คำตอบ เพราะ partition เลี่ยงไม่ได้ในระบบกระจาย การเลือกจริงจึงอยู่ระหว่าง CP กับ AP ส่วน PACELC ขยายความต่อว่า แม้ไม่มี partition (Else) ก็ยังต้องเลือกระหว่าง Latency กับ Consistency

Advance

Partitioning กับ Sharding ต่างกันตรงไหน

สองคำนี้มักถูกใช้ปนกัน แต่ในข้อสอบแยกความหมายชัดเจน Partitioning คือแบ่งตารางใหญ่เป็นหลายส่วนภายในฐานข้อมูลตัวเดียว แอปยังมองเห็นเป็นตารางเดียว Sharding คือแบ่งข้อมูลไปไว้คนละเซิร์ฟเวอร์ แต่ละเครื่องเก็บข้อมูลเพียงบางส่วน จึงขยายได้ทั้งพื้นที่เก็บและความสามารถในการเขียน แต่แลกกับความซับซ้อนมาก นอกจากนี้ยังมี vertical partitioning คือแยกคอลัมน์ที่ใช้ไม่บ่อยหรือใหญ่ (เช่น รูป หรือ description ยาว) ออกไปอีกตาราง

Partitioning ใน PostgreSQL

CREATE TABLE orders (
  order_id BIGINT, customer_id BIGINT, created_at TIMESTAMPTZ NOT NULL, total NUMERIC(12,2),
  PRIMARY KEY (order_id, created_at)        -- PK ต้องมีคอลัมน์ partition key
) PARTITION BY RANGE (created_at);

CREATE TABLE orders_2026_10 PARTITION OF orders FOR VALUES FROM ('2026-10-01') TO ('2026-11-01');
CREATE TABLE orders_2026_11 PARTITION OF orders FOR VALUES FROM ('2026-11-01') TO ('2026-12-01');

-- query ที่มีเงื่อนไข created_at จะอ่านเฉพาะ partition ที่เกี่ยว (partition pruning)
SELECT SUM(total) FROM orders WHERE created_at >= '2026-11-01' AND created_at < '2026-12-01';

-- ลบข้อมูลเก่าทั้งเดือนในพริบตา แทน DELETE ล้านแถวที่สร้าง WAL มหาศาล
ALTER TABLE orders DETACH PARTITION orders_2024_01;
DROP TABLE orders_2024_01;

Sharding และการเลือก shard key

shard key คือการตัดสินใจที่แก้ทีหลังยากที่สุด key ที่ดีต้องมีค่าหลากหลายพอ (high cardinality) กระจายโหลดเท่ากัน และทำให้ query ส่วนใหญ่ไปถึง shard เดียว

tenant_id สำหรับ SaaSดี ถ้าแต่ละ tenant มีขนาดใกล้กัน ถ้ามีลูกค้ารายใหญ่มาก ต้องแยก tenant นั้นไปไว้ shard ของตัวเอง
  • Hash sharding กระจายดี แต่ query เป็นช่วงต้องถามทุก shard
  • Range sharding query ช่วงดี แต่เสี่ยง hot spot
  • Directory / lookup มีตารางบอกว่าคีย์ไหนอยู่ shard ไหน ยืดหยุ่น แต่ตาราง lookup จะกลายเป็นจุดสำคัญที่ล่มไม่ได้
  • Consistent hashing เพิ่มหรือลด node แล้วย้ายข้อมูลแค่ประมาณ 1/N แทนที่จะย้ายเกือบทั้งหมดแบบ hash % N

ผลข้างเคียงที่ต้องวางแผนคือ join และ transaction ข้าม shard ทำได้ยากหรือช้า, unique constraint ข้าม shard ทำไม่ได้โดยตรง (ใช้ ID แบบ UUID หรือ Snowflake แทน auto increment), การ resharding เมื่อ shard เต็ม และการ backup ทุก shard ให้ได้จุดเวลาเดียวกัน เครื่องมือที่ช่วยได้ ได้แก่ Citus สำหรับ PostgreSQL, Vitess สำหรับ MySQL และ MongoDB ที่มี sharding ในตัว

ETL (Extract, Transform, Load) แปลงข้อมูลก่อนโหลดเข้า warehouse เหมาะเมื่อพื้นที่ใน warehouse มีราคาแพง หรือต้องกรองข้อมูลอ่อนไหวออกก่อน ส่วน ELT (Extract, Load, Transform) โหลดข้อมูลดิบเข้าไปก่อน แล้วใช้กำลังประมวลผลของ cloud warehouse แปลงข้อมูลข้างใน มักใช้ dbt เขียน transformation เป็น SQL ที่มี test และ version control ส่วนการจัดลำดับงานใช้ Airflow, Dagster หรือบริการของ cloud สำหรับข้อมูลที่ต้องสดระดับวินาที ใช้ streaming ผ่าน Kafka ร่วมกับ CDC

แหล่งข้อมูล (OLTP, SaaS, log)
   │  ingest: CDC (Debezium) / batch (Fivetran, Airbyte) / stream (Kafka)
   ▼
Object storage: Bronze → Silver → Gold   (Iceberg / Delta, Parquet)
   │  transform: Spark / dbt     orchestrate: Airflow
   ▼
Warehouse / SQL engine → BI (Power BI, Tableau, Looker) · ML · Reverse ETL
   └──── catalog, lineage, access policy, data quality (governance ครอบทุกชั้น)

โจทย์ 1 (GROUP BY + HAVING) หาลูกค้าที่มีออร์เดอร์สถานะ PAID อย่างน้อย 2 ใบ และยอดซื้อรวมของออร์เดอร์ PAID ตั้งแต่ 1,000 ขึ้นไป แสดง customer_id, จำนวนออร์เดอร์, ยอดรวม เรียงยอดรวมจากมากไปน้อย 
SELECT o.customer_id,
       COUNT(DISTINCT o.order_id)              AS paid_orders,
       SUM(oi.quantity * oi.unit_price)        AS total_spent
FROM orders o
JOIN order_item oi ON oi.order_id = o.order_id
WHERE o.status = 'PAID'                         -- กรองแถวก่อนจัดกลุ่ม
GROUP BY o.customer_id
HAVING COUNT(DISTINCT o.order_id) >= 2         -- กรองกลุ่มหลังรวม
   AND SUM(oi.quantity * oi.unit_price) >= 1000
ORDER BY total_spent DESC, o.customer_id;

ผลลัพธ์: (1, 3, 2820) และ (3, 2, 1630)

จุดที่คนพลาดบ่อยคือใช้ COUNT(*) แทน COUNT(DISTINCT o.order_id) หลัง join กับ order_item แต่ละออร์เดอร์จะกลายเป็นหลายแถวตามจำนวนรายการ ลูกค้า 1 จะนับได้ 4 แทนที่จะเป็น 3 อีกเรื่องคือ WHERE ใช้กับฟังก์ชันรวมไม่ได้ ต้องใช้ HAVING

โจทย์ 2 (ค่าสูงสุดอันดับสอง) หาราคาสินค้าที่สูงเป็นอันดับสองแบบไม่นับค่าซ้ำ ถ้าไม่มีให้แสดง NULL 
-- วิธีที่ 1: ใช้ได้ทุก dialect
SELECT MAX(price) AS second_highest
FROM product
WHERE price < (SELECT MAX(price) FROM product);

-- วิธีที่ 2: ครอบด้วย subquery เพื่อให้ได้ NULL เมื่อไม่มีแถว
SELECT (SELECT DISTINCT price FROM product ORDER BY price DESC LIMIT 1 OFFSET 1) AS second_highest;

-- วิธีที่ 3: ขยายเป็นอันดับที่ N ได้
SELECT DISTINCT price FROM (
  SELECT price, DENSE_RANK() OVER (ORDER BY price DESC) AS rnk FROM product
) t WHERE rnk = 2;

ผลลัพธ์: 890

ราคา 890 มีสองชิ้น (Hoodie กับ Lamp) ถ้าใช้ ORDER BY price DESC LIMIT 1 OFFSET 1 โดยไม่มี DISTINCT ก็ได้ 890 เหมือนกัน แต่เป็นความบังเอิญ แต่ถ้าค่าสูงสุดซ้ำกันจะได้คำตอบผิด วิธีที่ 3 ถ้าไม่มีอันดับสองจะได้ผลว่าง ไม่ใช่ NULL ต้องดูว่าโจทย์ต้องการแบบไหน

โจทย์ 3 (Top N ต่อกลุ่ม) หาสินค้าที่ทำรายได้สูงสุด 2 อันดับแรกในแต่ละหมวด นับเฉพาะออร์เดอร์ PAID ถ้ารายได้เท่ากันให้ได้อันดับเดียวกัน 
WITH rev AS (
  SELECT p.category, p.name, SUM(oi.quantity * oi.unit_price) AS revenue
  FROM order_item oi
  JOIN orders  o ON o.order_id = oi.order_id AND o.status = 'PAID'
  JOIN product p ON p.product_id = oi.product_id
  GROUP BY p.category, p.name
), ranked AS (
  SELECT *, DENSE_RANK() OVER (PARTITION BY category ORDER BY revenue DESC) AS rnk
  FROM rev
)
SELECT category, name, revenue, rnk
FROM ranked
WHERE rnk <= 2                              -- ใส่ใน WHERE ของ SELECT เดียวกับ window ไม่ได้
ORDER BY category, rnk, name;

ผลลัพธ์: Apparel Hoodie 1780 (1), Apparel Tee 870 (2), Home Rug 1200 (1), Home Lamp 890 (2)

ฟังก์ชันค่าที่เท่ากัน 100, 90, 90, 80
ROW_NUMBER()1, 2, 3, 4 ไม่สนว่าเท่ากัน
RANK()1, 2, 2, 4 ข้ามเลข
DENSE_RANK()1, 2, 2, 3 ไม่ข้ามเลข

สังเกตว่าเงื่อนไข o.status = 'PAID' อยู่ใน ON ของ inner join ซึ่งให้ผลเหมือนใส่ใน WHERE แต่ถ้าเป็น LEFT JOIN สองแบบนี้ให้ผลต่างกัน (ดูโจทย์ 10)

โจทย์ 4 (Anti-join) หาลูกค้าที่ไม่เคยสั่งซื้อเลย และอธิบายว่าทำไม NOT IN อาจได้ผลว่าง 
-- แบบ LEFT JOIN
SELECT c.customer_id, c.full_name
FROM customer c
LEFT JOIN orders o ON o.customer_id = c.customer_id
WHERE o.order_id IS NULL
ORDER BY c.customer_id;

-- แบบ NOT EXISTS (ปลอดภัยกับ NULL และอ่านเจตนาได้ชัด)
SELECT c.customer_id, c.full_name
FROM customer c
WHERE NOT EXISTS (SELECT 1 FROM orders o WHERE o.customer_id = c.customer_id);

ผลลัพธ์: (4, Ann2) และ (5, Eve)

ถ้าเขียน WHERE customer_id NOT IN (SELECT customer_id FROM blacklist) แล้วใน blacklist มีแถวที่เป็น NULL แม้แถวเดียว ผลลัพธ์จะว่างทั้งหมด เพราะ x NOT IN (2, NULL) เท่ากับ x <> 2 AND x <> NULL และการเทียบกับ NULL ได้ UNKNOWN ไม่ใช่ TRUE ลองกับข้อมูลชุดนี้แล้ว NOT IN ได้ 0 แถว ส่วน NOT EXISTS ได้ 4 แถวตามที่ควร

โจทย์ 5 (Running total) แสดงรายได้รายวันของออร์เดอร์ PAID พร้อมยอดสะสมตั้งแต่วันแรก 
WITH daily AS (
  SELECT o.created_at AS day, SUM(oi.quantity * oi.unit_price) AS revenue
  FROM orders o JOIN order_item oi ON oi.order_id = o.order_id
  WHERE o.status = 'PAID'
  GROUP BY o.created_at
)
SELECT day, revenue,
       SUM(revenue) OVER (ORDER BY day
                          ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS running_total
FROM daily
ORDER BY day;

ผลลัพธ์: 09-01 730/730, 09-02 890/1620, 09-03 2590/4210, 10-06 290/4500, 10-20 1340/5840

ถ้าไม่ระบุ frame ค่าเริ่มต้นเมื่อมี ORDER BY คือ RANGE BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW ซึ่งรวมทุกแถวที่มีค่า ORDER BY เท่ากันเข้าไปพร้อมกัน ถ้ามีวันที่ซ้ำกัน ยอดสะสมจะกระโดดข้ามไป การเขียน ROWS ให้ชัดจึงปลอดภัยกว่า

โจทย์ 6 (Pivot ด้วย CASE) นับจำนวนออร์เดอร์แต่ละสถานะในแต่ละเดือน ให้สถานะเป็นคอลัมน์ 
SELECT DATE_FORMAT(created_at, '%Y-%m') AS month,      -- PostgreSQL: to_char(created_at, 'YYYY-MM')
       SUM(CASE WHEN status = 'PAID'      THEN 1 ELSE 0 END) AS paid,
       SUM(CASE WHEN status = 'PENDING'   THEN 1 ELSE 0 END) AS pending,
       SUM(CASE WHEN status = 'CANCELLED' THEN 1 ELSE 0 END) AS cancelled
FROM orders
GROUP BY DATE_FORMAT(created_at, '%Y-%m')
ORDER BY month;

ผลลัพธ์: 2026-09 → 4, 0, 0 และ 2026-10 → 2, 1, 1

รูปแบบนี้เรียกว่า conditional aggregation ใช้ได้ทุก dialect PostgreSQL ยังเขียนสั้นลงได้ด้วย COUNT(*) FILTER (WHERE status = 'PAID') โจทย์ฝึกบน HackerRank ที่ให้จัดรายชื่อคนตามอาชีพเป็นคอลัมน์ก็ใช้หลักเดียวกัน แต่ต้องใส่ ROW_NUMBER ก่อนเพื่อให้แต่ละแถวจับคู่กันได้

โจทย์ 7 (ข้อมูลซ้ำ) หาอีเมลที่ซ้ำกัน แล้วเขียนคำสั่งลบแถวซ้ำโดยเก็บ customer_id ที่น้อยที่สุดไว้ 
SELECT email, COUNT(*) AS cnt
FROM customer GROUP BY email HAVING COUNT(*) > 1;           -- a@x.com 2

-- ดูก่อนว่าจะลบแถวไหน
SELECT customer_id, email FROM (
  SELECT customer_id, email,
         ROW_NUMBER() OVER (PARTITION BY email ORDER BY customer_id) AS rn
  FROM customer
) t WHERE rn > 1;                                               -- 4 a@x.com

-- ลบ (PostgreSQL / SQL Server / MySQL 8 ใช้ได้)
DELETE FROM customer
WHERE customer_id IN (
  SELECT customer_id FROM (
    SELECT customer_id, ROW_NUMBER() OVER (PARTITION BY email ORDER BY customer_id) AS rn
    FROM customer
  ) t WHERE rn > 1
);
ALTER TABLE customer ADD CONSTRAINT uq_customer_email UNIQUE (email);   -- กันไม่ให้เกิดอีก

ในงานจริงต้องย้ายออร์เดอร์ของลูกค้าที่ซ้ำไปผูกกับรหัสที่เก็บไว้ก่อนลบ ไม่อย่างนั้น foreign key จะไม่ยอมให้ลบ หรือถ้ามี cascade ประวัติการซื้อจะหายไปด้วย

โจทย์ 8 (LAG) หารายได้รายเดือนของออร์เดอร์ PAID และอัตราการเติบโตเทียบเดือนก่อนเป็นเปอร์เซ็นต์ ทศนิยม 2 ตำแหน่ง 
WITH m AS (
  SELECT DATE_FORMAT(o.created_at, '%Y-%m') AS month,
         SUM(oi.quantity * oi.unit_price)  AS revenue
  FROM orders o JOIN order_item oi ON oi.order_id = o.order_id
  WHERE o.status = 'PAID'
  GROUP BY DATE_FORMAT(o.created_at, '%Y-%m')
)
SELECT month, revenue,
       LAG(revenue) OVER (ORDER BY month) AS prev_revenue,
       ROUND(100.0 * (revenue - LAG(revenue) OVER (ORDER BY month))
             / NULLIF(LAG(revenue) OVER (ORDER BY month), 0), 2) AS growth_pct
FROM m ORDER BY month;

ผลลัพธ์: 2026-09 4210, NULL, NULL และ 2026-10 1630, 4210, -61.28

ใช้ 100.0 เพื่อบังคับให้เป็นการหารทศนิยม และ NULLIF(..., 0) กันการหารด้วยศูนย์ ส่วน LEAD() ทำงานกลับกัน คือดึงค่าจากแถวถัดไป

โจทย์ 9 (Gaps and islands) หาลูกค้าที่สั่งซื้อติดต่อกันอย่างน้อย 3 วัน แสดงวันเริ่ม วันสุดท้าย และจำนวนวัน 
WITH d AS (
  SELECT DISTINCT customer_id, CAST(created_at AS DATE) AS day FROM orders
), g AS (
  SELECT customer_id, day,
         -- วันที่ต่อเนื่องกัน ลบด้วยลำดับแล้วจะได้ค่าเดียวกัน
         DATE_SUB(day, INTERVAL ROW_NUMBER() OVER (PARTITION BY customer_id ORDER BY day) DAY) AS grp
         -- PostgreSQL: day - (ROW_NUMBER() OVER (...))::int
  FROM d
)
SELECT customer_id, MIN(day) AS start_day, MAX(day) AS end_day, COUNT(*) AS streak
FROM g
GROUP BY customer_id, grp
HAVING COUNT(*) >= 3;

ผลลัพธ์: (1, 2026-09-01, 2026-09-03, 3)

หลักคิดคือถ้าวันที่เรียงติดกัน ค่า “วันที่ ลบ ลำดับที่” จะเท่ากันตลอดช่วงนั้น เมื่อวันที่ขาดช่วง ค่านี้จะเปลี่ยนไป จึงใช้ค่านี้จัดกลุ่มได้ เทคนิคนี้ใช้กับโจทย์ล็อกอินติดต่อกัน หรือหาช่วงเวลาที่ระบบล่มต่อเนื่องได้ด้วย

โจทย์ 10 (CASE + LEFT JOIN) จัดระดับลูกค้าทุกคนตามยอด PAID: ตั้งแต่ 2,000 เป็น Gold ตั้งแต่ 1,000 เป็น Silver ที่เหลือเป็น Bronze รวมคนที่ไม่เคยซื้อด้วย 
SELECT c.customer_id,
       COALESCE(SUM(oi.quantity * oi.unit_price), 0) AS spent,
       CASE
         WHEN COALESCE(SUM(oi.quantity * oi.unit_price), 0) >= 2000 THEN 'Gold'
         WHEN COALESCE(SUM(oi.quantity * oi.unit_price), 0) >= 1000 THEN 'Silver'
         ELSE 'Bronze'                         -- CASE ตรวจจากบนลงล่าง เจอข้อแรกที่จริงก็หยุด
       END AS tier
FROM customer c
LEFT JOIN orders o      ON o.customer_id = c.customer_id AND o.status = 'PAID'   -- เงื่อนไขอยู่ใน ON
LEFT JOIN order_item oi ON oi.order_id = o.order_id
GROUP BY c.customer_id
ORDER BY spent DESC, c.customer_id;

ผลลัพธ์: 1 2820 Gold · 3 1630 Silver · 2 1390 Silver · 4 0 Bronze · 5 0 Bronze

D1. จากตาราง employee(id, name, department_id, salary) และ department(id, name) หาพนักงานที่ได้เงินเดือนสูงกว่าค่าเฉลี่ยของแผนกตัวเอง แสดงชื่อแผนก ชื่อพนักงาน เงินเดือน เรียงตามแผนกแล้วเงินเดือนจากมากไปน้อย 
SELECT d.name AS department, e.name, e.salary
FROM employee e
JOIN department d ON d.id = e.department_id
JOIN (SELECT department_id, AVG(salary) AS avg_sal
      FROM employee GROUP BY department_id) a
  ON a.department_id = e.department_id
WHERE e.salary > a.avg_sal
ORDER BY d.name, e.salary DESC;

-- หรือใช้ window function
SELECT department, name, salary FROM (
  SELECT d.name AS department, e.name, e.salary,
         AVG(e.salary) OVER (PARTITION BY e.department_id) AS avg_sal
  FROM employee e JOIN department d ON d.id = e.department_id
) t WHERE salary > avg_sal
ORDER BY department, salary DESC;
D2. ตาราง login(user_id, login_at) หาจำนวนผู้ใช้ที่กลับมาใช้งานภายใน 7 วันหลังการล็อกอินครั้งแรก (7-day retention) ต่อวันที่สมัคร 
WITH first_login AS (
  SELECT user_id, MIN(CAST(login_at AS DATE)) AS first_day
  FROM login GROUP BY user_id
)
SELECT f.first_day,
       COUNT(DISTINCT f.user_id) AS new_users,
       COUNT(DISTINCT CASE WHEN CAST(l.login_at AS DATE) > f.first_day
                            AND CAST(l.login_at AS DATE) <= f.first_day + INTERVAL '7' DAY
                           THEN f.user_id END) AS retained_7d
FROM first_login f
LEFT JOIN login l ON l.user_id = f.user_id
GROUP BY f.first_day
ORDER BY f.first_day;

จุดสำคัญคือ LEFT JOIN เพื่อไม่ทิ้งผู้ใช้ที่ไม่กลับมา และ COUNT(DISTINCT CASE ... END) ที่นับเฉพาะคนที่เข้าเงื่อนไข เพราะ CASE ที่ไม่มี ELSE ให้ค่า NULL ซึ่ง COUNT ไม่นับ (MySQL เขียนเป็น DATE_ADD(f.first_day, INTERVAL 7 DAY))

ชุดนี้สุ่ม 40 ข้อจากคลังทั้งหมด ทั้งแนวข้อสอบท้ายหัวข้อ โจทย์สถานการณ์ และข้อสอบสไตล์ Deloitte / HackerRank ตัวเลือกจะสลับใหม่ทุกครั้ง ข้อที่เลือกคำตอบเดียวจะเฉลยทันทีที่กดเลือก ส่วนข้อที่มีป้าย “เลือกหลายข้อ” ให้เลือกให้ครบแล้วกด “ตรวจข้อนี้” ต้องได้ครบทุกข้อที่ถูกและไม่มีข้อผิดปนจึงได้คะแนน ทุกข้อมีป้ายระดับ และเมื่อกด “ดูสรุปผล” จะแยกคะแนนตามระดับให้ด้วย

อ่านทวนสั้น ๆ ก่อนลงมือ

BeginnerKey-value เหมาะกับ session, cache และตัวนับ ส่วน document เหมาะกับข้อมูลที่โครงสร้างไม่ตายตัว ข้อมูลที่อ่านพร้อมกันและมีจำนวนจำกัดให้ฝังไว้ด้วยกัน ข้อมูลที่เพิ่มขึ้นไม่จำกัดให้แยกออก ความสัมพันธ์ M:N ต้องมีตารางกลาง เงินให้ใช้ DECIMAL RPO คือข้อมูลที่ยอมให้หาย RTO คือเวลาที่ยอมให้ระบบหยุด replica ใช้แทน backup ไม่ได้ ส่วน ACID คือ ทำครบหรือไม่ทำเลย ข้อมูลถูกตามกฎ transaction ไม่รบกวนกัน และ COMMIT แล้วข้อมูลไม่หาย
Intermediate2NF ตัด partial dependency ออก 3NF ตัด transitive dependency ออก และ BCNF กำหนดว่าทุก determinant ต้องเป็นคีย์ การเขียนสองระบบตรง ๆ (dual write) ไม่ปลอดภัย ให้ใช้ outbox หรือ CDC ใน cache-aside ให้ลบ key ตอนแก้ข้อมูล composite index ใช้ได้จากคอลัมน์ซ้ายสุด การครอบคอลัมน์ด้วยฟังก์ชันทำให้ใช้ index ไม่ได้ pool ใหญ่ไม่ได้แปลว่าเร็ว ไฟล์ migration ที่รันแล้วห้ามแก้ และการเปลี่ยน schema โดยไม่ปิดระบบใช้วิธี expand and contract
Advancetune query ก่อนเพิ่มเครื่อง แล้วจึงใช้ replica และ partition ส่วน sharding เป็นทางเลือกสุดท้าย read-your-writes แก้ปัญหา replication lag ถ้า R + W > N การอ่านจะได้ค่าล่าสุด shard key ที่ดีต้องกระจายโหลดเท่ากันและให้ query ส่วนใหญ่ไปถึง shard เดียว cache ต้องระวัง stampede, penetration และ avalanche ในด้าน governance owner เป็นผู้อนุมัติ steward ดูแลคุณภาพ custodian ดูแลด้านเทคนิค และ PDPA กำหนดให้แจ้งเหตุภายใน 72 ชั่วโมง ใน warehouse fact เก็บตัวเลข dimension เก็บบริบท SCD Type 2 เก็บประวัติ และ lakehouse คือ data lake ที่มี ACID
